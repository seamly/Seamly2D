SKYLINE BOTTOM-LEFT PACKING ALGORITHM
=====================================

DATA STRUCTURES:
----------------
SkylineSegment {
    xStart: double    // Left X coordinate of segment
    xEnd: double      // Right X coordinate of segment
    y: double         // Y height of this segment (top of placed items)
}

PlacementRecord {
    pieceId: string
    pieceX: double
    pieceY: double
    pieceWidth: double
    pieceHeight: double
}


ALGORITHM:
----------

FUNCTION SkylinePacking(layoutPieces, minColX, maxColX, minRowY, maxRowY):
    
    // Initialize skyline with a single segment covering the entire top area
    skyline = [SkylineSegment(minColX, maxColX, minRowY)]
    placementHistory = []
    gapPx = 10.0
    placedCount = 0
    
    // Process each piece in order
    FOR EACH piece IN layoutPieces:
        
        width = piece.bounds.width()
        height = piece.bounds.height()
        
        // Skip invalid pieces
        IF width <= 0 OR height <= 0:
            CONTINUE
        END IF
        
        // ============================================================
        // PHASE 1: FIND BEST POSITION (Bottom-Left Heuristic)
        // ============================================================
        
        bestX = -1
        bestY = INFINITY
        bestStartIdx = -1
        bestEndIdx = -1
        
        // Try placing at each segment's left edge
        FOR startIdx = 0 TO skyline.size() - 1:
            
            tryX = skyline[startIdx].xStart
            
            // Check if piece would extend beyond right boundary
            IF tryX + width + gapPx > maxColX:
                CONTINUE
            END IF
            
            // Find maximum Y among all segments the piece would span
            maxY = 0.0
            endIdx = startIdx
            coveredX = skyline[startIdx].xStart
            
            // Find all segments that the piece would cover horizontally
            WHILE endIdx < skyline.size() AND coveredX < tryX + width:
                
                seg = skyline[endIdx]
                
                // Check if segment overlaps with piece position
                IF seg.xStart < tryX + width AND seg.xEnd > tryX:
                    IF seg.y > maxY:
                        maxY = seg.y
                    END IF
                    coveredX = seg.xEnd
                END IF
                
                IF coveredX >= tryX + width:
                    BREAK
                END IF
                
                endIdx = endIdx + 1
            END WHILE
            
            // Check if piece fits vertically
            IF maxY + height + gapPx > maxRowY:
                CONTINUE  // Piece would extend beyond bottom boundary
            END IF
            
            // Check if this is a better position (lower Y, then lower X)
            IF maxY < bestY OR (maxY == bestY AND tryX < bestX):
                bestX = tryX
                bestY = maxY
                bestStartIdx = startIdx
                bestEndIdx = endIdx
            END IF
            
        END FOR
        
        // ============================================================
        // PHASE 2: PLACE THE PIECE
        // ============================================================
        
        // If no valid position found, skip this piece
        IF bestStartIdx < 0:
            LOG "No position found for piece"
            CONTINUE
        END IF
        
        // Place the piece at the best position (add gap above previous items)
        placeX = bestX
        placeY = bestY + gapPx
        
        APPLY_TRANSFORM(piece, placeX, placeY)
        placedCount = placedCount + 1
        
        // Record placement for debug output
        placementHistory.APPEND(PlacementRecord(piece.id, placeX, placeY, width, height))
        
        // ============================================================
        // PHASE 3: UPDATE THE SKYLINE
        // ============================================================
        
        // Calculate piece boundaries
        newY = placeY + height  // Top of the placed piece
        pieceLeft = placeX
        pieceRight = placeX + width
        
        // Build new skyline by modifying segments affected by the placed piece
        newSkyline = []
        
        FOR i = 0 TO skyline.size() - 1:
            
            seg = skyline[i]
            
            // Case 1: Segment is entirely to the left of the piece
            IF seg.xEnd <= pieceLeft:
                newSkyline.APPEND(seg)
                CONTINUE
            END IF
            
            // Case 2: Segment is entirely to the right of the piece
            IF seg.xStart >= pieceRight:
                newSkyline.APPEND(seg)
                CONTINUE
            END IF
            
            // Case 3: Segment overlaps with the piece - need to split/modify
            
            // Add left portion (if any) before the piece
            IF seg.xStart < pieceLeft:
                newSkyline.APPEND(SkylineSegment(seg.xStart, pieceLeft, seg.y))
            END IF
            
            // Add the piece's segment (only once, avoid duplicates)
            pieceSegmentAdded = FALSE
            FOR EACH existing IN newSkyline:
                IF existing.xStart == pieceLeft AND 
                   existing.xEnd == pieceRight AND 
                   existing.y == newY:
                    pieceSegmentAdded = TRUE
                    BREAK
                END IF
            END FOR
            
            IF NOT pieceSegmentAdded:
                newSkyline.APPEND(SkylineSegment(pieceLeft, pieceRight, newY))
            END IF
            
            // Add right portion (if any) after the piece
            IF seg.xEnd > pieceRight:
                newSkyline.APPEND(SkylineSegment(pieceRight, seg.xEnd, seg.y))
            END IF
            
        END FOR
        
        // ============================================================
        // PHASE 4: MERGE ADJACENT SEGMENTS WITH SAME Y HEIGHT
        // ============================================================
        
        mergedSkyline = []
        
        FOR EACH seg IN newSkyline:
            IF mergedSkyline IS EMPTY:
                mergedSkyline.APPEND(seg)
            ELSE:
                last = mergedSkyline.LAST()
                // Merge if adjacent and same height
                IF last.xEnd == seg.xStart AND last.y == seg.y:
                    last.xEnd = seg.xEnd  // Extend previous segment
                ELSE:
                    mergedSkyline.APPEND(seg)
                END IF
            END IF
        END FOR
        
        skyline = mergedSkyline
        
        // Sort skyline by xStart for consistent processing
        SORT skyline BY xStart ASCENDING
        
    END FOR
    
    // ============================================================
    // PHASE 5: DEBUG OUTPUT (optional)
    // ============================================================
    
    WRITE placementHistory TO debug file
    WRITE final skyline segments TO debug file
    
    RETURN placedCount
    
END FUNCTION


NOTES:
------
- The algorithm uses a Bottom-Left heuristic: prefers lowest Y position, 
  then leftmost X position
- A 10px gap is maintained between pieces and boundaries
- The skyline efficiently tracks the top contour of placed items
- Segments are merged when adjacent and at the same height to keep 
  the skyline compact
- Time complexity: O(n * m) where n = number of pieces, 
  m = average number of skyline segments
