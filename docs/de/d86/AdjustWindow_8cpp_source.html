57/// @brief Reload the scene with a new SVG and bbox JSON (reuses this window).
58void AdjustWindow::reload(const QString& svgPath, const QString& bboxJson)
59{
60 qDebug() << "[AdjustWindow] reload() — reloading scene from" << svgPath;
61
62 // Capture current viewport state before reloading so piece operations do not
63 // force a zoom reset. This preserves the user's current zoom/pan context.
64 const QTransform previousViewTransform = m_view->transform();
65 const QPointF previousViewCenter = m_view->mapToScene(m_view->viewport()->rect().center());
66
67 // Reset flag so the next close (Save/Cancel/×) is handled correctly.
68 m_closeHandled = false;
69 // Load the new layout into the scene; this clears all existing items and creates new ones.
70 m_scene->loadLayout(svgPath, bboxJson);
71 // Dump overlay state after reload for debugging.
72 m_scene->dumpOverlayData();
73
74 // Restore the previous viewport state after reload so zoom level is maintained
75 // while users iteratively apply operations to pieces.
76 m_view->setTransform(previousViewTransform);
77 m_view->centerOn(previousViewCenter);
78
79 updateUndoRedoActions(m_scene->canUndo(), m_scene->canRedo());
80} // reload

86/// @brief Collect piece transforms, clear overlays, and emit accepted().
87///
88/// Piece overlays are cleared immediately so the canvas shows only the
89/// background SVG while the async reload (triggered by adjustApplied in QML)
90/// is in flight. The reload re-adds PieceOverlayItems at their new positions.
91void AdjustWindow::onApplyClicked()
92{
93 qDebug() << "[AdjustWindow] onApplyClicked: 1 Apply triggered (button or Enter)";
94
95
96
97 // Get the transform JSON string for the overlay piece that has moved or rotated, e.g. [{"id":"piece1","transform":"translate(10.0000 5.0000) rotate(15.0000 50.0000 30.0000)"},...]
98 qDebug() << "[AdjustWindow] onApplyClicked: 2 calling getMovedTransform()";
99 const QString transform_str = m_scene->getMovedTransform();
100
101 // Dump overlay state before applying transforms for debugging.
102 m_scene->dumpOverlayData();
103
104 // Clear piece overlays immediately so the user sees the background SVG while the async reload is in flight.
105 qDebug() << "[AdjustWindow] onApplyClicked: 3 clearing overlay data";
106 m_scene->clearPieces();
107
108 // Emit the accepted() signal with the transforms JSON, which QML forwards as applyRequested() to the AdjustController, which forwards it to the main application's appController, which applies the transforms to the layout_dom and triggers the right canvas reload.
109 qDebug() << "[AdjustWindow] onApplyClicked: 4 emitting accepted() with transforms:" << transform_str;
110 // signal the transforms to QML, which forwards to the main application controller
111 emit accepted(transform_str);
112} // onApplyClicked

125} // onSaveClicked

163/// @brief Emit cancelled() and close the window.
164void AdjustWindow::onCancelClicked()
165{
166 m_closeHandled = true;
167 // Emit the cancelled() signal, which QML forwards as cancelRequested() to the AdjustController, which forwards it to the main application'sappController, which discards any pending transforms and triggers a reload of the original layout in right canvas of the main application window.
168 emit cancelled();
169 close();
170} // onCancelClicked

172/// @brief Treat OS × close as Save so exitAdjustMode() is always called.
173///
174/// Without this, closing the window via the title-bar × button leaves
175/// isAdjustMode = true and all toolbar buttons permanently disabled.
176/// The close is deferred — QML will call closeWindow() after the right
177/// canvas finishes loading the updated SVG.
178void AdjustWindow::closeEvent(QCloseEvent* event)
179{
180 if (!m_closeHandled) {
181 // OS × button — treat as Save: defer close until canvas reloads.
182 m_closeHandled = true;
183 emit abandoned();
184 event->accept();
185 return;
186 } // if close not already handled by a button
187
188 QMainWindow::closeEvent(event);
189} // closeEvent

406/// @brief Intercept wheel events on the view viewport to zoom instead of scroll.
407bool AdjustWindow::eventFilter(QObject* watched, QEvent* event)
408{
409 if (watched == m_view->viewport() && event->type() == QEvent::Wheel) {
410 QWheelEvent* wheel = static_cast<QWheelEvent*>(event);
411
412 if (wheel->angleDelta().y() > 0) {
413 // Scroll up — zoom in.
414 zoomIn();
415 } else {
416 // Scroll down — zoom out.
417 zoomOut();
418 } // if scroll direction
419
420 return true; // event consumed; do not scroll
421 } // if wheel event on viewport
422
423 return QMainWindow::eventFilter(watched, event);
424} // eventFilter
