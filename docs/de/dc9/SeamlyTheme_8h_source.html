5// @file SeamlyTheme.h
6// @brief Header-only Seamly branding palette constants and QPalette factory.
7//
8// Mirrors the colours defined in Theme.qml exactly so that QtWidgets windows
9// (e.g. AdjustWindow) match the QML frontend visually.
10//
11// Usage:
12// widget->setPalette(SeamlyTheme::makeSeamlyPalette());
13// widget->setStyle(QStyleFactory::create("Fusion"));
14//
15// Do NOT call QApplication::setPalette() globally — that would affect QML's
16// native-control rendering.
17

62/// @brief Build a QPalette that applies the Seamly branding to a QtWidgets widget.
63///
64/// Apply this palette to individual windows/widgets only:
65/// @code
66/// window->setPalette(SeamlyTheme::makeSeamlyPalette());
67/// window->setStyle(QStyleFactory::create("Fusion"));
68/// @endcode
69///
70/// @return Configured QPalette instance.
71inline QPalette makeSeamlyPalette()
72{
73 QPalette p;
74
75 // Window chrome and general backgrounds.
76 p.setColor(QPalette::Window, SEAMLY_VIOLET_DARK);
77 p.setColor(QPalette::WindowText, SEAMLY_GRAY_LIGHT);
78
79 // Text-entry and list backgrounds.
80 p.setColor(QPalette::Base, SEAMLY_VIOLET);
81 p.setColor(QPalette::AlternateBase, SEAMLY_VIOLET_MEDIUM);
82 p.setColor(QPalette::Text, SEAMLY_GRAY_LIGHT);
83
84 // Push-buttons and tool-buttons.
85 p.setColor(QPalette::Button, SEAMLY_VIOLET_MEDIUM);
86 p.setColor(QPalette::ButtonText, SEAMLY_GRAY_LIGHT);
87
88 // Selection.
89 p.setColor(QPalette::Highlight, SEAMLY_VIOLET_LIGHT);
90 p.setColor(QPalette::HighlightedText, SEAMLY_GRAY_LIGHT);
91
92 // Tooltips.
93 p.setColor(QPalette::ToolTipBase, SEAMLY_VIOLET_DARK);
94 p.setColor(QPalette::ToolTipText, SEAMLY_GRAY_LIGHT);
95
96 // Disabled state.
97 p.setColor(QPalette::Disabled, QPalette::Button, SEAMLY_GRAY_MEDIUM);
98 p.setColor(QPalette::Disabled, QPalette::ButtonText, SEAMLY_GRAY_LIGHT);
99 p.setColor(QPalette::Disabled, QPalette::Text, SEAMLY_GRAY_MEDIUM);
100 p.setColor(QPalette::Disabled, QPalette::WindowText, SEAMLY_GRAY_MEDIUM);
101
102 return p;
103} // makeSeamlyPalette
104
105} // namespace SeamlyTheme
