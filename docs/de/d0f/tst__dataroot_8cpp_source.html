79 * @brief initTestCase creates the scratch directory every case works inside, and redirects
80 * the QSettings base directory so the developer's real settings file is never touched.

83 * therefore never build a path from QDir::homePath() — see the class documentation.

85void TST_DataRoot::initTestCase()
86{

102 // QSettings::setPath() — it lives under GenericConfigLocation, which cannot be

310 * @brief DefaultDataRootIsSeamlyUnderDocuments checks the built-in default.
311 *
312 * The lineage: ~/seamly2d (original) → ~/seamly (Task 34) → ~/seamlyData (Task 53) →
313 * <Documents>/Seamly (Task 60). The last move is the one with a principle behind it —
314 * these are documents the user creates, opens and backs up, so they belong where every
315 * other application puts documents, while internal state stays in the platform's
316 * application-data locations.
317 *
318 * The expected value is built from QStandardPaths rather than hard-coded, deliberately:
319 * hard-coding "Documents" would pass on this machine and fail on a localized Linux system
320 * or a redirected Windows profile — the very cases DocumentsLocation exists to handle.
321 */
322void TST_DataRoot::DefaultDataRootIsSeamlyUnderDocuments() const
323{
324 QString documents = QStandardPaths::writableLocation(QStandardPaths::DocumentsLocation);
325 if (documents.isEmpty())
326 {
327 documents = QDir::homePath();
328 }
329 QCOMPARE(VCommonSettings::getDefaultDataRoot(), QDir::cleanPath(documents) + QStringLiteral("/Seamly"));
330
331 // None of the superseded names may come back.
332 const QString root = VCommonSettings::getDefaultDataRoot();
333 QVERIFY(!root.endsWith(QStringLiteral("seamly2d")));
334 QVERIFY(!root.endsWith(QStringLiteral("seamlyData")));
335 // "seamly" alone collides too easily with an unrelated user folder of the same name;
336 // the capitalised family name under Documents does not.
337 QVERIFY(!root.endsWith(QStringLiteral("/seamly")));
