69/**
70 * @brief initTestCase creates the scratch directory every case works inside, and redirects
71 * the QSettings base directory so the developer's real settings file is never touched.
72 *
73 * The home directory is deliberately NOT redirected: it cannot be, on Windows. Tests must
74 * therefore never build a path from QDir::homePath() — see the class documentation.
75 */
76void TST_DataRoot::initTestCase()
77{
78 m_scratch.reset(new QTemporaryDir());
79 m_settings.reset(new QTemporaryDir());
80 QVERIFY2(m_scratch->isValid(), "Could not create the scratch directory");
81 QVERIFY2(m_settings->isValid(), "Could not create the temporary settings directory");
82
83 // QSettings::setPath() has no getter, so the current base directory is recovered from a
84 // probe instance — its file lands at <base>/<organization>/<application>.ini — and put
85 // back in cleanupTestCase().
86 const QSettings probe(QSettings::IniFormat, QSettings::UserScope,
87 QStringLiteral("SeamlyProbeOrganization"), QStringLiteral("SeamlyProbeApp"));
88 m_originalSettingsBase = QFileInfo(QFileInfo(probe.fileName()).absolutePath()).absolutePath();
89
90 QSettings::setPath(QSettings::IniFormat, QSettings::UserScope, m_settings->path());
91
92 // Task SettingsFiles.1: the common settings file no longer resolves through
93 // QSettings::setPath() — it lives under GenericConfigLocation, which cannot be
94 // redirected per test — so it gets its own override at the same temporary base.
95 VCommonSettings::setCommonSettingsBaseDir(m_settings->path());
96
97 QVERIFY2(!QCoreApplication::organizationName().isEmpty(),
98 "The data root resolves through the application organization name, which must be set");
99}

367/**
368 * @brief InstallerDataRootIsCleanOrEmpty checks the shape of whatever the Windows installer
369 * recorded, on any platform. Pins the contract of InstallerRecord::dataRoot().
370 *
371 * The value cannot be arranged from a test: it lives under HKLM, which an unelevated process
372 * cannot write. What the test can hold is the contract every caller depends on — the result
373 * is either empty or a path already cleaned into Qt's '/' form, because initializeDataRoot()
374 * stores it in the settings file verbatim and every getDefault*Path() then appends to it.
375 *
376 * Off Windows the result is always empty, and there the assertions still run: a change that
377 * made the function return something on Linux or macOS would fail here.
378 */
379void TST_DataRoot::InstallerDataRootIsCleanOrEmpty() const
380{
381 const QString recorded = InstallerRecord::dataRoot();
382 if (recorded.isEmpty())
383 {
384 return;
385 }
386
387 QVERIFY2(!recorded.contains(QLatin1Char('\\')),
388 "the recorded root must be converted out of native separators");
389 QCOMPARE(recorded, QDir::cleanPath(recorded));
390 QVERIFY2(QDir::isAbsolutePath(recorded), "the installer records an absolute path");
391}

515/**
516 * @brief writeTestFile creates a file with known contents, making any parent directories.
517 *
518 * @param path file to write.
519 * @param contents text to put in it.
520 * @return true when the file was written.
521 */
522static bool writeTestFile(const QString &path, const QString &contents)
523{
524 if (!QDir().mkpath(QFileInfo(path).absolutePath()))
525 {
526 return false;
527 }
528 QFile file(path);
529 if (!file.open(QIODevice::WriteOnly | QIODevice::Text))
530 {
531 return false;
532 }
533 file.write(contents.toUtf8());
534 file.close();
535 return true;
536}
537
538//---------------------------------------------------------------------------------------------------------------------
539/**
540 * @brief readTestFile returns a file's contents, or an empty string when it cannot be read.
541 */
542static QString readTestFile(const QString &path)
543{
544 QFile file(path);
545 if (!file.open(QIODevice::ReadOnly | QIODevice::Text))
546 {
547 return QString();
548 }
549 const QString contents = QString::fromUtf8(file.readAll());
550 file.close();
551 return contents;
552}

738/**
739 * @brief PerAppPathsPersistToTheOwnSettingsFile guards against VSettings's per-app path
740 * getters and setters (pattern, layout, the SeamlyLayout executable, graphical output)
741 * reopening a fresh QSettings from format()/scope()/organizationName()/applicationName()
742 * instead of reading and writing "this".
743 *
744 * Application2D builds VSettings from an explicit file path (see
745 * Application2D::InitTrVars()), so organizationName() and applicationName() are both empty
746 * on it. Reopening QSettings from those empty values landed the four settings under the
747 * literal "Unknown Organization" file instead of the app's own qt6_seamly2d.ini, which is
748 * also why qt6_seamly2d.ini's [paths] section was missing pattern and layout.
749 */
750void TST_DataRoot::PerAppPathsPersistToTheOwnSettingsFile() const
751{
752 const QString iniPath = scratchPath(QStringLiteral("per-app-paths/qt6_seamly2d.ini"));
753
754 VSettings settings(iniPath, QSettings::IniFormat);
755 settings.SetPathPattern(QStringLiteral("G:/My Drive/seamlyData/patterns"));
756 settings.SetPathLayout(QStringLiteral("G:/My Drive/seamlyData/layouts"));
757 settings.setSeamlyLayoutAppPath(QStringLiteral("C:/Program Files/SeamlyApps/SeamlyLayout.exe"));
758 settings.SetGraphicalOutput(false);
759 settings.sync();
760
761 QCOMPARE(settings.getPatternPath(), QStringLiteral("G:/My Drive/seamlyData/patterns"));
762 QCOMPARE(settings.getLayoutPath(), QStringLiteral("G:/My Drive/seamlyData/layouts"));
763 QCOMPARE(settings.getSeamlyLayoutAppPath(), QStringLiteral("C:/Program Files/SeamlyApps/SeamlyLayout.exe"));
764 QCOMPARE(settings.GetGraphicalOutput(), false);
765
766 // Re-open the file directly, bypassing the VSettings instance, to prove the values
767 // landed on disk in the app's own ini rather than only in a live QSettings cache.
768 const QSettings onDisk(iniPath, QSettings::IniFormat);
769 QCOMPARE(onDisk.value(QStringLiteral("paths/pattern")).toString(),
770 QStringLiteral("G:/My Drive/seamlyData/patterns"));
771 QCOMPARE(onDisk.value(QStringLiteral("paths/layout")).toString(),
772 QStringLiteral("G:/My Drive/seamlyData/layouts"));
773 QCOMPARE(onDisk.value(QStringLiteral("paths/seamlyLayoutApp")).toString(),
774 QStringLiteral("C:/Program Files/SeamlyApps/SeamlyLayout.exe"));
775 QCOMPARE(onDisk.value(QStringLiteral("pattern/graphicalOutput")).toBool(), false);
776}
