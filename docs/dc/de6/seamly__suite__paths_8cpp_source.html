41/**
42 * @brief sourceTreeBuildSubPath is the path of a SeamlyLayout development build
43 * relative to the root of a Seamly2D source checkout, without the build
44 * configuration directory.
45 *
46 * The Qt frontend's CMake build writes its executable to
47 * `<checkout>/src/app/seamlylayout/qt_frontend/build/<config>/SeamlyLayout(.exe)`.
48 */
49const QLatin1String sourceTreeBuildSubPath("/src/app/seamlylayout/qt_frontend/build/");
50
51/**
52 * @brief maxUpwardLevels bounds how far the development-build lookup walks up
53 * from the running executable's directory while searching for the checkout root.
54 *
55 * The deepest layout in use is the debug shadow build, whose executable sits at
56 * `<checkout>/scripts/seamly2d-debug/src/app/seamly2d/bin/` — six levels
57 * below the checkout root. The release shadow build (`<checkout>/build/...`) is
58 * five. Eight leaves room for a differently nested build tree while still
59 * terminating quickly, and keeps the walk from climbing out of the checkout and
60 * probing unrelated parts of the filesystem.
61 */
62const int maxUpwardLevels = 8;

86/**
87 * @brief locateSeamlyLayout looks for the SeamlyLayout executable in the two
88 * layouts an installation can use, relative to one install directory.
89 *
90 * Lookup order:
91 * 1. Flat: `<directory>/SeamlyLayout(.exe)` — the layout used when every app
92 * shares one Qt runtime. Since Task 30 that is every supported install:
93 * the Windows MSI, the Linux Flatpak's `/app/bin`, and a local development
94 * tree, because all three apps now build against the same Qt release.
95 * 2. Subdirectory: `<directory>/SeamlyLayout/SeamlyLayout(.exe)` — the layout
96 * the Windows MSI used before Task 30 (Task 13). Back then SeamlyLayout was
97 * built against a different Qt release than seamly2d/seamlyme, so its Qt
98 * runtime DLLs could not share a flat directory with the parent apps'
99 * (identical file names, different versions) and the installer gave it its
100 * own subdirectory with its own Qt runtime. No current installer produces
101 * this layout; the branch is kept so a seamly2d upgraded in place over such
102 * an install — or any future packaging that isolates the daughter app —
103 * still finds the executable.
104 *
105 * Both candidates must be existing regular files — a *directory* that happens
106 * to be named like the executable (e.g. the "SeamlyLayout" subdirectory itself
107 * on non-Windows platforms, where the exe name has no ".exe" suffix) is never
108 * a match.
109 *
110 * @param directory absolute path of the directory to search (typically the
111 * directory of the running seamly2d executable).
112 * @return absolute path of the SeamlyLayout executable, or an empty string
113 * when neither layout contains it.
114 */
115QString locateSeamlyLayout(const QString &directory)
116{
117 const QString exeName = seamlyLayoutExeName();
118
119 // Layout 1 — flat: the executable directly beside the caller's apps.
120 const QFileInfo flat(directory + QLatin1Char('/') + exeName);
121 if (flat.exists() && flat.isFile())
122 {
123 return flat.absoluteFilePath();
124 }
125
126 // Layout 2 — MSI subdirectory: SeamlyLayout isolated with its own Qt runtime.
127 const QFileInfo nested(directory + QLatin1String("/SeamlyLayout/") + exeName);
128 if (nested.exists() && nested.isFile())
129 {
130 return nested.absoluteFilePath();
131 }
132
133 return QString(); // Neither layout present; the caller decides what to do.
134}

137/**
138 * @brief locateSeamlyLayoutDevBuild looks for a SeamlyLayout executable built
139 * from the same source checkout as the running application.
140 *
141 * This is the development convenience that lets a locally built seamly2d hand
142 * off to a locally built SeamlyLayout with no configuration, on any machine.
143 * It replaces the single hard-coded developer path this function grew out of
144 * (Task 50), which named one contributor's checkout and therefore helped
145 * exactly one machine.
146 *
147 * The checkout root is derived from the caller's directory rather than assumed:
148 * the walk starts at @p startDirectory and climbs up to ::maxUpwardLevels
149 * parents, treating each as a candidate checkout root and testing whether it
150 * contains a SeamlyLayout build. Both shadow-build layouts the project uses
151 * resolve this way — the release build at `<checkout>/build/...` (five levels)
152 * and a debug shadow build at `<checkout>/scripts/seamly2d-debug/...` (six) —
153 * without either being named here, so a differently nested build tree still
154 * works.
155 *
156 * At each level **Release is preferred over Debug**: a developer who has built
157 * both almost always wants the current release binary, and the old hard-coded
158 * path silently pinned Debug, which could be arbitrarily stale.
159 *
160 * This lookup is deliberately the *last* resort in Application2D's chain — a
161 * configured setting and an installed copy both outrank it — so a source tree
162 * that happens to sit above an installed application can never shadow the
163 * installation. Because it only ever matches inside a checkout that has been
164 * built, it is inert on an end user's machine.
165 *
166 * @param startDirectory absolute path to start the upward walk from (typically
167 * the directory of the running executable). Passed in rather than read
168 * from QCoreApplication so the tests can point it at a QTemporaryDir.
169 * @return absolute path of the SeamlyLayout development build, or an empty
170 * string when no checkout above @p startDirectory contains one.
171 */
172QString locateSeamlyLayoutDevBuild(const QString &startDirectory)
173{
174 if (startDirectory.isEmpty())
175 {
176 return QString();
177 }
178
179 const QString exeName = seamlyLayoutExeName();
180
181 // Release first, so a stale Debug build never wins over a current Release one.
182 const QStringList configurations{QLatin1String("Release"), QLatin1String("Debug")};
183
184 QDir directory(startDirectory);
185
186 // Climb one level per iteration, testing each ancestor as a checkout root.
187 for (int level = 0; level <= maxUpwardLevels; ++level)
188 {
189 const QString candidateRoot = directory.absolutePath();
190
191 for (const QString &configuration : configurations)
192 {
193 const QFileInfo candidate(candidateRoot + sourceTreeBuildSubPath + configuration
194 + QLatin1Char('/') + exeName);
195 // Must be an existing regular file — a directory of the same name is not a match.
196 if (candidate.exists() && candidate.isFile())
197 {
198 return candidate.absoluteFilePath();
199 }
200 }
201
202 // cdUp() fails at the filesystem root, which ends the walk early.
203 if (!directory.cdUp())
204 {
205 break;
206 }
207 }
208
209 return QString(); // No development build above the caller; not a source checkout.
210}

213/**
214 * @brief patternDocumentName returns the name Layout Mode gives the piece-mode
215 * document it hands to SeamlyLayout.
216 *
217 * The handoff carries no file, so SeamlyLayout has no file name to build its
218 * default export names from. This is that name: the pattern's complete base
219 * name, so "shirt.v2.sm2d" hands over "shirt.v2" and a later SVG export in
220 * SeamlyLayout still defaults to a name the user recognises.
221 *
222 * completeBaseName() is used deliberately — it keeps everything up to the *last*
223 * dot, so a version segment in the pattern name survives.
224 *
225 * @param patternFilePath path of the open pattern file; may be relative.
226 * @return the pattern's complete base name, or an empty string when
227 * @p patternFilePath is empty (no pattern has been saved yet).
228 */
229QString patternDocumentName(const QString &patternFilePath)
230{
231 if (patternFilePath.isEmpty())
232 {
233 return QString(); // Unsaved pattern; SeamlyLayout falls back to its own default.
234 }
235
236 return QFileInfo(patternFilePath).completeBaseName();
237}

240/**
241 * @brief seamlyLayoutLaunchArguments builds the argument vector seamly2d passes
242 * to the SeamlyLayout executable.
243 *
244 * This is the seamly2d side of the two-app launch contract. Layout Mode does not
245 * write a handoff file: it serialises piece mode to one stringified SVG and
246 * sends that document on the child process's standard input (Seamly2D.5). The
247 * arguments only say how to read it:
248 *
249 * SeamlyLayout --svg-stdin --document-name <pattern base name>
250 *
251 * SeamlyLayout's `StartupOptions` class
252 * (`src/app/seamlylayout/qt_frontend/src/StartupOptions.cpp`) parses exactly
253 * these two options, and rejects a positional file alongside them; both halves
254 * are pinned by tests. This function must never grow an argument silently.
255 *
256 * --document-name is omitted for an unsaved pattern rather than passed empty, so
257 * SeamlyLayout keeps its own default export name instead of an empty one.
258 *
259 * The name is passed as its own list element rather than embedded in a command
260 * string, so QProcess quotes it and a pattern name containing spaces needs no
261 * special handling.
262 *
263 * @param documentName pattern base name, as returned by patternDocumentName();
264 * may be empty.
265 * @return the argument list for QProcess::start().
266 */
267QStringList seamlyLayoutLaunchArguments(const QString &documentName)
268{
269 QStringList arguments(QStringLiteral("--svg-stdin"));
270
271 if (!documentName.isEmpty())
272 {
273 arguments << QStringLiteral("--document-name") << documentName;
274 }
275
276 return arguments;
277}
