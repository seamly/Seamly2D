57 /**
58 * @brief canShowSplash reports whether this process has a screen to put a window on.
59 *
60 * openSettings() runs from the application constructor, before VCommandLine has parsed
61 * anything, so Application2D::isGUIMode() is not yet answerable — it returns false for
62 * every run at this point, GUI or not. The platform plugin is the one thing that is
63 * already settled, and it is what actually decides whether a window can appear: the test
64 * suites and the CI jobs run with QT_QPA_PLATFORM=offscreen.
65 *
66 * A console export therefore does get the splash. That is accepted: it is not modal, it
67 * closes itself, and it only ever appears on the single run that migrates the tree.
68 */
69 bool canShowSplash()
70 {
71 if (qobject_cast<QApplication *>(QCoreApplication::instance()) == nullptr)
72 {
73 return false;
74 }
75
76 const QString platform = QGuiApplication::platformName();
77 return !platform.isEmpty()
78 && platform != QLatin1String("offscreen")
79 && platform != QLatin1String("minimal");
80 }

83 /**
84 * @brief MigrationSplash is the "please wait" window, or nothing at all.
85 *
86 * Constructing it on a headless run gives an object whose every method does nothing, so
87 * the caller needs no branches of its own.
88 *
89 * The colours are fixed rather than taken from the palette. The splash is created before
90 * Application2D::setTheme() runs, so there is no application palette to read yet, and a
91 * light panel with dark text is legible under either theme.
92 */
93 class MigrationSplash
94 {
95 public:
96 MigrationSplash();
97 ~MigrationSplash();
98
99 void show(const QString &message);
100 void hold(int milliseconds);
101
102 private:
103 Q_DISABLE_COPY_MOVE(MigrationSplash)
104
105 QSplashScreen *m_splash;
106 };
107
108 //-----------------------------------------------------------------------------------------------------------------
109 MigrationSplash::MigrationSplash()
110 : m_splash(nullptr)
111 {
112 if (!canShowSplash())
113 {
114 return;
115 }
116
117 const QSize logicalSize(560, 220);
118 const qreal ratio = qApp->devicePixelRatio();
119
120 QPixmap panel(logicalSize * ratio);
121 panel.setDevicePixelRatio(ratio);
122 panel.fill(QColor(0xFA, 0xFA, 0xFA));
123
124 QPainter painter(&panel);
125 painter.setPen(QPen(QColor(0x60, 0x60, 0x60), 1));
126 painter.drawRect(QRect(QPoint(0, 0), logicalSize - QSize(1, 1)));
127 painter.end();
128
129 m_splash = new QSplashScreen(panel);
130 m_splash->setFont(QFont(m_splash->font().family(), 10));
131 }
132
133 //-----------------------------------------------------------------------------------------------------------------
134 MigrationSplash::~MigrationSplash()
135 {
136 if (m_splash != nullptr)
137 {
138 m_splash->close();
139 delete m_splash;
140 }
141 }

143 //-----------------------------------------------------------------------------------------------------------------
144 /**
145 * @brief show puts a message on the splash and paints it now.
146 *
147 * The event loop has not started yet — this all runs from the application constructor —
148 * so without the explicit processEvents() the window would be created and never drawn.
149 */
150 void MigrationSplash::show(const QString &message)
151 {
152 if (m_splash == nullptr)
153 {
154 return;
155 }
156
157 m_splash->show();
158 m_splash->showMessage(message, Qt::AlignLeft | Qt::AlignVCenter | Qt::TextWordWrap,
159 QColor(0x20, 0x20, 0x20));
160 QCoreApplication::processEvents();
161 }
162
163 //-----------------------------------------------------------------------------------------------------------------
164 /**
165 * @brief hold keeps the last message on screen long enough to read.
166 *
167 * processEvents() in a loop rather than a nested QEventLoop: this runs during application
168 * construction, and a nested event loop there can re-enter code that is not built yet.
169 */
170 void MigrationSplash::hold(int milliseconds)
171 {
172 if (m_splash == nullptr)
173 {
174 return;
175 }
176
177 QElapsedTimer timer;
178 timer.start();
179 while (timer.elapsed() < milliseconds)
180 {
181 QCoreApplication::processEvents(QEventLoop::AllEvents, 50);
182 }
183 }

187/**
188 * @brief run copies the legacy tree out, then archives what it left behind.
189 *
190 * Called only when initializeDataRoot() has just adopted a legacy tree, so it is the first
191 * run of the first app after an upgrade. Every step is fail-safe: a failure at any point
192 * leaves the user's files where the previous step put them, and the worst case is that the
193 * app carries on from ~/seamly2d exactly as the old build did.
194 *
195 * The legacy tree is never deleted here. migrateAdoptedLegacyTree() already leaves it in
196 * place with a MIGRATED-TO-SEAMLY.txt marker; the .zip this function adds is a second,
197 * portable backup beside it, not a replacement for it.
198 *
199 * @param legacyRoot the adopted tree, normally ~/seamly2d or <Documents>/Seamly.
200 * @param newRoot where it should live now, normally <Documents>/SeamlyData.
201 * @return the data root actually in force afterwards.
202 */
203QString LegacyDataMigration::run(const QString &legacyRoot, const QString &newRoot)
204{
205 if (legacyRoot.isEmpty() || newRoot.isEmpty() || !QFileInfo(legacyRoot).isDir())
206 {
207 return legacyRoot;
208 }
209
210 MigrationSplash splash;
211 splash.show(QStringLiteral("Moving your work to\n\n %1\n\nPlease wait. A large collection of "
212 "patterns can take a few minutes.")
213 .arg(QDir::toNativeSeparators(newRoot)));
214
215 const QString root = VCommonSettings::migrateAdoptedLegacyTree(legacyRoot, newRoot);
216 if (root != newRoot)
217 {
218 // migrateAdoptedLegacyTree() has already said why. Nothing was moved, so there is
219 // nothing to back up yet.
220 return root;
221 }
222
223 splash.show(QStringLiteral("Backing up your old folder\n\n %1\n\nChecking every file as it is "
224 "packed into a backup.")
225 .arg(QDir::toNativeSeparators(legacyRoot)));
226
227 QString errorMessage;
228 const QString archive = LegacyDataArchive::archive(legacyRoot, root, &errorMessage);
229
230 if (archive.isEmpty())
231 {
232 qWarning() << "Could not back up the old data folder" << QDir::toNativeSeparators(legacyRoot)
233 << ':' << errorMessage;
234 splash.show(QStringLiteral("Your work is now in\n\n %1\n\nYour old folder could not be backed "
235 "up as a .zip, but nothing was changed there. It is still at\n\n %2")
236 .arg(QDir::toNativeSeparators(root), QDir::toNativeSeparators(legacyRoot)));
237 }
238 else
239 {
240 qInfo() << "Backed up the old data folder to" << QDir::toNativeSeparators(archive) << "- it remains at"
241 << QDir::toNativeSeparators(legacyRoot);
242 splash.show(QStringLiteral("Your work is now in\n\n %1\n\nYour old folder is kept at\n\n %2\n\n"
243 "and backed up in\n\n %3")
244 .arg(QDir::toNativeSeparators(root), QDir::toNativeSeparators(legacyRoot),
245 QDir::toNativeSeparators(archive)));
246 }
247
248 splash.hold(finalMessageHoldMs);
249 return root;
250}
