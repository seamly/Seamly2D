39/// @brief Open or raise the preferences dialog.
40void PreferencesController::openPreferences()
41{
42 if (!m_model) {
43 Logger::log(QStringLiteral("PreferencesController::openPreferences(): no model set"));
44 return;
45 } // if no model
46
47 if (!m_window) {
48 // First open — create the window and wire signals.
49 Logger::log(QStringLiteral("PreferencesController::openPreferences(): creating PreferencesWindow"));
50
51 m_window = new PreferencesWindow(m_model);
52
53 // Forward PreferencesWindow::saved() as saved().
54 connect(m_window, &PreferencesWindow::saved,
55 this, &PreferencesController::saved);
56
57 // Forward PreferencesWindow::defaultsReset() as defaultsReset().
58 connect(m_window, &PreferencesWindow::defaultsReset,
59 this, &PreferencesController::defaultsReset);
60
61 // Forward PreferencesWindow::discarded() as discarded().
62 connect(m_window, &PreferencesWindow::discarded,
63 this, &PreferencesController::discarded);
64
65 // Clear our pointer when the dialog is destroyed,
66 // so the next openPreferences() call creates a fresh dialog.
67 connect(m_window, &QObject::destroyed,
68 this, [this]() {
69 m_window = nullptr;
70 Logger::log(QStringLiteral("PreferencesController: PreferencesWindow destroyed"));
71 }); // destroyed lambda
72 } else {
73 // Subsequent call — reload fields from the model in case they changed.
74 Logger::log(QStringLiteral("PreferencesController::openPreferences(): raising PreferencesWindow"));
75 m_window->reloadFromModel();
76 } // if no window yet
77
78 m_window->show();
79 m_window->raise();
80 m_window->activateWindow();
81} // openPreferences
82
