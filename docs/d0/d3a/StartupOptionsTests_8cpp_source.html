5// @file StartupOptionsTests.cpp
6// @brief Qt tests for StartupOptions — SeamlyLayout's command-line contract.
7//
8// This suite locks the SeamlyLayout half of the Seamly2D Layout Mode handoff
9// (Task 49). Seamly2D launches
10//
11// SeamlyLayout <absolute path to <pattern>.pieces.svg>
12//
13// detached, and until Task 49 that argument was read by nobody: the daughter
14// app opened an empty canvas. The seamly2d half of the same contract — that
15// the launch really is "one positional argument, the .pieces.svg path" — is
16// locked by TST_SeamlySuitePaths in the Seamly2DTest suite.
17//
18// Covers:
19// • No argument / empty argument list → start with an empty canvas
20// • A real .svg (including the ".pieces.svg" double extension and an
21// upper-case ".SVG") → Status::OpenFile with an ABSOLUTE path
22// • A relative argument resolved against the working directory
23// • Missing file, directory, unreadable file, non-SVG suffix → Status::Failed
24// with a message naming the file
25// • More than one positional argument → Status::Failed
26// • Unknown option → Status::Failed carrying the parser's own text
27// • --help / -h / --version / -v → Status::ShowInformation
28//
29// StartupOptions needs no QObject, no window and no event loop, so this suite
30// runs guiless — QTEST_GUILESS_MAIN, not QTEST_MAIN.

82QString StartupOptionsTests::makeFile(const QTemporaryDir &directory, const QString &fileName)
83{
84 const QString path = directory.filePath(fileName);
85 QFile file(path);
86 // Verified rather than assumed: a silent failure here would make every
87 // assertion below test the "missing file" path by accident.
88 Q_ASSERT(file.open(QIODevice::WriteOnly));
89 file.write("<svg xmlns=\"http://www.w3.org/2000/svg\"/>");
90 file.close();
91 return QFileInfo(path).absoluteFilePath();
92} // StartupOptionsTests::makeFile

179// @brief A relative argument is resolved against the process working directory.
180void StartupOptionsTests::relativePath_isResolvedToAbsolute()
181{
182 QTemporaryDir directory;
183 QVERIFY(directory.isValid());
184 const QString svgPath = makeFile(directory, QStringLiteral("relative.svg"));
185
186 // Restored before leaving the test so the following tests are unaffected.
187 const QString previousWorkingDirectory = QDir::currentPath();
188 QVERIFY(QDir::setCurrent(directory.path()));
189
190 const StartupOptions options = StartupOptions::parse(
191 QStringList{QStringLiteral("SeamlyLayout"), QStringLiteral("relative.svg")});
192
193 QVERIFY(QDir::setCurrent(previousWorkingDirectory));
194
195 QCOMPARE(options.status(), StartupOptions::Status::OpenFile);
196 QVERIFY(QFileInfo(options.filePath()).isAbsolute());
197 // Compare the canonical forms: a temporary directory may sit under a
198 // symlinked path (/tmp, /var on macOS), which QFileInfo does not resolve.
199 QCOMPARE(QFileInfo(options.filePath()).canonicalFilePath(),
200 QFileInfo(svgPath).canonicalFilePath());
201} // relativePath_isResolvedToAbsolute()

253// @brief A file the process cannot read is reported rather than silently failing later.
254void StartupOptionsTests::unreadableFile_statusIsFailed()
255{
256#ifdef Q_OS_WIN
257 QSKIP("NTFS does not deny the owner read access through QFile::setPermissions");
258#else
259 QTemporaryDir directory;
260 QVERIFY(directory.isValid());
261 const QString svgPath = makeFile(directory, QStringLiteral("locked.svg"));
262 QVERIFY(QFile::setPermissions(svgPath, QFileDevice::WriteOwner));
263
264 const StartupOptions options =
265 StartupOptions::parse(QStringList{QStringLiteral("SeamlyLayout"), svgPath});
266
267 // Restore before asserting so QTemporaryDir can always clean up.
268 QFile::setPermissions(svgPath, QFileDevice::ReadOwner | QFileDevice::WriteOwner);
269
270 // Running as root defeats permission bits entirely; skip rather than fail.
271 if (options.status() == StartupOptions::Status::OpenFile) {
272 QSKIP("Process can read a file with no read permission (running as root?)");
273 } // if the permission bit had no effect
274
275 QCOMPARE(options.status(), StartupOptions::Status::Failed);
276 QVERIFY(options.message().contains(QStringLiteral("locked.svg")));
277#endif
278} // unreadableFile_statusIsFailed()

357// @brief The version text carries the metadata the application set.
358void StartupOptionsTests::versionOption_messageCarriesTheVersion()
359{
360 // Set on the test's own QCoreApplication, exactly as main.cpp does before
361 // parsing, so the value reaching the user is the one under test.
362 const QString previousName = QCoreApplication::applicationName();
363 const QString previousVersion = QCoreApplication::applicationVersion();
364 QCoreApplication::setApplicationName(QStringLiteral("SeamlyLayout"));
365 QCoreApplication::setApplicationVersion(QStringLiteral("9.9.9"));
366
367 const StartupOptions options = StartupOptions::parse(
368 QStringList{QStringLiteral("SeamlyLayout"), QStringLiteral("-v")});
369
370 QCoreApplication::setApplicationName(previousName);
371 QCoreApplication::setApplicationVersion(previousVersion);
372
373 QCOMPARE(options.status(), StartupOptions::Status::ShowInformation);
374 QVERIFY(options.message().contains(QStringLiteral("SeamlyLayout")));
375 QVERIFY(options.message().contains(QStringLiteral("9.9.9")));
