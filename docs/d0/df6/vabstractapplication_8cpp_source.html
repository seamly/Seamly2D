232/**
233 * @brief MigrateSeamlySettingsLocation resolves this app's settings-file path under the
234 * unified "Seamly" organization (Task 15), migrating the file forward from the
235 * pre-unification "Seamly2DTeam" organization folder on first run after an upgrade.
236 *
237 * Before Task 15, seamly2d and seamlyme each stored one flat .ini file (named after the
238 * application) as a sibling inside a single shared "Seamly2DTeam" organization folder
239 * (Qt's native IniFormat/UserScope resolution). After Task 15 every Seamly application
240 * gets its own directory nested under the "Seamly" organization instead — QStandardPaths
241 * ::AppConfigLocation already resolves to AppData/Local/Seamly/<AppName> on Windows once
242 * the organization name is "Seamly" and the application name is set, matching the layout
243 * seamlyLayout already uses via QStandardPaths internally. This function creates that new
244 * per-app directory and — only if its settings file does not already exist there — copies
245 * it in from whichever legacy filename candidate is found, so upgrading is non-destructive
246 * and re-entrant (running it again after the file exists is a cheap no-op).
247 *
248 * @param appIniFileName filename to use for this app's settings in the new location, e.g.
249 * "qt6_seamly2d.ini".
250 * @param legacyAppIniFileNames candidate filenames to look for inside the legacy
251 * organization folder, tried in order (newest format first), e.g. {"qt6_seamly2d.ini",
252 * "Seamly2D.ini"}.
253 * @param migrated optional out-parameter set to true if a legacy file was copied into the
254 * new location; leave null if the caller does not need to know (e.g. test harnesses, which
255 * must never show the migration notice dialog).
256 * @return absolute path to this app's settings file in the new unified location.
257 */
258QString VAbstractApplication::MigrateSeamlySettingsLocation(const QString &appIniFileName,
259 const QStringList &legacyAppIniFileNames,
260 bool *migrated)
261{
262 if (migrated != nullptr)
263 {
264 *migrated = false;
265 }
266
267 // New home: one directory per application, nested under the shared "Seamly"
268 // organization folder (AppData/Local/Seamly/<AppName> on Windows).
269 const QString newAppDir = QStandardPaths::writableLocation(QStandardPaths::AppConfigLocation);
270 QDir().mkpath(newAppDir);
271 const QString newAppIniPath = QDir(newAppDir).filePath(appIniFileName);
272
273 if (QFileInfo::exists(newAppIniPath))
274 {
275 // Already migrated on a previous run, or a fresh install with nothing to bring
276 // forward — either way there is nothing left to do.
277 return newAppIniPath;
278 }
279
280 // Legacy home: pre-Task-15 builds used organization name "Seamly2DTeam", with both
281 // apps' settings living as sibling flat files in that one shared folder.
282 static const QString kLegacyOrganizationName = QStringLiteral("Seamly2DTeam");
283 const QSettings legacyProbe(QSettings::IniFormat, QSettings::UserScope,
284 kLegacyOrganizationName, QCoreApplication::applicationName());
285 const QString legacyDir = QFileInfo(legacyProbe.fileName()).absolutePath();
286
287 for (const QString &legacyFileName : legacyAppIniFileNames)
288 {
289 const QString legacyPath = QDir(legacyDir).filePath(legacyFileName);
290 if (QFileInfo::exists(legacyPath) && QFile::copy(legacyPath, newAppIniPath))
291 {
292 if (migrated != nullptr)
293 {
294 *migrated = true;
295 }
296 break;
297 } // if legacy file found and copied
298 } // for each candidate legacy filename
299
300 return newAppIniPath;
301}

305 * @brief NotifySeamlySettingsMigrated shows a one-time informational dialog after Task 15
306 * settings migration moved a user's preferences to the new unified "Seamly" location.
307 * Only call this from a confirmed GUI-mode code path — never from a headless/CLI export
308 * run or an automated test, since a modal dialog with no automatic dismissal would hang
309 * a scripted caller waiting on process exit.
310 * @param appDisplayName human-readable app name to mention in the message (e.g. "Seamly2D").

316 tr("Settings moved"),
317 tr("%1's settings and preferences have moved to a new shared location "
318 "(the \"Seamly\" folder). Nothing was lost — this happens automatically, "
319 "once, after upgrading.").arg(appDisplayName));
320}

360void VAbstractApplication::loadTranslations(const QString &locale)
361{
362 if (locale.isEmpty())
363 {
364 qInfo() << "Locale is empty.";
365 return;
366 }
367 qInfo() << "Checked locale:" << locale;
368
369 ClearTranslation();
370
371 qtTranslator = new QTranslator(this);
372 qtxmlTranslator = new QTranslator(this);
373 qtBaseTranslator = new QTranslator(this);
374 appTranslator = new QTranslator(this);
375 pmsTranslator = new QTranslator(this);
376
377#if defined(Q_OS_WIN) || defined(Q_OS_MAC)
378 // Explicitly cast to void to suppress clang warnings
379 (void)qtTranslator->load("qt_" + locale, translationsPath(locale));
380 (void)qtxmlTranslator->load("qtxmlpatterns_" + locale, translationsPath(locale));
381 (void)qtBaseTranslator->load("qtbase_" + locale, translationsPath(locale));
382#else
383 // Explicitly cast to void to suppress clang warnings
384 (void)qtTranslator->load("qt_" + locale, QLibraryInfo::location(QLibraryInfo::TranslationsPath));
385 (void)qtxmlTranslator->load("qtxmlpatterns_" + locale, QLibraryInfo::location(QLibraryInfo::TranslationsPath));
386 (void)qtBaseTranslator->load("qtbase_" + locale, QLibraryInfo::location(QLibraryInfo::TranslationsPath));
387#endif
388
389 // Explicitly cast to void to suppress clang warnings
390 (void)appTranslator->load("seamly2d_" + locale, translationsPath(locale));
391 (void)pmsTranslator->load("measurements_" + locale, translationsPath(locale));
392
393 installTranslator(qtTranslator);
394 installTranslator(qtxmlTranslator);
395 installTranslator(qtBaseTranslator);
396 installTranslator(appTranslator);
397 installTranslator(pmsTranslator);
398
399 initTranslateVariables();//Very important do it after load QM files.
400}
