116// withGrain + rotationEnabled=true → alongGrainline

119// withoutGrain + (any) → alongGrainline

313//
314// These tests verify the fix for the Phase 5 bug: SettingsModel::load()
315// must emit all property notify signals AND settingsLoaded() unconditionally
316// at the end of a successful load, even for fields whose loaded value equals
317// the current model value (where the setter guard would otherwise suppress
318// the signal). Without these unconditional emissions, QML controls with
319// broken runtime bindings (modified by user interaction) never refresh.

324{

329 // Write a file with default unit so the setter guard would suppress unitChanged()
330 // without the BC.1 fix.

333 QFile f(path);

336 f.close();

587// @brief save() writes all 22 expected JSON keys after a file load.
588// A missing key in save() would break Rust LayoutSettings deserialization
589// and cause silent default-value fallback in the layout engine.
590void SettingsModelTests::bc2_saveAfterLoad_savedJsonContainsAllKeys()
591{
592 QTemporaryDir dir;
593 QVERIFY(dir.isValid());
594 const QString loadPath = dir.filePath(QStringLiteral("input.json"));
595 const QString savePath = dir.filePath(QStringLiteral("output.json"));
596
597 // Minimal fixture — only a few fields; the rest keep defaults.
598 QJsonObject fixture;
599 fixture[QStringLiteral("unit")] = QStringLiteral("cm");
600 fixture[QStringLiteral("layoutMode")] = QStringLiteral("withNap");
601 QFile f(loadPath);
602 QVERIFY(f.open(QIODevice::WriteOnly | QIODevice::Truncate));
603 f.write(QJsonDocument(fixture).toJson());
604 f.close();
605
606 SettingsModel m;
607 QVERIFY(m.load(loadPath));
608 QVERIFY(m.save(savePath));
609
610 // Parse the saved file.
611 QFile saved(savePath);
612 QVERIFY(saved.open(QIODevice::ReadOnly));
613 QJsonParseError err;
614 const QJsonDocument doc = QJsonDocument::fromJson(saved.readAll(), &err);
615 QCOMPARE(err.error, QJsonParseError::NoError);
616 const QJsonObject savedObj = doc.object();
617
618 // All 22 keys that save() is contractually required to write.
619 const QStringList expectedKeys = {
620 QStringLiteral("layoutMode"), QStringLiteral("rotationStep"),
621 QStringLiteral("fabricFolded"), QStringLiteral("unit"),
622 QStringLiteral("mediaType"), QStringLiteral("paperType"),
623 QStringLiteral("sheetName"), QStringLiteral("pageWidth"),
624 QStringLiteral("pageHeight"), QStringLiteral("rollSize"),
625 QStringLiteral("rollWidth"), QStringLiteral("tileSize"),

733// @brief Full pipeline: load file → edit fields via setters → save → reload → all fields match.
734// Simulates the actual user workflow: open a settings file, change values in
