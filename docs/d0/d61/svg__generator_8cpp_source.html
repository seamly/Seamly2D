62/**
63 * @brief Merge all the SVGs in the m_domList list into a single SVG
64 * @return The merged SVG as a DOM document
65 * @details m_domList contains DOM representations of multiple SVGs
66 Assuming each svg contains a main group containing every graphical item of the svg,
67 this function adds to the first svg of the list all the main groups of the other svgs,
68 thus creating a single svg with each svg of the list in it, every svg being in its own group.
69 This function is used in order to create svgs containing groups
70 */
71QDomDocument SvgGenerator::mergeSvgDoms()
72{
73 if (m_domList.isEmpty()) {
74 qDebug() << "Error : the SVG list is empty";
75 return QDomDocument();
76 }
77
78 QDomDocument mergedSvg = m_domList.at(0).cloneNode().toDocument();
79
80 QDomElement mergedSvgRoot = mergedSvg.documentElement();
81 if (mergedSvgRoot.tagName() != "svg") {
82 qDebug() << "Error : the first SVG does not contain a <svg> tag.";
83 return QDomDocument();
84 }
85
86 QDomNodeList mergedSvgGroups = mergedSvgRoot.elementsByTagName("g");
87 if (mergedSvgGroups.isEmpty()) {
88 qDebug() << "Error : the SVG does not contain a <g> tag.";
89 return QDomDocument();
90 }
91 // Drop the clone's own main group; it is re-added from the list below.
92 mergedSvgGroups.at(0).parentNode().removeChild(mergedSvgGroups.at(0));
93
94 // Piece-based exports get one pattern group wrapping all piece groups, tagged
95 // with the SVG data-* attributes so downstream tools can identify the pattern.
96 // Whole-scene exports (draft blocks) keep the legacy flat structure.
97 QDomElement parentElement = mergedSvgRoot;
98 if (m_pieceCount > 0)
99 {
100 QDomElement patternGroup = mergedSvg.createElement("g");
101 patternGroup.setAttribute("id", "pattern-1");
102 patternGroup.setAttribute("data-type", "pattern");
103 patternGroup.setAttribute("data-type-number", "1");
104 setAttribute(patternGroup, "data-name", m_patternName);
105 mergedSvgRoot.appendChild(patternGroup);
106 parentElement = patternGroup;
107 }
108
109 for (int i = 0; i < m_domList.size(); ++i) {
110 QDomDocument domSvg = m_domList.at(i);
111 QDomElement svgRoot = domSvg.documentElement();
112 if (svgRoot.tagName() != "svg") {
113 qDebug() << "Error : the SVG does not contain a <svg> tag.";
114 return QDomDocument();
115 }
116 QDomNodeList svgGroups = svgRoot.elementsByTagName("g");
117 if (svgGroups.isEmpty()) {
118 qDebug() << "Error : the SVG does not contain a <g> tag.";
119 return QDomDocument();
120 }
121 QDomElement mainGroup = svgGroups.at(0).toElement();
122 cleanSvg(mainGroup);
123 // importNode: nodes must be cloned into the target document before appending;
124 // appending a node owned by another document is silently ignored by QDom.
125 parentElement.appendChild(mergedSvg.importNode(mainGroup, true));
126 }
127
128 return mergedSvg;
129}

132/**
133 * @brief Remove empty groups from the SVG
134 * @return void
135 * @details This function removes empty unuseful groups from the exported SVG.
136 * Those empty groups were generated by Qt svg generator.
137 */
138void SvgGenerator::removeEmptyGroups(QDomElement &mainGroup)
139{
140 bool svgCleaned = false;
141
142 //We must remove the empty groups one by one until no more empty group is found.
143 //Since removing a group modifies the QDomNodeList indexation, we start over
144 //the search from the beginning each time a group is removed.
145 while(!svgCleaned)
146 {
147 svgCleaned = true;
148 QDomNodeList groups = mainGroup.elementsByTagName("g");
149
150 for (int i = 0; i < groups.size(); ++i) {
151 QDomElement group = groups.at(i).toElement();
152 if (group.childNodes().isEmpty()) {
153 // Remove from the group's real parent: empty groups can be nested at
154 // any depth, not only directly below mainGroup.
155 if (group.parentNode().removeChild(group).isNull()) {
156 qDebug() << "Error : could not remove empty group";
157 }
158 svgCleaned = false;
159 break;
160 }
161 }
162 }
163}

166/**
167 * @brief Remove the M0,0 origin paths from the SVG
168 * @return void
169 * @details This function removes every empty "M0,0" path from the SVG. Qt's SVG
170 * generator emits such a path for each QGraphicsPathItem with an empty
171 * path (e.g. the invisible piece container and label group items). If
172 * these paths are not removed, the bounding box of the exported pattern
173 * piece when opened in a svg editing software can be wrong.
174 */
175void SvgGenerator::removeEmptyOriginPath(QDomElement &mainGroup)
176{
177 QDomNodeList paths = mainGroup.elementsByTagName("path");
178 // Iterate backwards: QDomNodeList is live, removing a node reindexes the list.
179 for (int i = paths.size() - 1; i >= 0; --i) {
180 QDomElement path = paths.at(i).toElement();
181 const QString d = path.attribute("d");
182 if (d.isEmpty() || d == "M0,0") {
183 QDomElement parentGroup = path.parentNode().toElement();
184 parentGroup.removeChild(path);
185 // Drop the wrapping group as well when the path was its only content.
186 if (parentGroup != mainGroup && parentGroup.childNodes().isEmpty()) {
187 parentGroup.parentNode().removeChild(parentGroup);
188 }
189 }
190 }
191}

225/**
226 * @brief Render a graphics scene into an SVG DOM document.
227 *
228 * The scene is painted through Qt's QSvgGenerator into an in-memory buffer and
229 * the resulting SVG markup is parsed into a QDomDocument for post-processing.
230 *
231 * @param scene the scene to render.
232 * @return the rendered SVG as a DOM document; a null document when parsing failed.
233 */
234QDomDocument SvgGenerator::renderSceneToDom(QGraphicsScene *scene)
235{
236 QByteArray byteArray;
237 QBuffer buffer(&byteArray);
238 buffer.open(QIODevice::WriteOnly);
239
240 QSvgGenerator svgGenerator;
241 svgGenerator.setOutputDevice(&buffer);
242 svgGenerator.setSize(m_paper->rect().size().toSize());
243 svgGenerator.setViewBox(m_paper->rect());
244 svgGenerator.setTitle(QString());
245 svgGenerator.setDescription(QString());
246 svgGenerator.setResolution(m_resolution);
247
248 QFont appFont;
249 appFont.setPointSize(8);
250
251 QPainter painter;
252 painter.begin(&svgGenerator);
253 painter.setFont(appFont);
254 painter.setRenderHint(QPainter::Antialiasing, true);
255 painter.setBrush ( QBrush ( Qt::NoBrush ) );
256 scene->render(&painter, m_paper->rect(), m_paper->rect(), Qt::IgnoreAspectRatio);
257 painter.end();
258
259 QDomDocument domDoc;
260 if (!domDoc.setContent(byteArray))
261 {
262 qDebug() << "Error : Impossible to load the SVG content in the QDomDocument.";
263 return QDomDocument();
264 }
265
266 buffer.close();
267 return domDoc;
268}

271/**
272 * @brief Render each component of a piece separately and append the tagged groups to the piece group.
273 *
274 * Qt's QSvgGenerator creates <g> elements from painter state changes, not from
275 * graphics items, so a single render of the whole piece cannot be mapped back to
276 * its components. Instead each direct child of the piece root (seamline, cutline,
277 * notches, internal paths, labels, grainline) is rendered in its own pass with
278 * every sibling hidden. Each pass therefore produces exactly one <g> that belongs
279 * to one known component, which is tagged with the SVG data-* attributes:
280 * data-type, data-type-number (per-type counter within the piece), data-parent
281 * (the piece id) and a structured unique id ("<pieceId>-<type>-<n>").
282 *
283 * @param scene scene containing (only) this piece's item tree.
284 * @param item piece root item whose children are the components.
285 * @param pieceDoc DOM document of the piece; receives the component groups.
286 * @param pieceGroup piece <g> element the component groups are appended to.
287 * @param pieceId SVG id of the piece group, used to build data-parent and component ids.
288 */
289void SvgGenerator::addComponentGroups(QGraphicsScene *scene, QGraphicsItem *item, QDomDocument &pieceDoc,
290 QDomElement &pieceGroup, const QString &pieceId)
291{
292 const QList<QGraphicsItem *> components = item->childItems();
293 QHash<QString, int> typeCounters; // per data-type counter within this piece
294
295 for (int i = 0; i < components.size(); ++i)
296 {
297 // Show only the current component; hide every sibling.
298 for (int j = 0; j < components.size(); ++j)
299 {
300 components.at(j)->setVisible(i == j);
301 }
302
303 QDomDocument componentDoc = renderSceneToDom(scene);
304 if (componentDoc.isNull())
305 {
306 continue;
307 }
308
309 QDomNodeList groups = componentDoc.documentElement().elementsByTagName("g");
310 if (groups.isEmpty())
311 {
312 continue;
313 }
314
315 QDomElement componentGroup = groups.at(0).toElement();
316 cleanSvg(componentGroup);
317 if (componentGroup.childNodes().isEmpty())
318 {
319 // Component painted nothing (e.g. a piece without notches) — skip it.
320 continue;
321 }
322
323 // Tag the group so downstream tools (SeamlyLayout) can identify the component.
324 QString type = components.at(i)->data(PieceItemData::ItemType).toString();
325 if (type.isEmpty())
326 {
327 type = QStringLiteral("unknown");
328 }
329 const int typeNumber = ++typeCounters[type];
330 componentGroup.setAttribute("id", QString("%1-%2-%3").arg(pieceId, type).arg(typeNumber));
331 componentGroup.setAttribute("data-type", type);
332 componentGroup.setAttribute("data-type-number", QString::number(typeNumber));
333 componentGroup.setAttribute("data-parent", pieceId);
334
335 pieceGroup.appendChild(pieceDoc.importNode(componentGroup, true));
336 }
337
338 // Restore visibility so later exports (PDF, DXF, previews) see the whole piece.
339 for (int j = 0; j < components.size(); ++j)
340 {
341 components.at(j)->setVisible(true);
342 }
343}

346/**
347 * @brief Add a new SVG to the list of SVGs to be merged into a single SVG.
348 *
349 * When a piece item is given, the piece becomes a <g> element carrying the SVG
350 * data-* attributes (data-type="piece", data-type-number, data-parent,
351 * data-name and data-letter) and each of the piece's components is rendered
352 * and tagged individually (see addComponentGroups()). Without an item the
353 * scene is rendered as-is in a single untagged group (whole-scene exports,
354 * e.g. draft blocks).
355 *
356 * @param scene the scene that must be converted to SVG.
357 * @param item piece root item the scene contains; nullptr for whole-scene exports.
358 */
359void SvgGenerator::addSvgFromScene(QGraphicsScene *scene, QGraphicsItem *item)
360{
361 if (item == nullptr)
362 {
363 // Whole-scene export: keep the legacy single-group behavior.
364 QDomDocument domDoc = renderSceneToDom(scene);
365 if (!domDoc.isNull())
366 {
367 m_domList.append(domDoc);
368 }
369 return;
370 }
371
372 // One piece: build "<g id='piece-<n>' data-type='piece' ...>" holding one tagged group per component.
373 ++m_pieceCount;
374 const QString pieceId = QString("piece-%1").arg(m_pieceCount);
375
376 // Full render only to obtain a document with the correct <svg> root attributes.
377 QDomDocument pieceDoc = renderSceneToDom(scene);
378 if (pieceDoc.isNull())
379 {
380 return;
381 }
382
383 QDomElement svgRoot = pieceDoc.documentElement();
384 QDomNodeList rootGroups = svgRoot.elementsByTagName("g");
385 if (!rootGroups.isEmpty())
386 {
387 // Drop the untagged full-piece group; it is replaced by tagged component groups.
388 rootGroups.at(0).parentNode().removeChild(rootGroups.at(0));
389 }
390
391 QDomElement pieceGroup = pieceDoc.createElement("g");
392 pieceGroup.setAttribute("id", pieceId);
393 pieceGroup.setAttribute("data-type", "piece");
394 pieceGroup.setAttribute("data-type-number", QString::number(m_pieceCount));
395 pieceGroup.setAttribute("data-parent", "pattern-1");
396 setAttribute(pieceGroup, "data-name", item->data(PieceItemData::ObjectName).toString());
397 setAttribute(pieceGroup, "data-letter", item->data(PieceItemData::PieceLetter).toString());
398 svgRoot.appendChild(pieceGroup);
399
400 addComponentGroups(scene, item, pieceDoc, pieceGroup, pieceId);
401
402 m_domList.append(pieceDoc);
403}

407/**
408 * @brief Generate the merged SVG where each previously given scene is grouped separately.
409 * @return void
410 * @details This function merges the SVGs of the m_domList list and writes the result
411 in a file at the path given in the constructor.
412*/
413void SvgGenerator::generate()
414{
415 QDomDocument mergedSvg = mergeSvgDoms();
416
417 QFile outputFile(m_filepath);
418 if (!outputFile.open(QIODevice::WriteOnly | QIODevice::Text)) {
419 qDebug() << "Error : Couldn't write the output file.";
420 return;
421 }
422
423 QTextStream stream(&outputFile);
424 stream << mergedSvg.toString();
425 outputFile.close();
426
427 qDebug() << "Merged SVG Generated!";
428}
