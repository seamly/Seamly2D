41// QZipWriter and QZipReader are Qt private API. They moved from QtGui to QtCore in Qt 6, and
42// the build pins Qt 6.11.1, so the header path is fixed for as long as that pin holds. The
43// alternatives were a third-party zip library or a hand-written store-only writer; neither is
44// worth a dependency or 200 lines for one first-run backup. vmisc.pro carries the matching

68 /**
69 * @brief hashOfFile streams a file through SHA-256 without holding it all in memory.
70 * @return the digest, or an empty array when the file could not be read.
71 */
72 QByteArray hashOfFile(const QString &path)
73 {
74 QFile file(path);
75 if (!file.open(QIODevice::ReadOnly))
76 {
77 return QByteArray();
78 }
79
80 QCryptographicHash hash(QCryptographicHash::Sha256);
81 while (!file.atEnd())
82 {
83 const QByteArray chunk = file.read(hashChunkSize);
84 if (chunk.isEmpty() && !file.atEnd())
85 {
86 return QByteArray();
87 }
88 hash.addData(chunk);
89 }
90 return hash.result();
91 }

94 /**
95 * @brief relativeEntries lists a tree as .zip entry names.
96 *
97 * Directories come back too, so an empty folder survives the round trip. Entry names use
98 * forward slashes and no leading slash, which is what the .zip format requires.
99 *
100 * @param root tree to walk.
101 * @param files out-parameter, relative names of the regular files.
102 * @param directories out-parameter, relative names of the directories.
103 * @param errorMessage out-parameter, set when a symbolic link was found.
104 * @return false when the tree holds a symbolic link, which cannot be archived.
105 */
106 bool relativeEntries(const QString &root, QStringList *files, QStringList *directories, QString *errorMessage)
107 {
108 const QDir base(root);
109 const QDir::Filters filters = QDir::Files | QDir::Dirs | QDir::Hidden | QDir::System | QDir::NoDotAndDotDot;
110
111 QDirIterator iterator(root, filters, QDirIterator::Subdirectories);
112 while (iterator.hasNext())
113 {
114 const QString entryPath = iterator.next();
115 const QFileInfo entry(entryPath);
116
117 // A .zip entry cannot reproduce a link, and a link that quietly resolved outside
118 // the tree would make the backup incomplete in a way nobody could see.
119 //
120 // isSymbolicLink() and isJunction(), NOT isSymLink(): the last one also reports
121 // a Windows .lnk shortcut, which is an ordinary file holding ordinary bytes. A
122 // user who drops a shortcut in their patterns folder must not block the backup.
123 if (entry.isSymbolicLink() || entry.isJunction())
124 {
125 return fail(errorMessage, QStringLiteral("'%1' is a symbolic link")
126 .arg(QDir::toNativeSeparators(entryPath)));
127 }
128
129 const QString relative = base.relativeFilePath(entryPath);
130 if (entry.isDir())
131 {
132 directories->append(relative);
133 }
134 else
135 {
136 files->append(relative);
137 }
138 }
139 return true;
140 }

144/**
145 * @brief archivePath names the backup .zip, avoiding any file already there.
146 *
147 * The timestamp makes the name meaningful to a person browsing the folder, and makes a second
148 * migration on the same machine a second file rather than an overwrite.
149 *
150 * @param destinationRoot folder the archive is written into, normally the new data root.
151 * @param when timestamp to put in the name; the caller passes it so tests can fix it.
152 * @return absolute path of a file that does not exist yet.
153 */
154QString LegacyDataArchive::archivePath(const QString &destinationRoot, const QDateTime &when)
155{
156 const QString root = cleaned(destinationRoot);
157 const QString stamp = when.toString(QStringLiteral("yyyyMMdd-HHmmss"));
158 const QString base = root + QStringLiteral("/seamly2d-backup-") + stamp;
159
160 QString candidate = base + QStringLiteral(".zip");
161 for (int suffix = 2; QFileInfo::exists(candidate); ++suffix)
162 {
163 candidate = base + QStringLiteral("-%1.zip").arg(suffix);
164 }
165 return candidate;
166}

169/**
170 * @brief create writes every file and folder below sourceRoot into one .zip.
171 *
172 * A partial archive is never left behind: any failure removes the file before returning, so
173 * the caller cannot mistake a truncated archive for a good one.
174 *
175 * @param sourceRoot tree to archive; must be an existing directory.
176 * @param archiveFile .zip to write; its parent directory must already exist.
177 * @param errorMessage out-parameter, set on failure; pass null when the caller does not care.
178 * @return true when the archive was written and closed without error.
179 */
180bool LegacyDataArchive::create(const QString &sourceRoot, const QString &archiveFile, QString *errorMessage)
181{
182 const QString source = cleaned(sourceRoot);
183 const QString archive = cleaned(archiveFile);
184
185 if (source.isEmpty() || !QFileInfo(source).isDir())
186 {
187 return fail(errorMessage, QStringLiteral("'%1' is not a directory").arg(QDir::toNativeSeparators(source)));
188 }
189
190 // Writing the archive into the tree it is archiving would either capture the archive in
191 // itself or grow without end.
192 if (archive.startsWith(source + QLatin1Char('/'), Qt::CaseInsensitive))
193 {
194 return fail(errorMessage, QStringLiteral("the archive would sit inside the tree it archives"));
195 }
196
197 QStringList files;
198 QStringList directories;
199 if (!relativeEntries(source, &files, &directories, errorMessage))
200 {
201 return false;
202 }
203
204 QZipWriter writer(archive);
205 if (!writer.isWritable())
206 {
207 return fail(errorMessage, QStringLiteral("could not open '%1' for writing")
208 .arg(QDir::toNativeSeparators(archive)));
209 }
210
211 for (const QString &directory : qAsConst(directories))
212 {
213 writer.addDirectory(directory);
214 }
215
216 for (const QString &relative : qAsConst(files))
217 {
218 QFile file(source + QLatin1Char('/') + relative);
219 if (!file.open(QIODevice::ReadOnly))
220 {
221 writer.close();
222 QFile::remove(archive);
223 return fail(errorMessage, QStringLiteral("could not read '%1'").arg(relative));
224 }
225
226 // The QIODevice overload streams, so one huge file does not become one huge
227 // allocation.
228 writer.addFile(relative, &file);
229 file.close();
230
231 if (writer.status() != QZipWriter::NoError)
232 {
233 const int status = static_cast<int>(writer.status());
234 writer.close();
235 QFile::remove(archive);
236 return fail(errorMessage, QStringLiteral("writing '%1' failed with status %2").arg(relative).arg(status));
237 }
238 }
239
240 writer.close();
241 if (writer.status() != QZipWriter::NoError)
242 {
243 const int status = static_cast<int>(writer.status());
244 QFile::remove(archive);
245 return fail(errorMessage, QStringLiteral("closing the archive failed with status %1").arg(status));
246 }
247
248 return true;
249}

252/**
253 * @brief verifyAgainst proves the archive can be read back and holds the whole tree.
254 *
255 * The size and CRC in a .zip central directory describe what the writer *meant* to store, so
256 * they are not enough to justify calling the backup good. Every entry is decompressed and its
257 * SHA-256 compared with the file it came from.
258 *
259 * @param sourceRoot tree the archive was made from.
260 * @param archiveFile .zip to check.
261 * @param errorMessage out-parameter, naming the first entry that did not match.
262 * @return true when every file and folder in the tree is in the archive, byte for byte.
263 */
264bool LegacyDataArchive::verifyAgainst(const QString &sourceRoot, const QString &archiveFile, QString *errorMessage)
265{
266 const QString source = cleaned(sourceRoot);
267 const QString archive = cleaned(archiveFile);
268
269 QStringList files;
270 QStringList directories;
271 if (!relativeEntries(source, &files, &directories, errorMessage))
272 {
273 return false;
274 }
275
276 QZipReader reader(archive);
277 if (!reader.isReadable() || reader.status() != QZipReader::NoError)
278 {
279 return fail(errorMessage, QStringLiteral("could not open '%1' for reading")
280 .arg(QDir::toNativeSeparators(archive)));
281 }
282
283 QHash<QString, QZipReader::FileInfo> entries;
284 const QList<QZipReader::FileInfo> infoList = reader.fileInfoList();
285 for (const QZipReader::FileInfo &info : infoList)
286 {
287 entries.insert(info.filePath, info);
288 }
289
290 for (const QString &relative : qAsConst(files))
291 {
292 const auto entry = entries.constFind(relative);
293 if (entry == entries.constEnd() || !entry->isFile)
294 {
295 return fail(errorMessage, QStringLiteral("'%1' is missing from the archive").arg(relative));
296 }
297
298 const QString filePath = source + QLatin1Char('/') + relative;
299 const qint64 sourceSize = QFileInfo(filePath).size();
300 if (entry->size != sourceSize)
301 {
302 return fail(errorMessage, QStringLiteral("'%1' is %2 bytes in the archive, %3 on disk")
303 .arg(relative)
304 .arg(entry->size)
305 .arg(sourceSize));
306 }
307
308 const QByteArray stored = reader.fileData(relative);
309 if (static_cast<qint64>(stored.size()) != sourceSize)
310 {
311 return fail(errorMessage, QStringLiteral("'%1' did not decompress to its recorded size").arg(relative));
312 }
313
314 const QByteArray sourceHash = hashOfFile(filePath);
315 if (sourceHash.isEmpty())
316 {
317 return fail(errorMessage, QStringLiteral("could not re-read '%1'").arg(relative));
318 }
319 if (QCryptographicHash::hash(stored, QCryptographicHash::Sha256) != sourceHash)
320 {
321 return fail(errorMessage, QStringLiteral("'%1' does not match the file it came from").arg(relative));
322 }
323 }
324
325 // QZipWriter::addDirectory() stores a trailing slash; accept both spellings so the check
326 // does not depend on that detail.
327 for (const QString &relative : qAsConst(directories))
328 {
329 if (!entries.contains(relative) && !entries.contains(relative + QLatin1Char('/')))
330 {
331 return fail(errorMessage, QStringLiteral("folder '%1' is missing from the archive").arg(relative));
332 }
333 }
334
335 return true;
336}

339/**
340 * @brief archive is the whole sequence: write the .zip, then read it back and verify it.
341 *
342 * The source tree is never touched. A failed verification takes the half-good archive with
343 * it, so a bad backup can never be mistaken for a good one.
344 *
345 * @param sourceRoot tree to back up, normally ~/seamly2d.
346 * @param destinationRoot folder to write the archive into, normally the new data root.
347 * @param errorMessage out-parameter, set on failure; pass null when the caller does not care.
348 * @return absolute path of the verified archive, or an empty string on failure.
349 */
350QString LegacyDataArchive::archive(const QString &sourceRoot, const QString &destinationRoot, QString *errorMessage)
351{
352 const QString source = cleaned(sourceRoot);
353 const QString destination = cleaned(destinationRoot);
354
355 if (source.isEmpty() || !QFileInfo(source).isDir())
356 {
357 fail(errorMessage, QStringLiteral("'%1' is not a directory").arg(QDir::toNativeSeparators(source)));
358 return QString();
359 }
360 if (destination.isEmpty() || !QFileInfo(destination).isDir())
361 {
362 fail(errorMessage, QStringLiteral("'%1' is not a directory").arg(QDir::toNativeSeparators(destination)));
363 return QString();
364 }
365
366 // A destination inside the source would try to archive the .zip into itself.
367 if (destination.compare(source, Qt::CaseInsensitive) == 0 ||
368 destination.startsWith(source + QLatin1Char('/'), Qt::CaseInsensitive))
369 {
370 fail(errorMessage, QStringLiteral("the destination is inside the tree being archived"));
371 return QString();
372 }
373
374 const QString archiveFile = archivePath(destination, QDateTime::currentDateTime());
375 if (!create(source, archiveFile, errorMessage))
376 {
377 return QString();
378 }
379
380 if (!verifyAgainst(source, archiveFile, errorMessage))
381 {
382 QFile::remove(archiveFile);
383 return QString();
384 }
385
386 return archiveFile;
