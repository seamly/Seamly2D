5// @file PieceOverlayItem.h
6// @brief QGraphicsRectItem subclass representing one pattern piece overlay in the adjust canvas.
7//
8// Handles drag-to-move (using scene coordinates for correct delta tracking),
9// context-menu rotation, and selection highlight. No coordinate-space feedback
10// loop: drag delta is always computed from scenePos() which is stable regardless
11// of the item's own position.
12

21/// @class PieceOverlayItem
22/// @brief Interactive graphics item representing a single pattern piece overlay.
23///
24/// Renders a labeled rectangle for one piece, supports drag-to-move and
25/// right-click context-menu rotation (90 CW, 90 CCW, 180, reset).
26/// Movement is tracked entirely in scene coordinates so that Qt's item
27/// transform never creates a feedback loop with the drag delta.
28class PieceOverlayItem : public QGraphicsRectItem
29{
30public:
31 struct TransformState
32 {
33 QPointF pos;
34 double rotation = 0.0;
35 double scaleX = 1.0; ///< Horizontal flip: -1.0 when flipped, 1.0 otherwise.
36 double scaleY = 1.0; ///< Vertical flip: -1.0 when flipped, 1.0 otherwise.
37 QString transform;
38 };
39
40 struct HistorySnapshot
41 {
42 QVector<TransformState> history;
43 int historyIndex = -1;
44 };

52 /// @brief Set the human-readable name shown for this piece.
53 /// @details Sourced from the layout's bbox JSON `label` field, which the Rust
54 /// side derives from the Seamly2D handoff attributes in the order
55 /// `data-name` → `data-letter` → `id`. Display only — the piece's
56 /// identity for history and transform bookkeeping stays `m_id`.
57 /// An empty string is ignored so the id remains the fallback.
58 /// @param label Display label, e.g. "Front Bodice".
59 void setDisplayLabel(const QString& label) { if (!label.isEmpty()) { m_displayLabel = label; } }
