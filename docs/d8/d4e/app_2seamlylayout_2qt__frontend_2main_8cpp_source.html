62// @brief Set the Windows title bar color using DWM API.
63// @param window The QQuickWindow to style.
64// @param color The color in COLORREF format (0x00BBGGRR).
65void setWindowsTitleBarColor(QQuickWindow *window, COLORREF color)
66{
67 if (!window) return;
68 HWND hwnd = reinterpret_cast<HWND>(window->winId());
69 DwmSetWindowAttribute(hwnd, DWMWA_CAPTION_COLOR, &color, sizeof(color));
70} // setWindowsTitleBarColor
71#endif // Q_OS_WIN
72
73#ifdef Q_OS_MACOS
74#include <Cocoa/Cocoa.h>
75
76// @brief Set the macOS title bar color using Cocoa APIs.
77// @param window The QQuickWindow to style.
78// @param r Red component (0-255).
79// @param g Green component (0-255).
80// @param b Blue component (0-255).
81void setMacOSTitleBarColor(QQuickWindow *window, int r, int g, int b)
82{
83 if (!window) return;
84
85 NSView *nsView = reinterpret_cast<NSView *>(window->winId());
86 if (!nsView) return;
87
88 NSWindow *nsWindow = [nsView window];
89 if (!nsWindow) return;
90
91 // Make titlebar blend with content
92 [nsWindow setTitlebarAppearsTransparent:YES];
93
94 // Set the window background color (affects titlebar when transparent)
95 NSColor *color = [NSColor colorWithCalibratedRed:r/255.0
96 green:g/255.0
97 blue:b/255.0
98 alpha:1.0];
99 [nsWindow setBackgroundColor:color];
100
101 // Use dark appearance for light title text
102 [nsWindow setAppearance:[NSAppearance appearanceNamed:NSAppearanceNameVibrantDark]];
103} // setMacOSTitleBarColor
104#endif // Q_OS_MACOS

129 // Enable and open debug log file.

131 // Set to false (default) to disable all logging with zero overhead.
132 Logger::debugEnabled = true;
133 Logger::init();

136 // Application metadata — used by QSettings, About dialogs, OS task managers
137 //
138 // Task 15: organization renamed from "Seamly Systems" to "Seamly" so
139 // QStandardPaths::AppConfigLocation resolves under the same shared "Seamly"
140 // organization folder as seamly2d/seamlyme (AppData/Local/Seamly/SeamlyLayout
141 // on Windows). PreferencesModel::appConfigRootPath()'s first-run migration
142 // bridges data forward from the old "Seamly Systems" folder automatically.
143 app.setOrganizationName("Seamly");
144 app.setOrganizationDomain("seamly.io");
145 app.setApplicationName("SeamlyLayout");
146 app.setApplicationVersion("0.1.0");

148 // Application icon — SeamlyLayout logo (SVG scales to all resolutions)

151 // Quick Controls 2 style — Fusion as cross-platform base;

155 // Register CXX-Qt bridge types with the QML engine before loading any QML.
156 // qmltyperegistrar cannot process AppController's generated header through
157 // CXX-Qt's template include chain, so registration is done here at runtime.
158 // The QMetaObject used here comes from the moc output already compiled into

195 // Load the root QML component from the registered SeamlyLayout module.
196 //
197 // Note: QQmlApplicationEngine internally creates a QQuickWindow as part of
198 // loading Main.qml, and Main.qml itself declares ApplicationWindow (which
199 // is a QQuickWindowQmlImpl subclass of QQuickWindow). This means
200 // xChanged(int) and yChanged(int) are registered once on QQuickWindow by Qt
201 // and again on QQuickWindowQmlImpl — producing four console messages at startup:
202 //
203 // QMetaObject::indexOfSignal: signal xChanged(int) from QQuickWindow
204 // redefined in QQuickWindowQmlImpl
205 // QMetaObject::indexOfSignal: signal yChanged(int) from QQuickWindow
206 // redefined in QQuickWindowQmlImpl
207 // (repeated once more for the second engine pass)
208 //
209 // These messages are Qt-internal noise from the QML meta-object system; they
210 // do not indicate a bug and have no effect on behaviour. They will disappear
211 // if Qt ever deduplicates the signal declarations between the two classes.
212 // Safe to ignore.
213 Logger::log(QStringLiteral("main(): loading QML module SeamlyLayout::Main"));
214 engine.loadFromModule(QStringLiteral("SeamlyLayout"), QStringLiteral("Main"));
215
216#ifdef Q_OS_WIN
217 // Set Windows title bar color to violetMedium (#7351ad)
218 // COLORREF format is 0x00BBGGRR, so #7351ad becomes 0x00ad5173
219 QObject::connect(&engine, &QQmlApplicationEngine::objectCreated, &app,
220 [](QObject *obj, const QUrl &) {
221 if (auto *window = qobject_cast<QQuickWindow *>(obj)) {
222 setWindowsTitleBarColor(window, 0x00ad5173);
223 } // if window cast succeeded
224 }, Qt::QueuedConnection);
225#endif
226
227#ifdef Q_OS_MACOS
228 // Set macOS title bar color to violetMedium (#7351ad = RGB 115, 81, 173)

230 [](QObject *obj, const QUrl &) {
231 if (auto *window = qobject_cast<QQuickWindow *>(obj)) {
232 setMacOSTitleBarColor(window, 115, 81, 173);

235#endif
236
237 return app.exec();
