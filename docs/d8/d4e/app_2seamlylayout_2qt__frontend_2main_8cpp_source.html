70// @brief Set the Windows title bar color using DWM API.
71// @param window The QQuickWindow to style.
72// @param color The color in COLORREF format (0x00BBGGRR).
73void setWindowsTitleBarColor(QQuickWindow *window, COLORREF color)
74{
75 if (!window) return;
76 HWND hwnd = reinterpret_cast<HWND>(window->winId());
77 DwmSetWindowAttribute(hwnd, DWMWA_CAPTION_COLOR, &color, sizeof(color));
78} // setWindowsTitleBarColor
79#endif // Q_OS_WIN

81#ifdef Q_OS_MACOS
82#include <Cocoa/Cocoa.h>

84// @brief Set the macOS title bar color using Cocoa APIs.
85// @param window The QQuickWindow to style.
86// @param r Red component (0-255).
87// @param g Green component (0-255).
88// @param b Blue component (0-255).
89void setMacOSTitleBarColor(QQuickWindow *window, int r, int g, int b)
90{
91 if (!window) return;
92
93 NSView *nsView = reinterpret_cast<NSView *>(window->winId());
94 if (!nsView) return;
95
96 NSWindow *nsWindow = [nsView window];
97 if (!nsWindow) return;
98
99 // Make titlebar blend with content
100 [nsWindow setTitlebarAppearsTransparent:YES];
101
102 // Set the window background color (affects titlebar when transparent)
103 NSColor *color = [NSColor colorWithCalibratedRed:r/255.0
104 green:g/255.0
105 blue:b/255.0
106 alpha:1.0];
107 [nsWindow setBackgroundColor:color];
108
109 // Use dark appearance for light title text
110 [nsWindow setAppearance:[NSAppearance appearanceNamed:NSAppearanceNameVibrantDark]];
111} // setMacOSTitleBarColor
112#endif // Q_OS_MACOS

115// The appearance depends on the desktop environment (GNOME, KDE, XFCE, etc.)
116// and its theme settings. For consistent branding on Linux, users should:
117// - Use GNOME Tweaks or similar to set window/titlebar colors, or
118// - The app could use a frameless window with custom titlebar (future feature)

120// @brief Application entry point.
121// @param argc Argument count from the OS.
122// @param argv Argument vector from the OS.
123// @return Exit code: 0 on clean exit, non-zero on error.
124int main(int argc, char *argv[])
125{
126 // Initialise Qt WebEngine — must be called before QApplication
127 QtWebEngineQuick::initialize();
128
129 // Initialise the Qt application; QApplication is a subclass of QGuiApplication
130 // and supports both QML Quick windows and QtWidgets windows (e.g. AdjustWindow)
131 // in the same process without any additional configuration.
132 QApplication app(argc, argv);
133
134 // Detect host OS once — use Platform::os throughout the application
135 Platform::init();
136
137 // Enable and open debug log file.
138 // Set debugEnabled = true to write debug lines to logs/log_{YYMMDDHHMM}.txt.
139 // Set to false (default) to disable all logging with zero overhead.
140 Logger::debugEnabled = true;
141 Logger::init();
142 qInstallMessageHandler(Logger::messageHandler);
143
144 // Application metadata — used by QSettings, About dialogs, OS task managers
145 //
146 // Task 15: organization renamed from "Seamly Systems" to "Seamly" so
147 // QStandardPaths::AppConfigLocation resolves under the same shared "Seamly"
148 // organization folder as seamly2d/seamlyme (AppData/Local/Seamly/SeamlyLayout
149 // on Windows). PreferencesModel::appConfigRootPath()'s first-run migration
150 // bridges data forward from the old "Seamly Systems" folder automatically.
151 app.setOrganizationName("Seamly");
152 app.setOrganizationDomain("seamly.io");
153 app.setApplicationName("SeamlyLayout");
154 app.setApplicationVersion("0.1.0");
155
156 // -----------------------------------------------------------------------
157 // Command line — Task 49: the Seamly2D Layout Mode handoff.
158 //
159 // Seamly2D writes "<pattern>.pieces.svg" beside the pattern file and then
160 // launches this application detached with that path as its single
161 // positional argument. Parsing happens here, AFTER the metadata above, so
162 // --version can report the application name and version set there, and
163 // BEFORE the QML engine is built, so --help / --version cost no startup.
164 //
165 // The parsed result is only dispatched once the event loop is running (see
166 // the QTimer::singleShot below): the QML window and its WebEngine canvases
167 // must exist before an SVG can be pushed into them.
168 // -----------------------------------------------------------------------
169 const StartupOptions startupOptions = StartupOptions::parse(app.arguments());
170
171 if (startupOptions.status() == StartupOptions::Status::ShowInformation) {
172 // --help / --version. This is a WIN32-subsystem binary with no console
173 // to print to, so the text goes in a dialog on every platform.
174 QMessageBox::information(nullptr, QStringLiteral("SeamlyLayout"), startupOptions.message());
175 return 0;
176 } // if information requested
177
178 Logger::log(QStringLiteral("main(): startup file = '%1', message = '%2'")
179 .arg(startupOptions.filePath(), startupOptions.message()));
180
181 // Application icon — SeamlyLayout logo (SVG scales to all resolutions)
182 app.setWindowIcon(QIcon(QStringLiteral(":/icons/seamly-layout.svg")));
183
184 // Quick Controls 2 style — Fusion as cross-platform base;
185 // branding colours are applied via Theme.qml overrides in QML
186 QQuickStyle::setStyle(QStringLiteral("Fusion"));
187
188 // Register CXX-Qt bridge types with the QML engine before loading any QML.
189 // qmltyperegistrar cannot process AppController's generated header through
190 // CXX-Qt's template include chain, so registration is done here at runtime.
191 // The QMetaObject used here comes from the moc output already compiled into
192 // the cxxqt_bridge staticlib — no duplicate moc output is generated.
193 qmlRegisterType<AppController>("SeamlyLayout", 1, 0, "AppController");
194 Logger::log(QStringLiteral("main(): registered AppController with QML engine"));
195
196 // Register SettingsModel with the QML engine.
197 // qmltyperegistrar cannot resolve SettingsModel through cmake SOURCES
198 // automatically, so registration is done here at runtime.
199 qmlRegisterType<SettingsModel>("SeamlyLayout", 1, 0, "SettingsModel");
200 Logger::log(QStringLiteral("main(): registered SettingsModel with QML engine"));
201
202 // Register PreferencesModel with the QML engine.
203 qmlRegisterType<PreferencesModel>("SeamlyLayout", 1, 0, "PreferencesModel");
204 Logger::log(QStringLiteral("main(): registered PreferencesModel with QML engine"));
205
206 // Register AdjustController with the QML engine.
207 // Owns the QtWidgets AdjustWindow; QML calls launchAdjustWindow() to open it.
208 qmlRegisterType<AdjustController>("SeamlyLayout", 1, 0, "AdjustController");
209 Logger::log(QStringLiteral("main(): registered AdjustController with QML engine"));
210
211 // Register PreferencesController with the QML engine.
212 // Owns the QtWidgets PreferencesWindow; QML calls openPreferences() to show it.
213 qmlRegisterType<PreferencesController>("SeamlyLayout", 1, 0, "PreferencesController");
214 Logger::log(QStringLiteral("main(): registered PreferencesController with QML engine"));
215
216 // QML application engine — loads the SeamlyLayout QML module
217 QQmlApplicationEngine engine;
218
219 // Abort if the root QML object fails to load
220 QObject::connect(
221 &engine,
222 &QQmlApplicationEngine::objectCreationFailed,
223 &app,
224 []() { QCoreApplication::exit(-1); },
225 Qt::QueuedConnection
226 );
227
228 // Load the root QML component from the registered SeamlyLayout module.
229 //
230 // Note: QQmlApplicationEngine internally creates a QQuickWindow as part of
231 // loading Main.qml, and Main.qml itself declares ApplicationWindow (which
232 // is a QQuickWindowQmlImpl subclass of QQuickWindow). This means
233 // xChanged(int) and yChanged(int) are registered once on QQuickWindow by Qt
234 // and again on QQuickWindowQmlImpl — producing four console messages at startup:
235 //
236 // QMetaObject::indexOfSignal: signal xChanged(int) from QQuickWindow
237 // redefined in QQuickWindowQmlImpl
238 // QMetaObject::indexOfSignal: signal yChanged(int) from QQuickWindow
239 // redefined in QQuickWindowQmlImpl
240 // (repeated once more for the second engine pass)
241 //
242 // These messages are Qt-internal noise from the QML meta-object system; they
243 // do not indicate a bug and have no effect on behaviour. They will disappear
244 // if Qt ever deduplicates the signal declarations between the two classes.
245 // Safe to ignore.
246 Logger::log(QStringLiteral("main(): loading QML module SeamlyLayout::Main"));
247 engine.loadFromModule(QStringLiteral("SeamlyLayout"), QStringLiteral("Main"));
248
249#ifdef Q_OS_WIN
250 // Set Windows title bar color to violetMedium (#7351ad)
251 // COLORREF format is 0x00BBGGRR, so #7351ad becomes 0x00ad5173
252 QObject::connect(&engine, &QQmlApplicationEngine::objectCreated, &app,
253 [](QObject *obj, const QUrl &) {
254 if (auto *window = qobject_cast<QQuickWindow *>(obj)) {
255 setWindowsTitleBarColor(window, 0x00ad5173);
256 } // if window cast succeeded
257 }, Qt::QueuedConnection);
258#endif
259
260#ifdef Q_OS_MACOS
261 // Set macOS title bar color to violetMedium (#7351ad = RGB 115, 81, 173)
262 QObject::connect(&engine, &QQmlApplicationEngine::objectCreated, &app,
263 [](QObject *obj, const QUrl &) {
264 if (auto *window = qobject_cast<QQuickWindow *>(obj)) {
265 setMacOSTitleBarColor(window, 115, 81, 173);
266 } // if window cast succeeded
267 }, Qt::QueuedConnection);
268#endif
269
270 // -----------------------------------------------------------------------
271 // Task 49 — act on the command line once the event loop is running.
272 //
273 // A zero-delay single shot fires on the first pass of the event loop, by
274 // which time Main.qml's root ApplicationWindow (and the two SvgCanvas
275 // WebEngineViews inside it) have been constructed and shown. Calling the
276 // QML functions directly after loadFromModule() would push the SVG into a
277 // canvas whose web view has not yet been realised.
278 //
279 // The lambda captures `engine` by reference: it lives on this stack frame
280 // for the whole of app.exec(), which is the only time the timer can fire.
281 // -----------------------------------------------------------------------
282 if (startupOptions.hasFile() || startupOptions.hasError()) {
283 QTimer::singleShot(0, &app, [&engine, startupOptions]() {
284 const QList<QObject *> rootObjects = engine.rootObjects();
285 if (rootObjects.isEmpty()) {
286 // QML failed to load; objectCreationFailed above is already
287 // tearing the application down. Nothing to hand the file to.
288 Logger::log(QStringLiteral("main(): no QML root object — startup file dropped"));
289 return;
290 } // if rootObjects.isEmpty()
291
292 QObject *const window = rootObjects.constFirst();
293
294 if (startupOptions.hasFile()) {
295 // Main.qml: function openSvgFile(localPath) — the same entry
296 // point the Import SVG file dialog uses, so the handoff and a
297 // manual import cannot diverge.
298 Logger::log(QStringLiteral("main(): opening startup file %1").arg(startupOptions.filePath()));
299 QMetaObject::invokeMethod(window,
300 "openSvgFile",
301 Q_ARG(QVariant, QVariant(startupOptions.filePath())));
302 } else {
303 // Main.qml: function reportStartupError(message) — shows the
304 // error dialog over the empty canvas instead of leaving the
305 // user to guess why nothing opened.
306 Logger::log(QStringLiteral("main(): startup error: %1").arg(startupOptions.message()));
307 QMetaObject::invokeMethod(window,
308 "reportStartupError",
309 Q_ARG(QVariant, QVariant(startupOptions.message())));
310 } // if hasFile
311 }); // QTimer::singleShot
312 } // if a startup file or error is pending
313
314 return app.exec();
315} // main
