5// @file StartupOptions.h
6// @brief Command-line parsing for SeamlyLayout's single positional <svg-file>
7// argument, plus the validation of that file.
8//
9// This is the SeamlyLayout half of the Seamly2D Layout Mode handoff (Task 49).
10// Seamly2D writes "<pattern>.pieces.svg" beside the pattern file and launches
11//
12// SeamlyLayout <absolute path to .pieces.svg>
13//
14// detached (MainWindow::exportPiecesToSeamlyLayout(), which builds the argument
15// vector through SeamlySuitePaths::seamlyLayoutLaunchArguments()). Before this
16// class existed, main.cpp handed argc/argv to QApplication and read nothing
17// else, so the handoff opened an empty canvas and the user had to re-find the
18// file Seamly2D had just written.
19//
20// The class is a plain value type — no QObject, no Qt application instance
21// required — so the whole contract can be unit tested (see
22// src/test/SeamlyLayoutTest/StartupOptionsTests.cpp) without starting the GUI.
23//
24// Accepted argument forms:
25// SeamlyLayout start with an empty canvas
26// SeamlyLayout <file.svg> open that file
27// SeamlyLayout -h | --help show usage, exit 0
28// SeamlyLayout -v | --version show the version, exit 0
29//
30// Usage:
31// const StartupOptions options = StartupOptions::parse(QCoreApplication::arguments());
32// switch (options.status()) { ... }
33

39// @brief Parsed and validated SeamlyLayout command line.
40//
41// Produced by the static parse() factory; the four private members are set
42// once there and only read afterwards. status() says what the caller must do;
43// filePath() and message() are the payload for the respective statuses.
44class StartupOptions
45{
46public:
47 // @brief What main() should do with the parsed command line.
48 enum class Status
49 {
50 // No file was given — start normally with an empty canvas.
51 NoFile,
52 // filePath() holds an existing, readable SVG file to open at startup.
53 OpenFile,
54 // message() holds --help / --version text; show it and exit 0.
55 ShowInformation,
56 // message() holds the reason the command line could not be honoured
57 // (missing file, unreadable file, not an SVG, too many arguments).
58 // The application still starts, with an empty canvas, after showing it.
59 Failed
60 }; // enum class Status
61
62 // @brief Parse and validate a command line.
63 // @param arguments Full argument list including the program name at index 0,
64 // exactly as QCoreApplication::arguments() returns it.
65 // @return A fully populated StartupOptions; never throws, never exits.
66 static StartupOptions parse(const QStringList &arguments);
67
68 // @brief What the caller must do with this command line.
69 Status status() const { return m_status; }
70
71 // @brief Absolute path of the SVG to open. Empty unless status() is OpenFile.
72 QString filePath() const { return m_filePath; }
73
74 // @brief Text to show the user. Empty unless status() is ShowInformation or Failed.
75 QString message() const { return m_message; }
76
77 // @brief True when an SVG file should be opened at startup.
78 bool hasFile() const { return m_status == Status::OpenFile; }
79
80 // @brief True when the command line could not be honoured.
81 bool hasError() const { return m_status == Status::Failed; }
82
83private:
84 // Result of the parse; NoFile is the "nothing was asked of us" default.
85 Status m_status = Status::NoFile;
86
87 // Absolute path of the file to open; set only for Status::OpenFile.
88 QString m_filePath;
89
90 // User-visible text; set only for Status::ShowInformation and Status::Failed.
91 QString m_message;
92}; // class StartupOptions
