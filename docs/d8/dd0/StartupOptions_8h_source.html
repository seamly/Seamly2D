5// @file StartupOptions.h
6// @brief Command-line parsing for SeamlyLayout's two startup transports — an
7// optional positional <svg-file>, or an SVG document read from standard
8// input — plus the validation of each.
9//
10// This is the SeamlyLayout half of the Seamly2D Layout Mode handoff.
11//
12// Transport 1 — a file path (Task 49, still supported):
13//
14// SeamlyLayout <absolute path to some.svg>
15//
16// Anyone can use it: a shell, a desktop file association, or a script. It is
17// the only way to open an SVG that already exists on disk.
18//
19// Transport 2 — a document on standard input (Seamly2D.5, the handoff):
20//
21// SeamlyLayout --svg-stdin [--document-name <name>]
22//
23// Seamly2D serialises piece mode to one stringified SVG, launches SeamlyLayout
24// with --svg-stdin, writes that string to the child's standard input and closes
25// the channel. Nothing is written to disk, so no ".pieces.svg" is left beside
26// the pattern and a read-only pattern directory no longer blocks Layout Mode.
27// The seamly2d side builds the argument vector through
28// SeamlySuitePaths::seamlyLayoutLaunchArguments().
29//
30// --document-name carries the pattern base name, because a document with no
31// file has no name to derive default export file names from.
32//
33// The class is a plain value type — no QObject, no Qt application instance
34// required — so the whole contract can be unit tested (see
35// src/test/SeamlyLayoutTest/StartupOptionsTests.cpp) without starting the GUI.
36// The standard-input read is a separate parse() overload taking a QIODevice, so
37// a test can feed it a QBuffer.
38//
39// Accepted argument forms:
40// SeamlyLayout start with an empty canvas
41// SeamlyLayout <file.svg> open that file
42// SeamlyLayout --svg-stdin open the SVG document on standard input
43// SeamlyLayout -h | --help show usage, exit 0
44// SeamlyLayout -v | --version show the version, exit 0
45//
46// Usage:
47// QFile standardInput;
48// standardInput.open(stdin, QIODevice::ReadOnly);
49// const StartupOptions options =
50// StartupOptions::parse(QCoreApplication::arguments(), &standardInput);
51// switch (options.status()) { ... }
52

60// @brief Parsed and validated SeamlyLayout command line.
61//
62// Produced by the static parse() factories; the private members are set once
63// there and only read afterwards. status() says what the caller must do;
64// filePath(), svgDocument() and message() are the payload for the respective
65// statuses.
66class StartupOptions
67{
68public:
69 // @brief What main() should do with the parsed command line.
70 enum class Status
71 {
72 // No file and no document were given — start with an empty canvas.
73 NoFile,
74 // filePath() holds an existing, readable SVG file to open at startup.
75 OpenFile,
76 // svgDocument() holds an SVG document read from standard input.
77 OpenDocument,
78 // message() holds --help / --version text; show it and exit 0.
79 ShowInformation,
80 // message() holds the reason the command line could not be honoured
81 // (missing file, unreadable file, not an SVG, too many arguments,
82 // empty or unreadable standard input).
83 // The application still starts, with an empty canvas, after showing it.
84 Failed
85 }; // enum class Status
86
87 // @brief Parse and validate a command line that cannot use standard input.
88 // Equivalent to parse(arguments, nullptr): --svg-stdin fails, because no
89 // channel was offered to read the document from.
90 // @param arguments Full argument list including the program name at index 0,
91 // exactly as QCoreApplication::arguments() returns it.
92 // @return A fully populated StartupOptions; never throws, never exits.
93 static StartupOptions parse(const QStringList &arguments);
94
95 // @brief Parse and validate a command line, reading standard input when
96 // --svg-stdin asks for it.
97 // @param arguments Full argument list including the program name at index 0.
98 // @param standardInput Open, readable device holding the SVG document. Read
99 // to end only when --svg-stdin is set; may be nullptr otherwise.
100 // @return A fully populated StartupOptions; never throws, never exits.
101 static StartupOptions parse(const QStringList &arguments, QIODevice *standardInput);
102
103 // @brief What the caller must do with this command line.
104 Status status() const { return m_status; }
105
106 // @brief Absolute path of the SVG to open. Empty unless status() is OpenFile.
107 QString filePath() const { return m_filePath; }
108
109 // @brief The SVG document read from standard input. Empty unless status()

111 QString svgDocument() const { return m_svgDocument; }
112
113 // @brief Name to use for default export file names. Set only for
114 // OpenDocument, and only when --document-name was given.

117 // @brief Text to show the user. Empty unless status() is ShowInformation or Failed.
118 QString message() const { return m_message; }

120 // @brief True when an SVG file should be opened at startup.
121 bool hasFile() const { return m_status == Status::OpenFile; }
122
123 // @brief True when an SVG document from standard input should be opened.
124 bool hasDocument() const { return m_status == Status::OpenDocument; }
125
126 // @brief True when the command line could not be honoured.
127 bool hasError() const { return m_status == Status::Failed; }

129private:
130 // @brief Read and validate the SVG document offered on standard input.
131 // @param documentName Value of --document-name; empty when not given.
132 // @param standardInput Device read to end; nullptr is a failure, because
133 // --svg-stdin asked for a channel the caller did not supply.
134 // @return OpenDocument with svgDocument() set, or Failed with a message.
135 static StartupOptions parseStandardInput(const QString &documentName,
136 QIODevice *standardInput);
137
138 // Result of the parse; NoFile is the "nothing was asked of us" default.
139 Status m_status = Status::NoFile;
140
141 // Absolute path of the file to open; set only for Status::OpenFile.
142 QString m_filePath;
143
144 // Stringified SVG document; set only for Status::OpenDocument.
145 QString m_svgDocument;
146
147 // Pattern base name for default export names; set only for OpenDocument.
148 QString m_documentName;
149
150 // User-visible text; set only for Status::ShowInformation and Status::Failed.
151 QString m_message;
152}; // class StartupOptions
