38/// @brief Construct a PieceOverlayItem and configure its appearance and interaction flags.
39PieceOverlayItem::PieceOverlayItem(const QString& id,
40 double originX,
41 double originY,
42 double x,
43 double y,
44 double w,
45 double h,
46 double rotationDeg,
47 const QString& transformStr,
48 QGraphicsItem* parent)
49 : QGraphicsRectItem(0.0, 0.0, w, h, parent)
50 , m_id(id)
51 , m_originX(originX)
52 , m_originY(originY)
53 , m_w(w)
54 , m_h(h)
55 , m_initialPos(x, y)
56 , m_initialRotation(rotationDeg)
57 , m_basePos(x, y)
58 , m_baseRotation(rotationDeg)
59 , m_dragging(false)
60 , m_transformStr(transformStr)
61{
62 // Position the item in the scene at initial position.
63 setPos(x, y);
64
65 // Set rotation pivot at the bounding-box center, not the upper-left
66 // corner. Rotating about the center is the user-friendly behavior:
67 // the piece swings in place instead of swinging around a corner.
68 // buildTransform()/applyTransformString() compute the matching
69 // rotate-about-center SVG matrix so overlays and SVG pieces stay
70 // aligned after accept_adjustments().
71 setTransformOriginPoint(m_w / 2.0, m_h / 2.0);
72
73 // Every piece overlay must be interactive, regardless of whether it already
74 // has a persisted SVG transform string. Without this, the view's
75 // ScrollHandDrag mode steals left-drag gestures and context-menu rotation
76 // updates are never reflected on the overlay item itself.
77 setFlags(QGraphicsItem::ItemIsSelectable | QGraphicsItem::ItemIsFocusable);
78 setAcceptedMouseButtons(Qt::LeftButton | Qt::RightButton);
79 setAcceptHoverEvents(true);
80 setCursor(Qt::SizeAllCursor);
81
82 // Appearance: 2 px violetMedium outline, semi-transparent fill.
83 setPen(QPen(kVioletMedium, 2.0));
84 setBrush(QBrush(kVioletFill));
85
86 if (!transformStr.isEmpty()) {
87 m_transformStr = transformStr;
88 this->applyTransformString(x, y, rotationDeg, m_transformStr);
89 m_initialPos = pos();
90 m_initialRotation = rotation();
91 } else {
92 // No prior transform — use rotationDeg if provided.
93 if (std::abs(rotationDeg) > 0.001) {
94 this->setRotation(rotationDeg);
95 }
96 // Build initial transform string from current state.
97 qDebug() << "[PieceOverlayItem] PieceOverlayItem constructor: no prior transform, calling buildTransform() to build transform string";
98 m_transformStr = this->buildTransform();
99 }
100
101 recordCurrentState();
102} // PieceOverlayItem constructor

108/// @brief Parse and apply all SVG transform commands in the string to (x, y, angle).
109void PieceOverlayItem::applyTransformString(double baseX, double baseY, double baseAngle, const QString& transformStr)
110{
111 QTransform cumulative;
112 QString rest = transformStr.trimmed();
113 while (!rest.isEmpty()) {
114 rest.remove(QRegularExpression(R"(^[\s,]+)"));
115 if (rest.isEmpty()) {
116 break;
117 }
118
119 const int paren = rest.indexOf('(');
120 if (paren < 0) {
121 break;
122 }
123
124 const QString funcName = rest.left(paren).trimmed();
125 rest = rest.mid(paren + 1);
126
127 const int close = rest.indexOf(')');
128 if (close < 0) {
129 break;
130 }
131
132 const QString paramsStr = rest.left(close);
133 rest = rest.mid(close + 1);
134
135 const QStringList parts = paramsStr.split(QRegularExpression(R"([\s,]+)"), Qt::SkipEmptyParts);
136 QVector<double> params;
137 params.reserve(parts.size());
138 for (const QString& part : parts) {
139 bool ok = false;
140 const double value = part.toDouble(&ok);
141 if (ok) {
142 params.append(value);
143 }
144 }
145
146 QTransform next;
147 if (funcName == "translate") {
148 const double tx = params.value(0, 0.0);
149 const double ty = params.value(1, 0.0);
150 next.translate(tx, ty);
151 } else if (funcName == "rotate") {
152 const double a = params.value(0, 0.0);
153 const double cx = params.value(1, 0.0);
154 const double cy = params.value(2, 0.0);
155 next.translate(cx, cy);
156 next.rotate(a);
157 next.translate(-cx, -cy);
158 } else if (funcName == "matrix" && params.size() >= 6) {
159 next = QTransform(params[0], params[1], params[2], params[3], params[4], params[5]);
160 } else {
161 continue;
162 }
163
164 cumulative = cumulative * next;
165 }
166
167 // Replay the SVG transform chain against the piece's bounding-box center
168 // (the rotation pivot set via setTransformOriginPoint()) and the local
169 // +X axis through that center, so overlay position/rotation match
170 // buildTransform()'s rotate-about-center convention and stay aligned
171 // with the rendered SVG.
172 const QPointF centerBase(baseX + m_w / 2.0, baseY + m_h / 2.0);
173 const QPointF center = cumulative.map(centerBase);
174 const QPointF axis = cumulative.map(QPointF(centerBase.x() + 1.0, centerBase.y()));
175 const QPointF dir = axis - center;
176 constexpr double kRadToDeg = 57.29577951308232;
177 const double angle = baseAngle + std::atan2(dir.y(), dir.x()) * kRadToDeg;
178
179 // pos() is the scene position of the item's local origin (0,0); recover
180 // it from the mapped center by subtracting the half-extent offset.
181 const QPointF origin(center.x() - m_w / 2.0, center.y() - m_h / 2.0);
182
183 this->setPos(origin);
184 this->setRotation(angle);
185
186} // PieceOverlayItem constructor

198/// @brief Return true if this piece has been moved, rotated, or flipped from its load position.
199bool PieceOverlayItem::hasMoved() const
200{
201 constexpr double kPosTol = 0.5; // half a pixel
202 constexpr double kRotTol = 0.001; // degrees
203 constexpr double kScaleTol = 0.001; // scale tolerance
204 const QPointF cur = pos();
205 return std::abs(cur.x() - m_initialPos.x()) > kPosTol
206 || std::abs(cur.y() - m_initialPos.y()) > kPosTol
207 || std::abs(rotation() - m_initialRotation) > kRotTol
208 || std::abs(m_scaleX - 1.0) > kScaleTol
209 || std::abs(m_scaleY - 1.0) > kScaleTol;
210} // hasMoved

242void PieceOverlayItem::recordCurrentState()
243{
244 const QString fullTransform = buildTransform();
245 const bool duplicateCurrent =
246 m_historyIndex >= 0
247 && m_historyIndex < m_transformHistory.size()
248 && std::abs(m_transformHistory[m_historyIndex].pos.x() - pos().x()) < 0.0001
249 && std::abs(m_transformHistory[m_historyIndex].pos.y() - pos().y()) < 0.0001
250 && std::abs(m_transformHistory[m_historyIndex].rotation - rotation()) < 0.001
251 && std::abs(m_transformHistory[m_historyIndex].scaleX - m_scaleX) < 0.0001
252 && std::abs(m_transformHistory[m_historyIndex].scaleY - m_scaleY) < 0.0001
253 && m_transformHistory[m_historyIndex].transform == fullTransform;
254
255 if (duplicateCurrent) {
256 return;
257 }
258
259 if (m_historyIndex + 1 < m_transformHistory.size()) {
260 m_transformHistory.resize(m_historyIndex + 1);
261 }
262
263 TransformState state;
264 state.pos = pos();
265 state.rotation = rotation();
266 state.scaleX = m_scaleX;
267 state.scaleY = m_scaleY;
268 state.transform = fullTransform;
269 m_transformHistory.append(state);
270 m_historyIndex = m_transformHistory.size() - 1;
271} // recordCurrentState
272
273bool PieceOverlayItem::undo()
274{
275 if (!canUndo()) {
276 return false;
277 }
278
279 --m_historyIndex;
280 applyState(m_transformHistory[m_historyIndex]);
281 return true;
282}

316/// @brief Build the SVG transform string for the item's current position, rotation, and flip state.
317QString PieceOverlayItem::buildTransform() const
318{
319 // Rebuild the piece's full canonical affine transform from its immutable
320 // geometry baseline so Apply can overwrite the SVG transform with a single
321 // matrix that both Qt replay and SVG rendering interpret identically.
322 const double tx = pos().x() - m_basePos.x();
323 const double ty = pos().y() - m_basePos.y();
324 const double angle = rotation() - m_baseRotation;
325 const bool hasTranslation = std::abs(tx) >= 0.0001 || std::abs(ty) >= 0.0001;
326 const bool hasRotation = std::abs(angle) >= 0.001;
327 const bool hasFlip = std::abs(m_scaleX - 1.0) > 0.0001 || std::abs(m_scaleY - 1.0) > 0.0001;
328
329 // Build the combined transform matrix as a rotate-about-center operation,
330 // matching the QGraphicsItem rotation pivot set via setTransformOriginPoint()
331 // in the constructor and contextMenuEvent() (bounding-box center, not the
332 // upper-left corner):
333 // 1. Translate the bbox-center pivot from its base scene position to its
334 // current scene position (pos() + half-extent).
335 // 2. Apply rotation and flip/scale, both pivoted at that same center.
336 // 3. Translate back from the bbox-center pivot's base scene position.
337 const double cx = m_w / 2.0;
338 const double cy = m_h / 2.0;
339
340 QTransform fullMatrix;
341 fullMatrix.translate(pos().x() + cx, pos().y() + cy);
342 fullMatrix.rotate(angle);
343
344 // Flip is pivoted at the same center as the rotation, so no extra
345 // translate pair is needed here.
346 if (hasFlip) {
347 fullMatrix.scale(m_scaleX, m_scaleY);
348 }
349
350 fullMatrix.translate(-m_basePos.x() - cx, -m_basePos.y() - cy);
351
352 qDebug() << "[PieceOverlayItem] buildTransform(): " << pieceId();
353 qDebug() << " local origin ox, oy: (" << m_originX << "," << m_originY << ")";
354 qDebug() << " base x, y: (" << m_basePos.x() << "," << m_basePos.y() << ")";
355 qDebug() << " full transform tx, ty: (" << tx << "," << ty << ")";
356 qDebug() << " old transformStr: " << m_transformStr;
357 qDebug() << " full angle: " << angle;
358 qDebug() << " scaleX, scaleY: (" << m_scaleX << "," << m_scaleY << ")";
359 qDebug() << " new pos.x, pos.y: (" << pos().x() << "," << pos().y() << ")";
360
361 if (!hasTranslation && !hasRotation && !hasFlip) {
362 return QString();
363 }
364
365 return QString("matrix(%1 %2 %3 %4 %5 %6)")
366 .arg(fullMatrix.m11(), 0, 'f', 6)
367 .arg(fullMatrix.m12(), 0, 'f', 6)
368 .arg(fullMatrix.m21(), 0, 'f', 6)
369 .arg(fullMatrix.m22(), 0, 'f', 6)
370 .arg(fullMatrix.dx(), 0, 'f', 6)
371 .arg(fullMatrix.dy(), 0, 'f', 6);
372} // buildTransform()

392/// @brief Move the item by the scene-coordinate delta from press origin.
393void PieceOverlayItem::mouseMoveEvent(QGraphicsSceneMouseEvent* event)
394{
395 if (event->buttons() & Qt::LeftButton) {
396 // Override default QGraphicsItem drag handling to track movement delta in scene coordinates, which prevents the feedback loop that would occur if we used item-space coordinates (event->pos()) that shift as the item moves under the cursor. This also allows for more intuitive dragging when zoomed or transformed.
397 // Fires multiple times as mouse is dragged, don't add debug messages here. The final position and rotation are logged in buildTransform() when the transform string is built for the current state.
398 const QPointF delta = event->scenePos() - m_dragStartScenePos;
399
400 if (!m_dragging && (std::abs(delta.x()) > DRAG_THRESHOLD ||
401 std::abs(delta.y()) > DRAG_THRESHOLD)) {
402 // Threshold crossed — engage drag mode.
403 m_dragging = true;
404 } // if threshold crossed
405
406 if (m_dragging) {
407 // Apply delta from the captured start position.
408 setPos(m_dragStartItemPos + delta);
409 } // if dragging
410 } // if left button held
411
412 // Call base class to preserve default processing for item selection and other flags.
413 QGraphicsRectItem::mouseMoveEvent(event);
414}

416/// @brief Restore z-order and clear drag state on button release.

418{

420
421 // Return item to its normal z-layer.
422 setZValue(1.0);
423 m_dragging = false;

427 if (AdjustScene* adjustScene = dynamic_cast<AdjustScene*>(scene())) {
428 adjustScene->notifyPieceStateCommitted(m_id, m_historyIndex);
429 }
430 }
431
432 QGraphicsRectItem::mouseReleaseEvent(event);
433}

439/// @brief Show the rotation / flip / align / abut context menu.
440void PieceOverlayItem::contextMenuEvent(QGraphicsSceneContextMenuEvent* event)
441{
442 QMenu menu;
443
444 // Header action — shows the piece's human-readable name ("Front Bodice")
445 // when the Seamly2D handoff supplied one, falling back to the id. Not interactive.
446 QAction* header = menu.addAction(QString("Piece: %1").arg(displayLabel()));
447 header->setEnabled(false);
448 menu.addSeparator();
449
450 // Rotation actions.
451 QAction* rot10C = menu.addAction("10 C");
452 QAction* rot10CC = menu.addAction("10 CC");
453 QAction* rot225C = menu.addAction("22.5 C");
454 QAction* rot225CC = menu.addAction("22.5 CC");
455 QAction* rot45C = menu.addAction("45 C");
456 QAction* rot45CC = menu.addAction("45 CC");
457 QAction* rot90C = menu.addAction("90 C");
458 QAction* rot90CC = menu.addAction("90 CC");
459 QAction* rot180 = menu.addAction("180");
460
461 menu.addSeparator();
462
463 // Flip actions.
464 QAction* flipH = menu.addAction("Flip Horizontal");
465 QAction* flipV = menu.addAction("Flip Vertical");
466
467 menu.addSeparator();
468
469 // Align to contentRect edge actions.
470 AdjustScene* adjustScene = dynamic_cast<AdjustScene*>(scene());
471 const bool hasContentRect = adjustScene && adjustScene->hasContentRect();
472
473 QAction* alignLeft = menu.addAction("Align Left Edge");
474 QAction* alignRight = menu.addAction("Align Right Edge");
475 QAction* alignTop = menu.addAction("Align Top Edge");
476 QAction* alignBottom = menu.addAction("Align Bottom Edge");
477
478 // Per-direction gating: enable only when contentRect exists AND the piece is
479 // not already flush with that edge (otherwise the action would be a no-op).
480 constexpr double kEdgeTol = 0.5;
481 if (hasContentRect) {
482 const QRectF cr = adjustScene->contentRect();
483 const QRectF bbox = sceneBoundingRect();
484 alignLeft->setEnabled(std::abs(bbox.left() - cr.left()) > kEdgeTol);
485 alignRight->setEnabled(std::abs(bbox.right() - cr.right()) > kEdgeTol);
486 alignTop->setEnabled(std::abs(bbox.top() - cr.top()) > kEdgeTol);
487 alignBottom->setEnabled(std::abs(bbox.bottom() - cr.bottom()) > kEdgeTol);
488 } else {
489 alignLeft->setEnabled(false);
490 alignRight->setEnabled(false);
491 alignTop->setEnabled(false);
492 alignBottom->setEnabled(false);
493 }
494
495 menu.addSeparator();
496
497 // Abut to nearest piece actions (enabled when a piece exists in that direction).
498 PieceOverlayItem* nearestLeft = adjustScene ? adjustScene->findNearestPieceLeft(this) : nullptr;
499 PieceOverlayItem* nearestRight = adjustScene ? adjustScene->findNearestPieceRight(this) : nullptr;
500 PieceOverlayItem* nearestAbove = adjustScene ? adjustScene->findNearestPieceAbove(this) : nullptr;
501 PieceOverlayItem* nearestBelow = adjustScene ? adjustScene->findNearestPieceBelow(this) : nullptr;
502
503 QAction* abutLeft = menu.addAction("Abut Left Piece");
504 QAction* abutRight = menu.addAction("Abut Right Piece");
505 QAction* abutAbove = menu.addAction("Abut Above Piece");
506 QAction* abutBelow = menu.addAction("Abut Below Piece");
507
508 abutLeft->setEnabled(nearestLeft != nullptr);
509 abutRight->setEnabled(nearestRight != nullptr);
510 abutAbove->setEnabled(nearestAbove != nullptr);
511 abutBelow->setEnabled(nearestBelow != nullptr);
512
513 menu.addSeparator();
514
515 // Z-order stacking actions.
516 QAction* raiseToTop = menu.addAction("Raise to Top");
517 QAction* lowerToBottom = menu.addAction("Lower to Bottom");
518
519 QAction* chosen = menu.exec(event->screenPos());
520
521 // Set rotation pivot at the bounding-box center (see constructor comment).
522 setTransformOriginPoint(m_w / 2.0, m_h / 2.0);
523
524 const double prevRotation = rotation();
525 const double prevScaleX = m_scaleX;
526 const double prevScaleY = m_scaleY;
527 const QPointF prevPos = pos();
528 const qreal prevZValue = zValue();
529 bool stateChanged = false;
530
531 if (chosen == rot10C) {
532 setRotation(rotation() + 10.0);
533 } else if (chosen == rot10CC) {
534 setRotation(rotation() - 10.0);
535 } else if (chosen == rot225C) {
536 setRotation(rotation() + 22.5);
537 } else if (chosen == rot225CC) {
538 setRotation(rotation() - 22.5);
539 } else if (chosen == rot45C) {
540 setRotation(rotation() + 45.0);
541 } else if (chosen == rot45CC) {
542 setRotation(rotation() - 45.0);
543 } else if (chosen == rot90C) {
544 setRotation(rotation() + 90.0);
545 } else if (chosen == rot90CC) {
546 setRotation(rotation() - 90.0);
547 } else if (chosen == rot180) {
548 setRotation(rotation() + 180.0);
549 } else if (chosen == flipH) {
550 // Toggle horizontal flip.
551 m_scaleX = (m_scaleX > 0) ? -1.0 : 1.0;
552 stateChanged = true;
553 } else if (chosen == flipV) {
554 // Toggle vertical flip.
555 m_scaleY = (m_scaleY > 0) ? -1.0 : 1.0;
556 stateChanged = true;
557 } else if (chosen == alignLeft && hasContentRect) {
558 // Move piece so its left edge touches contentRect left edge.
559 const QRectF contentRect = adjustScene->contentRect();
560 const QRectF bbox = sceneBoundingRect();
561 const double dx = contentRect.left() - bbox.left();
562 setPos(pos().x() + dx, pos().y());
563 stateChanged = true;
564 } else if (chosen == alignRight && hasContentRect) {
565 // Move piece so its right edge touches contentRect right edge.
566 const QRectF contentRect = adjustScene->contentRect();
567 const QRectF bbox = sceneBoundingRect();
568 const double dx = contentRect.right() - bbox.right();
569 setPos(pos().x() + dx, pos().y());
570 stateChanged = true;
571 } else if (chosen == alignTop && hasContentRect) {
572 // Move piece so its top edge touches contentRect top edge.
573 const QRectF contentRect = adjustScene->contentRect();
574 const QRectF bbox = sceneBoundingRect();
575 const double dy = contentRect.top() - bbox.top();
576 setPos(pos().x(), pos().y() + dy);
577 stateChanged = true;
578 } else if (chosen == alignBottom && hasContentRect) {
579 // Move piece so its bottom edge touches contentRect bottom edge.
580 const QRectF contentRect = adjustScene->contentRect();
581 const QRectF bbox = sceneBoundingRect();
582 const double dy = contentRect.bottom() - bbox.bottom();
583 setPos(pos().x(), pos().y() + dy);
584 stateChanged = true;
585 } else if (chosen == abutLeft && nearestLeft) {
586 // Move piece so its left edge touches the nearest piece's right edge.
587 const QRectF myBbox = sceneBoundingRect();
588 const QRectF otherBbox = nearestLeft->sceneBoundingRect();
589 const double dx = otherBbox.right() - myBbox.left();
590 setPos(pos().x() + dx, pos().y());
591 stateChanged = true;
592 } else if (chosen == abutRight && nearestRight) {
593 // Move piece so its right edge touches the nearest piece's left edge.
594 const QRectF myBbox = sceneBoundingRect();
595 const QRectF otherBbox = nearestRight->sceneBoundingRect();
596 const double dx = otherBbox.left() - myBbox.right();
597 setPos(pos().x() + dx, pos().y());
598 stateChanged = true;
599 } else if (chosen == abutAbove && nearestAbove) {
600 // Move piece so its top edge touches the nearest piece's bottom edge.
601 const QRectF myBbox = sceneBoundingRect();
602 const QRectF otherBbox = nearestAbove->sceneBoundingRect();
603 const double dy = otherBbox.bottom() - myBbox.top();
604 setPos(pos().x(), pos().y() + dy);
605 stateChanged = true;
606 } else if (chosen == abutBelow && nearestBelow) {
607 // Move piece so its bottom edge touches the nearest piece's top edge.
608 const QRectF myBbox = sceneBoundingRect();
609 const QRectF otherBbox = nearestBelow->sceneBoundingRect();
610 const double dy = otherBbox.top() - myBbox.bottom();
611 setPos(pos().x(), pos().y() + dy);
612 stateChanged = true;
613 } else if (chosen == raiseToTop && adjustScene) {
614 // Raise piece to top of z-order stack.
615 const qreal maxZ = adjustScene->maxPieceZValue();
616 if (zValue() < maxZ + 1.0) {
617 setZValue(maxZ + 1.0);
618 }
619 // Z-order change doesn't need state recording (visual only).
620 } else if (chosen == lowerToBottom && adjustScene) {
621 // Lower piece to bottom of z-order stack.
622 const qreal minZ = adjustScene->minPieceZValue();
623 // Keep above background (z=0), so use minZ - 0.5 but not below 0.5.
624 const qreal newZ = qMax(0.5, minZ - 0.5);
625 if (zValue() > newZ) {
626 setZValue(newZ);
627 }
628 // Z-order change doesn't need state recording (visual only).
629 }
