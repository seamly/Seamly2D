45/// @brief Create a section header widget matching the QML Rectangle + Text.
46/// @param text Section title.
47/// @return Owning pointer to the section header QWidget.
48static QWidget *makeSectionHeader(const QString &text)
49{
50 auto *widget = new QWidget;
51 widget->setFixedHeight(28);
52 widget->setAutoFillBackground(true);
53
54 // Translucent white stripe — matches QML Qt.rgba(1,1,1,0.08)
55 QPalette pal = widget->palette();
56 pal.setColor(QPalette::Window, QColor(255, 255, 255, 20));
57 widget->setPalette(pal);
58
59 auto *layout = new QHBoxLayout(widget);
60 layout->setContentsMargins(12, 0, 0, 0);
61
62 auto *label = new QLabel(text);
63 QFont font = label->font();
64 font.setPixelSize(12); // Theme.fontSizeSmall
65 font.setBold(true);
66 label->setFont(font);
67 label->setStyleSheet(
68 QStringLiteral("color: %1;").arg(SeamlyTheme::SEAMLY_VIOLET_LIGHT.name()));
69
70 layout->addWidget(label);
71 return widget;
72} // makeSectionHeader
