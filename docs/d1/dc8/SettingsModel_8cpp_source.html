88// @brief Copy a file if the destination does not already exist.
89// Task 15 helper — mirrors PreferencesModel.cpp's copyIfMissing() (kept local to this file
90// rather than shared, matching this codebase's existing per-file helper convention).
91bool copyIfMissing(const QString &sourcePath, const QString &destPath)
92{
93 if (QFileInfo::exists(destPath) || !QFileInfo::exists(sourcePath)) {
94 return false;
95 } // if no copy needed/possible
96
97 QDir destDir = QFileInfo(destPath).absoluteDir();
98 if (!destDir.exists()) {
99 destDir.mkpath(QStringLiteral("."));
100 } // if dest dir missing
101
102 return QFile::copy(sourcePath, destPath);
103} // copyIfMissing

105// @brief Recursively copy every entry from a legacy organization directory tree into the
106// new one, skipping anything the destination already has.
107//
108// Task 15: seamlyLayout's organization name changed from "Seamly Systems" to the shared
109// "Seamly" (see main.cpp), so QStandardPaths::AppConfigLocation resolves to a brand new,
110// empty directory. This bridges the settings file(s) forward from the old organization
111// folder the first time the new one is resolved. Safe to call unconditionally — once
112// everything has been copied across it is a cheap no-op.
113void migrateLegacyOrganizationTree(const QString &legacyRoot, const QString &newRoot)
114{
115 const QDir legacyDir(legacyRoot);
116 if (!legacyDir.exists()) {
117 return;
118 } // if nothing to migrate
119
120 const QFileInfoList entries = legacyDir.entryInfoList(QDir::Files | QDir::Dirs | QDir::NoDotAndDotDot);
121 for (const QFileInfo &entry : entries) {
122 const QString destPath = QDir(newRoot).filePath(entry.fileName());
123 if (entry.isDir()) {
124 QDir().mkpath(destPath);
125 migrateLegacyOrganizationTree(entry.absoluteFilePath(), destPath);
126 } else {
127 copyIfMissing(entry.absoluteFilePath(), destPath);
128 } // if directory vs file
129 } // for each legacy entry
130} // migrateLegacyOrganizationTree
131
132// @brief Bridge appConfigRoot forward from the pre-Task-15 "Seamly Systems" organization
133// folder into the new shared "Seamly" one, the first time appConfigRoot is resolved.
134void migrateLegacyOrganization(const QString &newRoot)
135{
136 const QString currentOrganization = QCoreApplication::organizationName();
137 static const QString kLegacyOrganizationName = QStringLiteral("Seamly Systems");
138 if (currentOrganization == kLegacyOrganizationName) {
139 return; // already running under the legacy name — nothing to bridge
140 } // if already legacy
141
142 QCoreApplication::setOrganizationName(kLegacyOrganizationName);
143 const QString legacyRoot = QDir(
144 QStandardPaths::writableLocation(QStandardPaths::AppConfigLocation)).absolutePath();
145 QCoreApplication::setOrganizationName(currentOrganization);
146
147 if (legacyRoot != newRoot) {
148 migrateLegacyOrganizationTree(legacyRoot, newRoot);
149 } // if legacy root resolves to a different directory
150} // migrateLegacyOrganization

219// @brief Convert a value in the active unit to pixels at 96 px/in.
220//
221// @param value Value in active unit (in / cm / mm).
222// @param unit Unit string: "in" | "cm" | "mm".
223// @return Pixel equivalent, rounded to nearest integer.
224static int toPixels(double value, const QString &unit)
225{
226 constexpr double PPI = 96.0;
227 if (unit == QStringLiteral("mm"))
228 return static_cast<int>(std::round(value / 25.4 * PPI)); // mm → in → px
229 if (unit == QStringLiteral("cm"))
230 return static_cast<int>(std::round(value / 2.54 * PPI)); // cm → in → px
231 return static_cast<int>(std::round(value * PPI)); // "in" or unknown
232} // toPixels()

292// @brief Snap a raw rotationStep value into the set valid for the given mode.
293// "withNap" → { 0, 180 } (0 = head-up, 180 = head-down)
294// other → value unchanged (rotationStep is unused for "alongGrainline")
295// Always returns the closest allowed value by absolute difference.
296static double snapRotationStepForMode(double raw, const QString &mode)
297{
298 static const double napAllowed[] = { 0.0, 180.0 };
299 const double *allowed = nullptr;
300 int count = 0;
301 if (mode == QStringLiteral("withNap")) {
302 allowed = napAllowed;
303 count = static_cast<int>(sizeof(napAllowed) / sizeof(napAllowed[0]));
304 } else {
305 return raw; // alongGrainline (or unknown): leave value alone
306 } // if mode
307
308 double best = allowed[0];
309 double bestDiff = std::abs(raw - allowed[0]);
310 for (int i = 1; i < count; ++i) {
311 const double d = std::abs(raw - allowed[i]);
312 if (d < bestDiff) { best = allowed[i]; bestDiff = d; }
313 } // for i
314 return best;
315} // snapRotationStepForMode()

317// @brief Set the piece-arrangement mode.
318// Unsupported values (including legacy "rotate") are coerced to
319// "alongGrainline" to keep behavior stable and deterministic.
320void SettingsModel::setLayoutMode(const QString &v)
321{
322 QString normalized = v;
323 if (normalized != QStringLiteral("alongGrainline")
324 && normalized != QStringLiteral("withNap")) {
325 normalized = QStringLiteral("alongGrainline");
326 } // if unsupported layout mode
327
328 if (m_layoutMode == normalized) return;
329 m_layoutMode = normalized;
330 // Coerce rotationStep when entering a mode that constrains it.
331 if (normalized == QStringLiteral("withNap")) {
332 const double snapped = snapRotationStepForMode(m_rotationStep, normalized);
333 if (!qFuzzyCompare(snapped, m_rotationStep)) {
334 m_rotationStep = snapped;
335 emit rotationStepChanged();
336 } // if snapped differs
337 } // if mode constrains rotationStep
338 emit layoutModeChanged();
339} // setLayoutMode()

565// @brief Load settings from JSON. Missing keys keep their current (default) values.
566bool SettingsModel::load(const QString &path)
567{
568 Logger::log(QStringLiteral("==========LOAD SETTINGS=========="));
569 Logger::log(QStringLiteral("SettingsModel::load(): path=") + path);
570
571 QFile f(path);
572 if (!f.open(QIODevice::ReadOnly)) {
573 // File absent — keep defaults (not an error on first run)
574 Logger::log(QStringLiteral("SettingsModel::load(): file not found, keeping defaults"));
575 return true;
576 } // if !open
577
578 QJsonParseError err;
579 const QJsonDocument doc = QJsonDocument::fromJson(f.readAll(), &err);
580 if (err.error != QJsonParseError::NoError) {
581 return false;
582 } // if parse error
583
584 const QJsonObject obj = doc.object();
585
586 // -----------------------------------------------------------------
587 // Layout mode + rotation step (with legacy migration)
588 //
589 // Pre-2026-05 schema:
590 // layoutMode ∈ { "withGrain", "withoutGrain" }
591 // rotationEnabled : bool
592 // rotationStep : double (any value)
593 //
594 // New schema:
595 // layoutMode ∈ { "alongGrainline", "withNap" }
596 // rotationStep : double — semantics depend on layoutMode:
597 // "withNap" → fixed offset ∈ { 0, 180 }
598 // "alongGrainline" → unused
599 //
600 // Migration mapping:
601 // withGrain + rotationEnabled=false → withNap (head-up: rotationStep=0)
602 // withGrain + rotationEnabled=true → alongGrainline
603 // withGrain + (no rotationEnabled) → alongGrainline (was the implicit default)
604 // withoutGrain + (any) → alongGrainline (rotate removed)
605 // -----------------------------------------------------------------
606 if (obj.contains(QStringLiteral("layoutMode"))) {
607 const QString rawMode = obj[QStringLiteral("layoutMode")].toString();
608 if (rawMode == QStringLiteral("withGrain")) {
609 // Legacy: choose between alongGrainline and withNap based on the
610 // old rotationEnabled flag.
611 const bool legacyRotationEnabled =
612 obj.contains(QStringLiteral("rotationEnabled"))
613 ? obj[QStringLiteral("rotationEnabled")].toBool()
614 : true; // default in the old schema
615 if (legacyRotationEnabled) {
616 setLayoutMode(QStringLiteral("alongGrainline"));
617 } else {
618 // Default withNap migration to head-up (rotationStep=0).
619 setRotationStep(0.0);
620 setLayoutMode(QStringLiteral("withNap"));
621 } // if legacyRotationEnabled
622 } else if (rawMode == QStringLiteral("withoutGrain")) {
623 // Legacy mode had free rotation; Rotate has been removed, so map
624 // to alongGrainline.
625 setLayoutMode(QStringLiteral("alongGrainline"));
626 } else {
627 // Modern value or unknown — pass through; QML radio group will
628 // ignore unrecognised values until the user picks one.
629 setLayoutMode(rawMode);
630 } // if legacy mode
631 } // if layoutMode present
632
633 // Rotation step: snap to the set valid for the resolved layoutMode.
634 // setLayoutMode() above has already coerced m_rotationStep into a valid
635 // value when switching modes, but the JSON's rotationStep may carry a more
636 // specific user choice (e.g. step=180 for withNap)
637 // that we should restore here.
638 if (obj.contains(QStringLiteral("rotationStep"))) {
639 const double raw = obj[QStringLiteral("rotationStep")].toDouble(m_rotationStep);
640 setRotationStep(snapRotationStepForMode(raw, m_layoutMode));
641 } // if rotationStep present
642
643 if (obj.contains(QStringLiteral("fabricFolded")))
644 setFabricFolded(obj[QStringLiteral("fabricFolded")].toBool());
645
646 if (obj.contains(QStringLiteral("unit")))
