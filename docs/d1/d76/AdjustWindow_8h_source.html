24/// @class AdjustWindow
25/// @brief Top-level QtWidgets window for interactive layout piece adjustment.
26///
27/// Opens the layout SVG with interactive PieceOverlayItems overlaid. The user can:
28/// - Drag pieces to new positions.
29/// - Right-click to rotate or reset a piece.
30/// - Zoom with the mouse wheel.
31/// - Fit everything in view via the toolbar or Ctrl+0.
32/// - Click Apply to confirm changes (emits accepted()) or Cancel to discard.
33///
34/// @par Typical usage from QML / C++:
35/// @code
36/// auto* win = new AdjustWindow(svgPath, bboxJson);
37/// connect(win, &AdjustWindow::accepted, this, [](const QString& json){ ... });
38/// connect(win, &AdjustWindow::cancelled, win, &AdjustWindow::close);
39/// win->show();
40/// @endcode
41class AdjustWindow : public QMainWindow
