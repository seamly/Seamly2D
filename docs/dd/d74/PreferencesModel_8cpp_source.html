44// @brief Read the DataRoot value the Windows installer recorded for SeamlyLayout.
45//
46// smsi_registry.wxs mirrors the same install breadcrumbs Seamly2D's key carries under
47// HKLM\SOFTWARE\Seamly\SeamlyLayout. This is that key's reader — the per-app counterpart to
48// InstallerRecord::dataRoot() (src/libs/vmisc/installer_record.cpp), which SeamlyLayout's
49// standalone CMake/Cargo build does not link against.
50//
51// Registry64Format rather than NativeFormat: the apps are 64-bit today, so the two agree,
52// but the MSI is x64 and always writes the 64-bit view.
53//
54// @return the recorded path cleaned into Qt's '/' separator form; empty when absent, blank,
55// or off Windows.
56QString installerDataRoot()
57{
58 const QSettings installKey(
59 QLatin1String("HKEY_LOCAL_MACHINE\\SOFTWARE\\Seamly\\SeamlyLayout"),
60 QSettings::Registry64Format);
61 const QString recorded = installKey.value(QStringLiteral("DataRoot")).toString().trimmed();
62 if (recorded.isEmpty()) {
63 // Layout.8.1: logged so a fresh-install repro can show whether this is an MSI
64 // custom-action ordering race (registry value not yet written) or a genuine
65 // absence (unpackaged/dev build, non-installer run).
66 Logger::log(QStringLiteral(
67 "installerDataRoot(): HKLM\\SOFTWARE\\Seamly\\SeamlyLayout\\DataRoot empty or absent"));
68 return QString();
69 } // if nothing recorded
70
71 const QString cleaned = QDir::cleanPath(QDir::fromNativeSeparators(recorded));
72 Logger::log(QStringLiteral("installerDataRoot(): read \"") + cleaned + QStringLiteral("\""));
73 return cleaned;
74} // installerDataRoot

81// @brief Recursively copy every entry from a legacy organization directory tree into the
82// new one, skipping anything the destination already has.
83//
84// Task 15: seamlyLayout's organization name changed from "Seamly Systems" to the shared
85// "Seamly" (see main.cpp), so QStandardPaths::AppConfigLocation now resolves to a brand
86// new, empty directory. This bridges every settings/preferences file forward from the old
87// organization folder the first time the new one is resolved. Safe to call unconditionally
88// on every appConfigRootPath() call — once everything has been copied across it is a cheap
89// no-op (copyIfMissing never overwrites an existing destination file).
90void migrateLegacyOrganizationTree(const QString &legacyRoot, const QString &newRoot)
91{
92 const QDir legacyDir(legacyRoot);
93 if (!legacyDir.exists()) {
94 return;
95 } // if nothing to migrate
96
97 const QFileInfoList entries = legacyDir.entryInfoList(QDir::Files | QDir::Dirs | QDir::NoDotAndDotDot);
98 for (const QFileInfo &entry : entries) {
99 const QString destPath = QDir(newRoot).filePath(entry.fileName());
100 if (entry.isDir()) {
101 QDir().mkpath(destPath);
102 migrateLegacyOrganizationTree(entry.absoluteFilePath(), destPath);
103 } else {
104 copyIfMissing(entry.absoluteFilePath(), destPath);
105 } // if directory vs file
106 } // for each legacy entry
107} // migrateLegacyOrganizationTree

109// @brief Return the writable app-config root path.
110// Falls back to <exeDir> only if Qt cannot provide AppConfigLocation.
111//
112// Task 15: also bridges data forward from the pre-unification "Seamly Systems"
113// organization folder into the new shared "Seamly" one on first use. The legacy root is
114// computed by asking QStandardPaths for AppConfigLocation under the old organization name
115// (temporarily swapped in), which reconstructs the exact platform-specific legacy path
116// (Windows/macOS/Linux each resolve AppConfigLocation differently) without hard-coding any
117// of that platform logic here.
118QString appConfigRootPath()
119{
120 QString root = QStandardPaths::writableLocation(QStandardPaths::AppConfigLocation);
121 if (root.isEmpty()) {
122 root = QCoreApplication::applicationDirPath();
123 } // if AppConfigLocation unavailable
124 root = QDir(root).absolutePath();
125 QDir().mkpath(root);
126
127 const QString currentOrganization = QCoreApplication::organizationName();
128 if (currentOrganization != QString::fromUtf8(kLegacyOrganizationName)) {
129 QCoreApplication::setOrganizationName(QString::fromUtf8(kLegacyOrganizationName));
130 const QString legacyRoot = QDir(
131 QStandardPaths::writableLocation(QStandardPaths::AppConfigLocation)).absolutePath();
132 QCoreApplication::setOrganizationName(currentOrganization);
133
134 if (legacyRoot != root) {
135 migrateLegacyOrganizationTree(legacyRoot, root);
136 } // if legacy root resolves to a different directory
137 } // if not already running under the legacy organization name
138
139 return root;
140} // appConfigRootPath

154// @brief Replace supported tokens in default path templates.
155//
156// Layout.8.3: only input_directory/layout_directory (user data, shared with the rest of
157// the suite) go through this. ${DATAROOT} resolves to the Windows installer's recorded
158// DataRoot when one exists, so a fresh MSI install nests those two under %DATAROOT%.
159// Layout.11: without an installer record (dev builds, Linux, macOS) the fallback is
160// <Documents>/SeamlyData — the suite's data-storage pattern — never the raw home
161// directory. preferences/settings paths are app-config, not user data, and are seeded
162// straight from appConfigRootPath() in seedFromBundledDefaults() below — never through
163// this function — so they carry no dependency on the installer registry key or MSI
164// custom-action ordering.
165QString expandDefaultPathTokens(const QString &value)
166{
167 QString documents = QStandardPaths::writableLocation(QStandardPaths::DocumentsLocation);
168 if (documents.isEmpty()) {
169 documents = QDir::homePath();
170 } // if the platform reports no documents location
171 QString dataRootBase = QDir::cleanPath(documents) + QStringLiteral("/SeamlyData");
172#ifdef Q_OS_WIN
173 const QString installedDataRoot = installerDataRoot();
174 if (!installedDataRoot.isEmpty()) {
175 dataRootBase = installedDataRoot;
176 } // if the installer recorded a data root
177#endif // Q_OS_WIN
178
179 QString result = value;
180 result.replace(QStringLiteral("${DATAROOT}"), dataRootBase);
181 return QDir::cleanPath(result);
182} // expandDefaultPathTokens

184// @brief Seed preferences.json from bundled resource defaults.
185//
186// Layout.8.3: settings_directory/preferences_directory/settings_file/preferences_file are
187// app-config, exactly like Seamly2D/SeamlyMe's own qt6_*.ini — they must resolve under
188// appConfigRootPath() (%LOCALAPPDATA%\Seamly\SeamlyLayout on Windows) regardless of
189// whether the Windows installer's DataRoot registry key has been written yet. Routing them
190// through expandDefaultPathTokens()/installerDataRoot() (as before) made them depend on
191// MSI custom-action ordering and silently fell back to the raw home directory when that
192// race lost. input_directory/layout_directory are genuine user data, shared with the rest
193// of the suite, so they keep the DataRoot-substituted template below.
194bool seedFromBundledDefaults(const QString &destPath)
195{
196 Logger::log(QStringLiteral("seedFromBundledDefaults(): destPath=") + destPath);
197
198 QFile bundled(QStringLiteral(":/defaults/default_preferences.json"));
199 if (!bundled.open(QIODevice::ReadOnly)) {
200 return false;
201 } // if cannot open resource
202
203 QJsonParseError parseError;
204 const QJsonDocument doc = QJsonDocument::fromJson(bundled.readAll(), &parseError);
205 bundled.close();
206 if (parseError.error != QJsonParseError::NoError || !doc.isObject()) {
207 return false;
208 } // if invalid JSON
209
210 const QJsonObject root = doc.object();
211 const QString key = platformDefaultsKey();
212 const QJsonObject source = root.value(key).isObject() ? root.value(key).toObject() : root;
213
214 // Build normalized OS-appropriate defaults for runtime use.
215 QJsonObject out;
216
217 const QString appConfigRoot = appConfigRootPath();
218 Logger::log(QStringLiteral("seedFromBundledDefaults(): appConfigRoot=") + appConfigRoot);
219
220 const QString defaultSettingsDir = QDir::toNativeSeparators(
221 QDir(appConfigRoot).filePath(QString::fromUtf8(kSettingsFolderName)));
222 const QString defaultPreferencesDir = QDir::toNativeSeparators(
223 QDir(appConfigRoot).filePath(QString::fromUtf8(kPreferencesFolderName)));
224 const QString defaultInputDir = QDir::toNativeSeparators(
225 expandDefaultPathTokens(source.value(QStringLiteral("input_directory"))
226 .toString(QStringLiteral("${DATAROOT}/layouts"))));
227 const QString defaultLayoutDir = QDir::toNativeSeparators(
228 expandDefaultPathTokens(source.value(QStringLiteral("layout_directory"))
229 .toString(QStringLiteral("${DATAROOT}/layouts"))));
230
231 const QString defaultSettingsFile = QDir::toNativeSeparators(
232 QFileInfo(QDir(defaultSettingsDir).filePath(QStringLiteral("default_settings.json"))).absoluteFilePath());
233 const QString defaultPreferencesFile = QDir::toNativeSeparators(
234 QFileInfo(QDir(defaultPreferencesDir).filePath(QStringLiteral("default_preferences.json"))).absoluteFilePath());
235
236 out[QStringLiteral("settings_directory")] = defaultSettingsDir;
237 out[QStringLiteral("preferences_directory")] = defaultPreferencesDir;
238 out[QStringLiteral("preferences_file")] = defaultPreferencesFile;
239 out[QStringLiteral("input_directory")] = defaultInputDir;
240 out[QStringLiteral("layout_directory")] = defaultLayoutDir;
241 out[QStringLiteral("settings_file")] = defaultSettingsFile;
242 out[QStringLiteral("dxf_viewer_path")] = source.value(QStringLiteral("dxf_viewer_path")).toString(QStringLiteral(""));
243 out[QStringLiteral("pdf_viewer_path")] = source.value(QStringLiteral("pdf_viewer_path")).toString(QStringLiteral(""));
244 out[QStringLiteral("png_viewer_path")] = source.value(QStringLiteral("png_viewer_path")).toString(QStringLiteral(""));
245 out[QStringLiteral("projector_path")] = source.value(QStringLiteral("projector_path")).toString(QStringLiteral(""));
246
247 // Ensure default folders exist at first run.
248 QDir().mkpath(defaultSettingsDir);
249 QDir().mkpath(defaultPreferencesDir);
250 QDir().mkpath(defaultInputDir);
251 QDir().mkpath(defaultLayoutDir);
252
253 QDir destDir = QFileInfo(destPath).absoluteDir();
254 if (!destDir.exists()) {
255 destDir.mkpath(QStringLiteral("."));
256 } // if missing
257
258 QFile outFile(destPath);
259 if (!outFile.open(QIODevice::WriteOnly | QIODevice::Truncate)) {
260 return false;
261 } // if cannot write
262
263 outFile.write(QJsonDocument(out).toJson(QJsonDocument::Indented));
264 outFile.close();
265 return true;
266} // seedFromBundledDefaults

411// @brief Load preferences from an INI file or a legacy JSON defaults file.
412bool PreferencesModel::load(const QString &path)
413{
414 Logger::log(QStringLiteral("==========LOAD PREFERENCES=========="));
415 Logger::log(QStringLiteral("PreferencesModel::load(): path=") + path);
416
417 const QString absolutePath = QFileInfo(path).absoluteFilePath();
418 if (QFileInfo(absolutePath).suffix().compare(QStringLiteral("json"), Qt::CaseInsensitive) == 0) {
419 return loadJsonPreferences(absolutePath);
420 } // if legacy JSON or defaults profile
421
422 if (!QFileInfo::exists(absolutePath)) {
423 // DEPRECATED first-run seeding (Task SettingsFiles.3, 2026-08-31). The
424 // Windows MSI seeds a complete qt6_seamlylayout.ini at install time
425 // (packaging/windows/smsi_seed_user_settings.ps1), so on an installed
426 // Windows machine this branch never runs. It stays only for packages
427 // with no install hook — the macOS dmg, the Linux AppImage/Flatpak,
428 // dev builds. Remove it when those packages gain install-time seeding.
429 if (QFileInfo(absolutePath).fileName() != QString::fromUtf8(kPreferencesFileName)) {
430 Logger::log(QStringLiteral("PreferencesModel::load(): INI file not found"));
431 return false;
432 } // if this is not the application preferences file
433
434 if (migrateLegacyPreferencesJson(absolutePath)) {
435 adoptInstallerDataRootIfEmpty(absolutePath);
436 return true;
437 } // if legacy preferences migrated
438
439 if (!resetToDefaults() || !save(absolutePath)) {
440 Logger::log(QStringLiteral("PreferencesModel::load(): failed to create the INI file"));
441 return false;
442 } // if defaults could not be saved
443 adoptInstallerDataRootIfEmpty(absolutePath);
444 return true;
445 } // if INI file is missing
446
447 QSettings settings(absolutePath, QSettings::IniFormat);
448 setInputDirectory(settings.value(QStringLiteral("input_directory"), m_inputDirectory).toString());
449 setLayoutDirectory(settings.value(QStringLiteral("layout_directory"), m_layoutDirectory).toString());
450 setPreferencesDirectory(
451 settings.value(QStringLiteral("preferences_directory"), m_preferencesDirectory).toString());
452 setSettingsDirectory(settings.value(QStringLiteral("settings_directory"), m_settingsDirectory).toString());
453 setSettingsFile(settings.value(QStringLiteral("settings_file"), m_settingsFile).toString());
454 setPreferencesFile(settings.value(QStringLiteral("preferences_file"), m_preferencesFile).toString());
455 setDxfViewerPath(settings.value(QStringLiteral("dxf_viewer_path"), m_dxfViewerPath).toString());
456 setPdfViewerPath(settings.value(QStringLiteral("pdf_viewer_path"), m_pdfViewerPath).toString());
457 setPngViewerPath(settings.value(QStringLiteral("png_viewer_path"), m_pngViewerPath).toString());
458 setProjectorPath(settings.value(QStringLiteral("projector_path"), m_projectorPath).toString());
459 setDataRoot(settings.value(QStringLiteral("data_root"), m_dataRoot).toString());
460
461 if (settings.status() != QSettings::NoError) {
462 Logger::log(QStringLiteral("PreferencesModel::load(): QSettings reported an error"));
463 return false;
464 } // if read failed
465
466 migrateLegacyPreferencePaths();
467 adoptInstallerDataRootIfEmpty(absolutePath);
468
469 // Task SettingsFiles.6: an installer-seeded ini skips the deprecated first-run
470 // seeding above, so nothing created the defaults profile the ini's
471 // preferences_file key names. Seed it here, on demand — the same app-side rule
472 // resolvedSettingsDirectory() applies to default_settings.json. Never overwrites
473 // an existing profile.
474 const QString defaultsProfile = preferencesFilePath();
475 if (!QFileInfo::exists(defaultsProfile) && seedFromBundledDefaults(defaultsProfile)) {
476 Logger::log(QStringLiteral("PreferencesModel::load(): seeded missing defaults profile ")
477 + defaultsProfile);
478 } // if defaults profile was missing
479
480 Logger::log(QStringLiteral("PreferencesModel::load(): loaded successfully"));
481 return true;
482} // load

590// @brief Adopt the Windows installer's recorded data root, once, if dataRoot is unset.
591//
592// Mirrors VCommonSettings::initializeDataRoot() (src/libs/vmisc/vcommonsettings.cpp) for

1062/// @brief Check if a string is an HTTP/HTTPS URL.
1063/// @param str The string to check.
1064/// @return true if the string starts with http:// or https://.
1065static bool isHttpUrl(const QString &str)

1067 return str.startsWith(QStringLiteral("http://"), Qt::CaseInsensitive)
1068 || str.startsWith(QStringLiteral("https://"), Qt::CaseInsensitive);

1071// @brief Launch a file in a viewer application or open an online viewer URL.
1072// If viewerPath is an HTTP/HTTPS URL, opens the URL in the default browser.
1073// Otherwise, the viewer is parsed:
1074// 1. A leading `file:///` prefix is stripped (Qt FileDialog may return
1075// URL-form paths; we tolerate that here too).
1076// 2. If the whole string resolves to an existing file (common case: bare
1077// Windows path with unquoted spaces like
1078// `C:\Program Files\Common Files\eDrawings2026\eDrawings.exe`), it is
1079// used as the executable directly with no preset arguments.
1080// 3. Otherwise the string is parsed as a shell-style command
1081// (executable + optional preset arguments) — supports the Chrome PWA
1082// shortcut form which always uses quotes around the path:
1083// `"C:\Program Files\Google\Chrome\Application\chrome_proxy.exe"
1084// --profile-directory=Default --app-id=<id>`.
1085// The file path, when non-empty, is appended as the final argument. An empty
1086// file path is allowed for launcher-only viewers (e.g. a projector
1087// application that takes no file argument).
1088//
1089// On Windows, detects UWP/Store apps (App Execution Aliases) and skips the
1090// slow QProcess::startDetached failure by going straight to QDesktopServices.
1091// @param viewerPath HTTP/HTTPS URL OR a path (bare or quoted) OR a
1092// shell-style command.
1093// @param filePath Absolute file path to append as the final argument, or
1094// empty when the viewer takes no file argument.
1095// @return true if the viewer was launched successfully.
1096bool PreferencesModel::openInViewer(const QString &viewerPath, const QString &filePath)
1097{
1098 if (viewerPath.isEmpty()) {
1099 Logger::log(QStringLiteral("PreferencesModel::openInViewer(): empty viewer path"));
1100 return false;
1101 } // if empty viewer
1102
1103 // Handle online viewer URLs (HTTP/HTTPS).
1104 if (isHttpUrl(viewerPath)) {
1105 Logger::log(QStringLiteral("PreferencesModel::openInViewer(): opening URL viewer=")
1106 + viewerPath);
1107 // Note: Online viewers like ShareCAD require users to upload files manually,
1108 // as local files cannot be passed directly to web services.
1109 return QDesktopServices::openUrl(QUrl(viewerPath));
1110 }
1111
1112 QStringList parsed = parseViewerCommand(viewerPath);
1113 if (parsed.isEmpty()) {
1114 Logger::log(QStringLiteral("PreferencesModel::openInViewer(): viewer command parsed to no tokens"));
1115 return false;
1116 }
1117 const QString executable = parsed.takeFirst();
1118 QStringList args = parsed; // remainder are preset args
1119
1120 QString nativeFile;
1121 if (!filePath.isEmpty()) {
1122 nativeFile = Platform::toNativePath(filePath);
1123 args.append(nativeFile);
1124 }
1125
1126 Logger::log(QStringLiteral("PreferencesModel::openInViewer(): viewer=")
1127 + executable
1128 + QStringLiteral(" args=[") + args.join(QStringLiteral(", "))
1129 + QStringLiteral("] file=") + nativeFile);
1130
1131 bool ok = false;
1132
1133 // On Windows, UWP/Store apps (e.g. MSPaint) are NTFS reparse points that
1134 // QProcess::startDetached cannot launch. Detect them up front and skip
1135 // straight to QDesktopServices to avoid the slow QProcess failure/timeout.
1136 if (Platform::isStoreApp(executable)) {
1137 Logger::log(QStringLiteral("PreferencesModel::openInViewer(): Store app detected, using QDesktopServices"));
1138 if (!nativeFile.isEmpty()) {
1139 ok = QDesktopServices::openUrl(QUrl::fromLocalFile(nativeFile));
1140 } else {
1141 // No file to hand the shell; ask QDesktopServices to launch the exe directly.
1142 ok = QDesktopServices::openUrl(QUrl::fromLocalFile(executable));
1143 }
1144 } else {
1145 // Traditional desktop executable — launch directly via QProcess.
1146 ok = QProcess::startDetached(executable, args);
1147 if (!ok && !nativeFile.isEmpty()) {
1148 // Fallback: open the file via the OS shell, which handles edge cases
1149 // (e.g. file-type associations) that QProcess cannot launch.
1150 Logger::log(QStringLiteral("PreferencesModel::openInViewer(): QProcess failed, falling back to QDesktopServices"));
1151 ok = QDesktopServices::openUrl(QUrl::fromLocalFile(nativeFile));
1152 } // if QProcess failed and we have a file
1153 } // if Store app vs traditional exe
1154
1155 if (!ok) {
1156 Logger::log(QStringLiteral("PreferencesModel::openInViewer(): failed to open file"));
1157 } // if all methods failed
1158 return ok;
1159} // openInViewer

1233 // hand back URL-form paths and users may also paste one in).

1235 const QString asLocal = QUrl(viewerStr).toLocalFile();

1237 viewerStr = asLocal;
1238 }
1239 }
1240

1242 // executable with no preset args. Handles Browse-picked Windows paths
1243 // with spaces (e.g. `C:\Program Files\Common Files\eDrawings2026\eDrawings.exe`)
1244 // without requiring quotes.
1245 if (QFileInfo::exists(viewerStr)) {

1250 // Chrome PWA shortcut: `"C:\Program Files\Google\Chrome\Application\chrome_proxy.exe"
1251 // --profile-directory=Default --app-id=<id>`).

1254 return QStringList();
1255 }

1257 out.append(Platform::toNativePath(tokens.takeFirst()));
1258 out.append(tokens);
