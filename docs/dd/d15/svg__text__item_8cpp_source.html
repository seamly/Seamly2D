48/**
49 * @brief paint draws the text with QPainter::drawText() so paint engines
50 * receive real text draw calls.
51 *
52 * Workflow: the item's font is applied to the painter, the fill color is
53 * applied as a solid pen (paint engines take the text fill from the painter's
54 * pen — QSvgGenerator even skips text entirely when the pen is NoPen), and
55 * every line of the text is drawn at its baseline with QPainter::drawText().
56 * QPainter forwards each call to QPaintEngine::drawTextItem(), which the SVG
57 * engine writes as a <text> element, the PDF engine as searchable text, and
58 * the DXF engine as a TEXT entity; raster engines (PNG, screen) rasterize the
59 * glyphs as before.
60 *
61 * Geometry matches QGraphicsSimpleTextItem: the first line's top edge sits at
62 * y = 0 (drawText() takes a baseline point, so the baseline is offset by the
63 * font's ascent) and subsequent lines advance by the font's line spacing.
64 *
65 * @param painter painter to draw with.
66 * @param option unused style options.
67 * @param widget unused widget pointer.
68 */
69void SvgTextItem::paint(QPainter *painter, const QStyleOptionGraphicsItem *option, QWidget *widget)
70{
71 Q_UNUSED(option)
72 Q_UNUSED(widget)
73
74 painter->setFont(font());
75 painter->setPen(QPen(brush().color()));
76
77 const QFontMetricsF metrics(font());
78 qreal baseline = metrics.ascent();
79
80 // Label lines are usually single-line strings, but honor embedded
81 // newlines the same way the base class does: one drawn line each.
82 const QStringList lines = text().split(QLatin1Char('\n'));
83 for (int i = 0; i < lines.size(); ++i)
84 {
85 painter->drawText(QPointF(0.0, baseline), lines.at(i));
86 baseline += metrics.lineSpacing();
87 }
88}
