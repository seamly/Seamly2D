5// @file LoggerTests.cpp
6// @brief Qt tests for Logger — where the session log file is written.
7//
8// Layout.10. SeamlyLayout used to write its session log to
9// %LOCALAPPDATA%\SeamlyLayout\output\log_<timestamp>.txt: outside the shared
10// "Seamly" organization folder that holds qt6_seamlylayout.ini, and in a
11// directory named "output" rather than "logs". Two separate causes:
12//
13// • Logger::init() appended "/output" to the AppConfigLocation root;
14// • main() called Logger::init() BEFORE it set the organization and
15// application names, and AppConfigLocation is built from those two names,
16// so the root itself was wrong.
17//
18// The suite locks the resulting path, which is
19// <AppConfigLocation>/logs/log_YYMMDDHHMMSS.txt — for an installed Windows
20// build, %LOCALAPPDATA%\Seamly\SeamlyLayout\logs.
21//
22// QStandardPaths test mode redirects AppConfigLocation into a throwaway tree,
23// so the suite never touches the real user configuration.
24//
25// Logger is the only writer of the log file. Rust lines arrive through
26// seamly_logger_write_utf8(), so concurrentWriters_lineArrivesWholeAndInOrder()
27// calls that entry point directly in place of the Rust bridge.
28//
29// Logger needs no QObject, no window and no event loop, so this suite runs
30// guiless — QTEST_GUILESS_MAIN, not QTEST_MAIN.

141// All three write paths run at the same time from two threads: Logger::log(),
142// the Qt message handler, and the Rust entry point. Every line must arrive
143// whole, and each (thread, source) pair must keep its order. The order between
144// threads is not defined, so the test does not check it.
145void LoggerTests::concurrentWriters_lineArrivesWholeAndInOrder()
146{
147 const QtMessageHandler previousHandler = qInstallMessageHandler(Logger::messageHandler);
148
149 constexpr int threadCount = 2;
150 constexpr int linesPerSource = 300;
151 const QString marker = QStringLiteral("[concurrent_writers]");
152
153 // Each thread cycles through the three sources, so the sources interleave too.
154 std::vector<std::unique_ptr<QThread>> writers;
155 for (int threadId = 0; threadId < threadCount; ++threadId) {
156 writers.emplace_back(QThread::create([threadId, marker]() {
157 for (int index = 0; index < linesPerSource; ++index) {
158 const QString body = QStringLiteral(" thread=%1 index=%2 end").arg(threadId).arg(index);
159 Logger::log(marker + QStringLiteral(" source=cpp") + body);
160 qDebug().noquote() << marker + QStringLiteral(" source=qt") + body;
161 const QByteArray rustLine = (marker + QStringLiteral(" source=rust") + body).toUtf8();
162 seamly_logger_write_utf8(rustLine.constData(),
163 static_cast<std::size_t>(rustLine.size()));
164 }
165 }));
166 }
167 for (auto &writer : writers) writer->start();
168 for (auto &writer : writers) QVERIFY(writer->wait(30000));
169
170 qInstallMessageHandler(previousHandler);
171
172 QFile logFile(m_logFilePath);
173 QVERIFY(logFile.open(QIODevice::ReadOnly | QIODevice::Text));
174 const QStringList lines = QString::fromUtf8(logFile.readAll()).split(QLatin1Char('\n'));
175
176 // A whole line: timestamp prefix, one source, and the "end" sentinel. The Qt
177 // handler may append " (file:line)" when the build carries message context.
178 const QRegularExpression wholeLine(QStringLiteral(
179 R"(^\[\d+\] DEBUG: \[concurrent_writers\] source=(cpp|qt|rust) )"
180 R"(thread=(\d+) index=(\d+) end( \‍(.+:\d+\))?$)"));
181
182 QHash<QString, int> nextIndex; // key: "<source>/<thread>"
183 int matched = 0;
184 for (const QString &line : lines) {
185 if (!line.contains(marker)) continue;
186 const QRegularExpressionMatch match = wholeLine.match(line);
187 QVERIFY2(match.hasMatch(), qPrintable(QStringLiteral("clipped line: '%1'").arg(line)));
188
189 const QString key = match.captured(1) + QLatin1Char('/') + match.captured(2);
190 const int index = match.captured(3).toInt();
191 QVERIFY2(index == nextIndex.value(key, 0),
192 qPrintable(QStringLiteral("out of order: '%1'").arg(line)));
193 nextIndex[key] = index + 1;
194 ++matched;
195 }
196
197 QCOMPARE(matched, threadCount * linesPerSource * 3);
198} // concurrentWriters_lineArrivesWholeAndInOrder()
