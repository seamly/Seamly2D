5// @file LoggerTests.cpp
6// @brief Qt tests for Logger — where the session log file is written.
7//
8// Layout.10. SeamlyLayout used to write its session log to
9// %LOCALAPPDATA%\SeamlyLayout\output\log_<timestamp>.txt: outside the shared
10// "Seamly" organization folder that holds qt6_seamlylayout.ini, and in a
11// directory named "output" rather than "logs". Two separate causes:
12//
13// • Logger::init() appended "/output" to the AppConfigLocation root;
14// • main() called Logger::init() BEFORE it set the organization and
15// application names, and AppConfigLocation is built from those two names,
16// so the root itself was wrong.
17//
18// The suite locks the resulting path, which is
19// <AppConfigLocation>/logs/log_YYMMDDHHMMSS.txt — for an installed Windows
20// build, %LOCALAPPDATA%\Seamly\SeamlyLayout\logs.
21//
22// QStandardPaths test mode redirects AppConfigLocation into a throwaway tree,
23// so the suite never touches the real user configuration.
24//
25// Logger needs no QObject, no window and no event loop, so this suite runs
26// guiless — QTEST_GUILESS_MAIN, not QTEST_MAIN.
27

60// @brief Open one log file under a throwaway AppConfigLocation root.
61// The metadata is set BEFORE init(), which is the order main() must use.
62void LoggerTests::initTestCase()
63{
64 QStandardPaths::setTestModeEnabled(true);
65
66 QCoreApplication::setOrganizationName(QStringLiteral("Seamly"));
67 QCoreApplication::setApplicationName(QStringLiteral("SeamlyLayout"));
68
69 const QString appConfigRoot =
70 QStandardPaths::writableLocation(QStandardPaths::AppConfigLocation);
71 QVERIFY2(!appConfigRoot.isEmpty(), "AppConfigLocation is unavailable in test mode");
72
73 // Plant a stale file so init_removesStaleLogFiles() has something to find.
74 const QString logsDir = appConfigRoot + QStringLiteral("/logs");
75 QVERIFY(QDir().mkpath(logsDir));
76 m_staleFilePath = logsDir + QStringLiteral("/log_240101000000.txt");
77 QFile staleFile(m_staleFilePath);
78 QVERIFY(staleFile.open(QIODevice::WriteOnly | QIODevice::Text));
79 staleFile.write("stale\n");
80 staleFile.close();
81
82 Logger::debugEnabled = true;
83 Logger::init();
84 QVERIFY2(Logger::debugEnabled, "Logger::init() failed to open the log file");
85
86 m_logFilePath = QString::fromUtf8(qgetenv("SEAMLY_LOG_FILE"));
87 QVERIFY2(!m_logFilePath.isEmpty(), "Logger::init() published no SEAMLY_LOG_FILE");
88} // initTestCase()

90void LoggerTests::logDirectory_isLogsUnderTheAppConfigRoot()
91{
92 const QString expected =
93 QStandardPaths::writableLocation(QStandardPaths::AppConfigLocation)
94 + QStringLiteral("/logs");
95
96 QCOMPARE(QFileInfo(m_logFilePath).absolutePath(), QDir(expected).absolutePath());
97 QVERIFY2(QFile::exists(m_logFilePath), "the log file was not created");
98} // logDirectory_isLogsUnderTheAppConfigRoot()

100// The organization folder is what the defect lost: without it the path was
101// .../SeamlyLayout/logs, not .../Seamly/SeamlyLayout/logs.
102void LoggerTests::logDirectory_carriesTheOrganizationAndApplication()
103{
104 const QString path = QDir::fromNativeSeparators(m_logFilePath);
105 QVERIFY2(path.contains(QStringLiteral("/Seamly/SeamlyLayout/logs/")),
106 qPrintable(QStringLiteral("log path was '%1'").arg(path)));
107} // logDirectory_carriesTheOrganizationAndApplication()
108
109void LoggerTests::logDirectory_isNotTheLegacyOutputDirectory()
110{
111 const QString path = QDir::fromNativeSeparators(m_logFilePath);
112 QVERIFY2(!path.contains(QStringLiteral("/output/")),
113 qPrintable(QStringLiteral("log path was '%1'").arg(path)));
114} // logDirectory_isNotTheLegacyOutputDirectory()

116void LoggerTests::logFileName_isTimestamped()
117{
118 const QString fileName = QFileInfo(m_logFilePath).fileName();
119 const QRegularExpression pattern(QStringLiteral("^log_\\d{12}\\.txt$"));
120 QVERIFY2(pattern.match(fileName).hasMatch(),
121 qPrintable(QStringLiteral("file name was '%1'").arg(fileName)));
122} // logFileName_isTimestamped()

124// Each run starts a clean directory, so an old session's file cannot be mistaken
125// for the current one.
126void LoggerTests::init_removesStaleLogFiles()
127{
128 QVERIFY2(!QFile::exists(m_staleFilePath), "the stale log file survived Logger::init()");
129} // init_removesStaleLogFiles()
