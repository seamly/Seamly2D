35/// @brief Open or reload the AdjustWindow with the provided output/adjust_dom.svg file.
36// - On first call, creates the window, connects its signals, and shows it.
37// - On subsequent calls, reloads the scene with the new SVG and bbox JSON
38// so the window reflects the latest layout state.
39
40void AdjustController::launchAdjustWindow(const QString& svgPath,
41 const QString& bboxJson)
42{
43 if (!m_window) {
44 // First launch — create the window and wire signals.
45 Logger::log(QStringLiteral("===========ENTER ADJUST MODE=========="));
46 Logger::log(QStringLiteral("AdjustController::launchAdjustWindow(): creating AdjustWindow"));
47
48 m_window = new AdjustWindow(svgPath, bboxJson);
49
50 // When user applies/enters, Forward AdjustWindow::accepted() as applyRequested().
51 connect(m_window, &AdjustWindow::accepted,
52 this, &AdjustController::applyRequested);
53
54 // When user saves, Forward AdjustWindow::saveRequested() as saveRequested().
55 connect(m_window, &AdjustWindow::saveRequested,
56 this, &AdjustController::saveRequested);
57
58 // Forward AdjustWindow::cancelled() as cancelRequested().
59 connect(m_window, &AdjustWindow::cancelled,
60 this, &AdjustController::cancelRequested);
61
62 // Forward title-bar X closes as immediate abandon/discard.
63 connect(m_window, &AdjustWindow::abandoned,
64 this, &AdjustController::abandonRequested);
65
66 // Clear our pointer when the user closes the window via the OS button,
67 // so the next launchAdjustWindow() call creates a fresh window.
68 connect(m_window, &QObject::destroyed,
69 this, [this]() {
70 m_window = nullptr;
71 Logger::log(QStringLiteral("AdjustController: AdjustWindow destroyed"));
72 }); // destroyed lambda
73 } else {
74 // Subsequent call — reload scene without destroying the window.
75 Logger::log(QStringLiteral("AdjustController::launchAdjustWindow(): reloading AdjustWindow"));
76 // Note: AdjustWindow::reload() clears all existing items and creates new ones, so the window reflects the latest layout state.
77 m_window->reload(svgPath, bboxJson);
78 } // if no window yet
79
80 m_window->show();
81 m_window->raise();
82 m_window->activateWindow();
83} // launchAdjustWindow
