78qreal NewPieceDefaults::fitGrainlineLength(qreal length, qreal arrowLength)
79{
80 qreal fitted = length;
81 if (length < arrowLength * 2)
82 {
83 fitted = arrowLength * 2.1;
84 }
85 return fitted;

89QVector<VLabelTemplateLine> NewPieceDefaults::readLabelTemplate(const QString &userFile, const QString &builtInFile)
90{
91 // The user's file wins. The resource copy covers a data root that was never seeded or a template that was deleted.
92 QString filename = userFile;
93 if (!QFileInfo::exists(filename))
94 {
95 filename = builtInFile;
96 }
97
98 QVector<VLabelTemplateLine> lines;
99 if (QFileInfo::exists(filename))
100 {
101 VLabelTemplate labelTemplate;
102 labelTemplate.setXMLContent(VLabelTemplateConverter(filename).Convert());
103 lines = labelTemplate.ReadLines();
104 }
105 return lines;
106}
