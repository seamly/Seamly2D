47 // @brief Convert a file path to the native OS separator format.
48 // Windows: forward slashes → backslashes.
49 // macOS/Linux: backslashes → forward slashes.
50 static QString toNativePath(const QString &path)
51 {
52 return QDir::toNativeSeparators(path);
53 } // toNativePath

55 // @brief Check if an executable is a Windows Store (UWP/MSIX) app.
56 // Windows Store apps are registered as App Execution Aliases — NTFS reparse
57 // points that QProcess::startDetached cannot launch directly. Detecting them
58 // lets us skip the slow QProcess failure and go straight to QDesktopServices.
59 // On non-Windows platforms, always returns false.
60 static bool isStoreApp(const QString &exePath)
61 {
62#ifdef Q_OS_WIN
63 // App Execution Aliases are NTFS reparse points; real .exe files are not.
64 QString native = toNativePath(exePath);
65 DWORD attrs = GetFileAttributesW(
66 reinterpret_cast<const wchar_t *>(native.utf16()));
67 if (attrs != INVALID_FILE_ATTRIBUTES
68 && (attrs & FILE_ATTRIBUTE_REPARSE_POINT)) {
69 return true; // if reparse point — UWP/Store app alias
70 } // if reparse point
71 return false; // traditional desktop executable
72#else
73 Q_UNUSED(exePath);
74 return false; // non-Windows — no Store apps
75#endif
76 } // isStoreApp
77
78 // @brief Detect whether this process is running from within a mounted AppImage.
79 //
80 // Task 17: an AppImage mounts its payload read-only (a FUSE-mounted squashfs), so any
81 // exe-relative writable path (settings, input/output folders, debug logs) that works for
82 // a normal Linux install fails silently inside one — the same problem Task 16 found for
83 // a signed, notarized macOS .app bundle. The AppImage runtime sets the APPIMAGE
84 // environment variable (absolute path to the .AppImage file) in every process it execs
85 // (see https://docs.appimage.org/packaging-guide/environment-variables.html), so checking
86 // for it is enough to tell the two cases apart.
87 //
88 // The check itself is a plain environment-variable read with no OS-specific API, so it is
89 // safe to call on every platform — it is simply always false on Windows/macOS, where the
90 // variable is never set. This also lets unit tests exercise the AppImage fallback path on
91 // any host by setting the variable directly, unlike the compile-time Q_OS_MACOS branches.
92 static bool isAppImage()
93 {
94 return qEnvironmentVariableIsSet("APPIMAGE");
95 } // isAppImage
96
97 // @brief Detect whether this process is running inside a Flatpak sandbox.
98 //
99 // Task 18: a Flatpak mounts the application payload at the /app prefix read-only (the
100 // executable lives at /app/bin/...), so any exe-relative writable path (settings,
101 // input/output folders, debug logs) that works for a normal Linux install fails silently
102 // inside one — the same problem Task 16 found for a signed, notarized macOS .app bundle
103 // and Task 17 for a FUSE-mounted AppImage. Like the AppImage case, this cannot be told
104 // apart from a native Linux install at compile time and must be detected at runtime.
105 //
106 // Every Flatpak sandbox has the /.flatpak-info file bind-mounted in and exports the
107 // FLATPAK_ID environment variable to the app process
108 // (see https://docs.flatpak.org/en/latest/flatpak-command-reference.html and the
109 // freedesktop app-sandbox conventions), so checking either is enough to recognise the
110 // sandbox. The file check is the robust production signal (present regardless of how the
111 // process was spawned); the environment-variable check additionally lets unit tests
112 // exercise the Flatpak fallback on any host by setting FLATPAK_ID directly, exactly as the
113 // AppImage tests set APPIMAGE.
114 //
115 // The check uses no OS-specific API, so it is safe to call on every platform — it is simply
116 // always false on Windows/macOS and on a non-Flatpak Linux install, where neither the
117 // variable nor the file is present.
118 static bool isFlatpak()
119 {
120 return qEnvironmentVariableIsSet("FLATPAK_ID")
121 || QFileInfo::exists(QStringLiteral("/.flatpak-info"));
122 } // isFlatpak
123
124private:
125 // Non-instantiable — all members are static.
126 Platform() = delete;
