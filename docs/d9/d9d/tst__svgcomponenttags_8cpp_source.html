49/**
50 * @brief squarePoints builds a closed square contour used as path geometry.
51 * @param x left coordinate.
52 * @param y top coordinate.
53 * @param side side length.
54 * @return the four corner points of the square.
55 */
56QVector<QPointF> squarePoints(qreal x, qreal y, qreal side)
57{
58 QVector<QPointF> points;
59 points << QPointF(x, y) << QPointF(x + side, y) << QPointF(x + side, y + side) << QPointF(x, y + side);
60 return points;
61}

76/**
77 * @brief makeTestPiece builds a minimal piece: a square contour holding one
78 * plain internal path and two cutout paths.
79 * @return the layout piece.
80 */
81VLayoutPiece makeTestPiece()
82{
83 VLayoutPiece piece;
84 // setMainPathPoints() is the setter behind getContourPoints() — it writes
85 // the piece's contour (d->contour) after removing duplicate points.
86 piece.setMainPathPoints(squarePoints(10, 10, 180));
87 piece.SetName(QStringLiteral("Test Piece"));
88
89 QVector<VLayoutPiecePath> internalPaths;
90 internalPaths.append(makePath(squarePoints(20, 20, 30), false));
91 piece.setInternalPaths(internalPaths);
92
93 QVector<VLayoutPiecePath> cutoutPaths;
94 cutoutPaths.append(makePath(squarePoints(70, 70, 30), true));
95 cutoutPaths.append(makePath(squarePoints(120, 120, 30), true));
96 piece.setCutoutPaths(cutoutPaths);
97
98 return piece;
99}

102/**
103 * @brief groupsOfType collects all <g> elements of the document carrying the
104 * given data-type attribute value, in document order.
105 * @param doc parsed SVG document.
106 * @param type wanted data-type value.
107 * @return the matching group elements.
108 */
109QVector<QDomElement> groupsOfType(const QDomDocument &doc, const QString &type)
110{
111 QVector<QDomElement> result;
112 const QDomNodeList groups = doc.elementsByTagName(QStringLiteral("g"));
113 for (int i = 0; i < groups.size(); ++i)
114 {
115 const QDomElement group = groups.at(i).toElement();
116 if (group.attribute(QStringLiteral("data-type")) == type)
117 {
118 result.append(group);
119 }
120 }
121 return result;
122}

136/**
137 * @brief CutoutTaggedAsCutPath checks the item-tree contract: cutout paths get
138 * the dedicated "cut_path" item type while plain internal paths keep
139 * "internal_path".
140 */
141void TST_SvgComponentTags::CutoutTaggedAsCutPath() const
142{
143 const VLayoutPiece piece = makeTestPiece();
144 const QScopedPointer<QGraphicsItem> root(piece.GetItem(true));
145
146 int internalCount = 0;
147 int cutoutCount = 0;
148 const QList<QGraphicsItem *> components = root->childItems();
149 for (int i = 0; i < components.size(); ++i)
150 {
151 const QString type = components.at(i)->data(PieceItemData::ItemType).toString();
152 if (type == QLatin1String("internal_path"))
153 {
154 ++internalCount;
155 }
156 else if (type == QLatin1String("cut_path"))
157 {
158 ++cutoutCount;
159 }
160 }
161
162 QCOMPARE(internalCount, 1);
163 QCOMPARE(cutoutCount, 2);
164}

167/**
168 * @brief ExportedSvgTagsCutPathGroups checks the exported SVG end to end:
169 * cutouts become data-type="cut_path" groups with their own per-piece counter
170 * and "piece-<n>-cut_path-<m>" ids, plain internal paths keep their
171 * independent "internal_path" counter, and every group points back to the
172 * piece via data-parent.
173 */
174void TST_SvgComponentTags::ExportedSvgTagsCutPathGroups() const
175{
176 QTemporaryDir tempDir;
177 QVERIFY2(tempDir.isValid(), "Could not create a temporary directory");
178 const QString filePath = tempDir.filePath(QStringLiteral("component_tags.svg"));
179
180 // Render the piece through the real export pipeline into a temp file.
181 const VLayoutPiece piece = makeTestPiece();
182 QGraphicsScene scene;
183 QGraphicsItem *item = piece.GetItem(true);
184 scene.addItem(item); // scene takes ownership
185
186 QGraphicsRectItem paper(QRectF(0, 0, 400, 400));
187 SvgGenerator generator(&paper, filePath, QStringLiteral("Test Pattern"), QString(), 96);
188 generator.addSvgFromScene(&scene, item);
189 generator.generate();
190
191 QFile file(filePath);
192 QVERIFY2(file.open(QIODevice::ReadOnly), "Generated SVG file could not be opened");
193 QDomDocument doc;
194 QVERIFY2(doc.setContent(&file), "Generated SVG could not be parsed");
195
196 // Two cutouts: own counter starting at 1, structured ids, piece as parent.
197 const QVector<QDomElement> cutouts = groupsOfType(doc, QStringLiteral("cut_path"));
198 QCOMPARE(cutouts.size(), 2);
199 for (int i = 0; i < cutouts.size(); ++i)
200 {
201 const QString number = QString::number(i + 1);
202 QCOMPARE(cutouts.at(i).attribute(QStringLiteral("id")), QStringLiteral("piece-1-cut_path-%1").arg(number));
203 QCOMPARE(cutouts.at(i).attribute(QStringLiteral("data-type-number")), number);
204 QCOMPARE(cutouts.at(i).attribute(QStringLiteral("data-parent")), QStringLiteral("piece-1"));
205 }
206
207 // The plain internal path keeps its own counter, unaffected by the cutouts.
208 const QVector<QDomElement> internals = groupsOfType(doc, QStringLiteral("internal_path"));
209 QCOMPARE(internals.size(), 1);
210 QCOMPARE(internals.at(0).attribute(QStringLiteral("id")), QStringLiteral("piece-1-internal_path-1"));
211 QCOMPARE(internals.at(0).attribute(QStringLiteral("data-type-number")), QStringLiteral("1"));
212 QCOMPARE(internals.at(0).attribute(QStringLiteral("data-parent")), QStringLiteral("piece-1"));
213}
