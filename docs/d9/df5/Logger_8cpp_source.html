5// @file Logger.cpp
6// @brief Implementation of the singleton debug logger.
7//
8// Log lines are written in the format used by the legacy Iced UI logs:
9// [unix_seconds] DEBUG: message
10//
11// The log file is opened once by Logger::init() and kept open for the
12// duration of the process. All writes go to a file named:
13// {appDir}/output/log_{YYMMDDHHMM}.txt
14// {appDir} is the writable AppConfigLocation root, not the install directory,
15// wherever the install directory is read-only or needs administrator rights:
16// on macOS (a notarized .app bundle — Task 16), on Windows (C:\Program Files),
17// inside a mounted Linux AppImage (Platform::isAppImage() — Task 17), and
18// inside a Flatpak sandbox (Platform::isFlatpak() — Task 18). Only an ordinary
19// Linux install still writes beside the executable.

58// @brief Open the log file.
59// The logs/ directory is created under the application binary directory
60// if it does not already exist. The file name encodes the startup time
61// as YYMMDDHHMM so each run gets its own file.
62void Logger::init()
63{
64 if (!debugEnabled) return; // logging disabled — do not create files
65
66#if defined(Q_OS_MACOS) || defined(Q_OS_WIN)
67 // The install root is not writable, so logs go under the writable
68 // AppConfigLocation root - the same "Seamly/SeamlyLayout" tree the settings
69 // and preferences files already use.
70 //
71 // macOS (Task 16): a signed, notarized .app bundle is read-only.
72 // Windows: the MSI installs into C:\Program Files\SeamlyApps, which a
73 // standard user cannot write. The exe-relative path appeared to work only
74 // for administrators, and left an output\ directory inside Program Files
75 // that no uninstall removes, because the installer does not own it.
76 QString logsDir = QStandardPaths::writableLocation(QStandardPaths::AppConfigLocation);
77 if (logsDir.isEmpty()) {
78 logsDir = QCoreApplication::applicationDirPath();
79 } // if AppConfigLocation unavailable
80 logsDir += QStringLiteral("/output");
81#else
82 // Task 17: a mounted Linux AppImage is read-only for the same reason a macOS bundle is
83 // — detect it at runtime (Platform::isAppImage(), since unlike macOS this can't be known
84 // at compile time) and fall back to the same writable AppConfigLocation root. A normal
85 // (non-AppImage) Linux install keeps writing logs next to the executable.
86 // Task 18: a Flatpak's /app prefix is read-only in the same way, so Platform::isFlatpak()
87 // (also runtime-only) is treated identically and writes logs under AppConfigLocation too.
88 QString logsDir;
89 if (Platform::isAppImage() || Platform::isFlatpak()) {
90 logsDir = QStandardPaths::writableLocation(QStandardPaths::AppConfigLocation);
91 if (logsDir.isEmpty()) {
92 logsDir = QCoreApplication::applicationDirPath();
93 } // if AppConfigLocation unavailable
94 logsDir += QStringLiteral("/output");
95 } else {
96 // Write log files to the output/ directory next to the executable
97 logsDir = QCoreApplication::applicationDirPath() + QStringLiteral("/output");
98 } // if running from a mounted AppImage
99#endif
100 QDir().mkpath(logsDir);
101 clearOutputDirectory(logsDir);
102
103 // Build the file name: log_{YYMMDDHHmmss}.txt
104 const QString timestamp =
105 QDateTime::currentDateTime().toString(QStringLiteral("yyMMddHHmmss"));
106 const QString filePath =
107 logsDir + QStringLiteral("/log_") + timestamp + QStringLiteral(".txt");
108
109 // Publish the path so Rust log_to_file() can append to the same file.
110 qputenv("SEAMLY_LOG_FILE", filePath.toUtf8());
111
112 s_file.setFileName(filePath);
113
114 if (!s_file.open(QIODevice::WriteOnly | QIODevice::Append | QIODevice::Text)) {
115 debugEnabled = false;
116 return;
117 } // if !open
118
119 s_stream.setDevice(&s_file);
120
121 // Write a session header so log files from multiple runs are easy to distinguish
122 const qint64 unixSec = QDateTime::currentSecsSinceEpoch();
123 s_stream << QStringLiteral("[") << QString::number(unixSec)
124 << QStringLiteral("] DEBUG: Logger::init(): SeamlyLayout session started\n");
125 s_stream << QStringLiteral("[") << QString::number(unixSec)
126 << QStringLiteral("] DEBUG: Logger::init(): cleared existing debug files from output/\n");
127 s_stream.flush();
128} // Logger::init

134// @brief Append one debug line to the open log file.
135// @param message Text written after the "[unix_seconds] DEBUG: " prefix.
136// Does nothing when debugEnabled is false or when the file is not open.
137void Logger::log(const QString &message)
138{
139 if (!debugEnabled) return; // logging disabled
140 if (!s_file.isOpen()) return; // file not open (init() not called or failed)
141
142 const qint64 unixSec = QDateTime::currentSecsSinceEpoch();
143 s_stream << QStringLiteral("[") << QString::number(unixSec)
144 << QStringLiteral("] DEBUG: ") << message << QStringLiteral("\n");
145 s_stream.flush(); // flush immediately so lines appear even if the app crashes
146} // Logger::log

152// @brief Qt message handler — routes all qDebug/qInfo/qWarning/qCritical/qFatal
153// and QML console.log/warn/error output to the log file.
154// Install via qInstallMessageHandler(Logger::messageHandler) after init().
155void Logger::messageHandler(QtMsgType type,
156 const QMessageLogContext &context,
157 const QString &msg)
158{
159 if (!debugEnabled || !s_file.isOpen()) return;
160
161 const qint64 unixSec = QDateTime::currentSecsSinceEpoch();
162 QString level;
163 switch (type) {
164 case QtDebugMsg: level = QStringLiteral("DEBUG"); break;
165 case QtInfoMsg: level = QStringLiteral("INFO"); break;
166 case QtWarningMsg: level = QStringLiteral("WARN"); break;
167 case QtCriticalMsg: level = QStringLiteral("CRIT"); break;
168 case QtFatalMsg: level = QStringLiteral("FATAL"); break;
169 }
170
171 // Include source file:line when available (C++ calls only; QML has no context)
172 QString location;
173 if (context.file) {
174 location = QStringLiteral(" (") + QString::fromUtf8(context.file)
175 + QStringLiteral(":") + QString::number(context.line)
176 + QStringLiteral(")");
177 }
178
179 s_stream << QStringLiteral("[") << QString::number(unixSec)
180 << QStringLiteral("] ") << level << QStringLiteral(": ")
181 << msg << location << QStringLiteral("\n");
182 s_stream.flush();
183
184 if (type == QtFatalMsg)
185 abort();
186} // Logger::messageHandler
