28/// @brief Build a minimal bbox JSON string with two non-overlapping pieces.
29QString buildBboxJson()
30{
31 QJsonArray pieces;
32
33 QJsonObject pieceA;
34 pieceA["id"] = QStringLiteral("pieceA");
35 pieceA["x"] = 10.0;
36 pieceA["y"] = 10.0;
37 pieceA["w"] = 30.0;
38 pieceA["h"] = 30.0;
39 pieceA["origin_x_px"] = 0.0;
40 pieceA["origin_y_px"] = 0.0;
41 pieceA["rotation_deg"] = 0.0;
42 pieceA["transform_str"] = QString();
43 pieces.append(pieceA);
44
45 QJsonObject pieceB;
46 pieceB["id"] = QStringLiteral("pieceB");
47 pieceB["x"] = 90.0;
48 pieceB["y"] = 10.0;
49 pieceB["w"] = 30.0;
50 pieceB["h"] = 30.0;
51 pieceB["origin_x_px"] = 0.0;
52 pieceB["origin_y_px"] = 0.0;
53 pieceB["rotation_deg"] = 0.0;
54 pieceB["transform_str"] = QString();
55 pieces.append(pieceB);
56
57 QJsonObject root;
58 root["pieces"] = pieces;
59 return QString::fromUtf8(QJsonDocument(root).toJson(QJsonDocument::Compact));
60}

62/// @brief Write a minimal SVG with a contentRect element to a temporary directory.
63/// @return Absolute path to the written SVG file, or empty string on failure.
64QString writeLayoutSvg(const QString& directoryPath)
65{
66 const QString svgPath = directoryPath + QStringLiteral("/layout.svg");
67 QFile svgFile(svgPath);
68 if (!svgFile.open(QIODevice::WriteOnly | QIODevice::Text)) {
69 return QString();
70 }
71
72 const QByteArray svg =
73 "<svg xmlns=\"http://www.w3.org/2000/svg\" width=\"200\" height=\"200\" viewBox=\"0 0 200 200\">"
74 "<rect id=\"contentRect\" x=\"0\" y=\"0\" width=\"200\" height=\"200\" fill=\"none\" stroke=\"none\"/>"
75 "</svg>";
76
77 svgFile.write(svg);
78 svgFile.close();
79 return svgPath;
80}

139void AdjustControllerTests::signalSpiesAreValid()
140{
141 AdjustController controller;
142
143 // QSignalSpy attaches to a signal by pointer — if the signal doesn't exist this fails.
144 QSignalSpy applySpy (&controller, &AdjustController::applyRequested);
145 QSignalSpy saveSpy (&controller, &AdjustController::saveRequested);
146 QSignalSpy cancelSpy (&controller, &AdjustController::cancelRequested);
147 QSignalSpy abandonSpy(&controller, &AdjustController::abandonRequested);
148
149 QVERIFY(applySpy.isValid());
150 QVERIFY(saveSpy.isValid());
151 QVERIFY(cancelSpy.isValid());
152 QVERIFY(abandonSpy.isValid());
153
154 // No spurious signals should have fired during construction.
155 QCOMPARE(applySpy.count(), 0);
156 QCOMPARE(saveSpy.count(), 0);
157 QCOMPARE(cancelSpy.count(), 0);
158 QCOMPARE(abandonSpy.count(), 0);

161void AdjustControllerTests::launchWindowWithValidSvgDoesNotCrash()
162{
163 QTemporaryDir tempDir;
164 QVERIFY2(tempDir.isValid(), "Temporary directory must be created");
165
166 const QString svgPath = writeLayoutSvg(tempDir.path());
167 QVERIFY2(!svgPath.isEmpty(), "Test SVG file must be written");
168
169 AdjustController controller;
170
171 // First call — creates AdjustWindow, connects signals, and calls show().
172 controller.launchAdjustWindow(svgPath, buildBboxJson());
173
174 // Immediately close to avoid interactive UI in the test runner.
175 controller.closeAdjustWindow();
176 QTest::qWait(0); // drain event loop so WA_DeleteOnClose window is destroyed before test returns
177 // Reaching here means no crash on the first-launch path.
178 QVERIFY(true);
179}

181void AdjustControllerTests::launchWindowTwiceDoesNotCrash()
182{
183 QTemporaryDir tempDir;
184 QVERIFY2(tempDir.isValid(), "Temporary directory must be created");
185
186 const QString svgPath = writeLayoutSvg(tempDir.path());
187 QVERIFY2(!svgPath.isEmpty(), "Test SVG file must be written");
188
189 const QString bboxJson = buildBboxJson();
190 AdjustController controller;
191
192 // First launch creates the window.
193 controller.launchAdjustWindow(svgPath, bboxJson);
194
195 // Second launch must call reload() on the existing window instead of creating a new one.
196 controller.launchAdjustWindow(svgPath, bboxJson);
197
198 // Clean up.
199 controller.closeAdjustWindow();
200 QTest::qWait(0); // drain event loop so WA_DeleteOnClose window is destroyed before test returns
201 // Reaching here means the reload path did not crash.
202 QVERIFY(true);
203}
