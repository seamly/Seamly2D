6// @brief QObject bridge between QML and the QtWidgets AdjustWindow.

8// Registered with the QML engine as "AdjustController" (SeamlyLayout 1.0).
9// QML calls launchAdjustWindow() to open (or reload) the adjust window.

17// Forward declaration — avoids pulling AdjustWindow headers into QML-visible code.

20/// @class AdjustController
21/// @brief QML-accessible controller that owns and manages the AdjustWindow.
22///
23/// Lifetime: one AdjustWindow instance is created on the first call to
24/// launchAdjustWindow() and reused on subsequent calls via AdjustWindow::reload().
25/// The pointer is cleared automatically when the user closes the window via the
26/// OS close button (destroyed() signal).
27///
28/// @par QML usage:
29/// @code
30/// AdjustController {
31/// id: adjustController
32/// onApplyRequested: function(transformsJson) { ... }
33/// onCancelRequested: { ... }
34/// }
35/// adjustController.launchAdjustWindow(svgPath, bboxJson)
36/// @endcode
37class AdjustController : public QObject
38{
39 Q_OBJECT
40
41public:
42 /// @brief Construct the controller (no window created yet).
43 /// @param parent Optional QObject parent.
44 explicit AdjustController(QObject* parent = nullptr);
45
46 /// @brief Open or reload the AdjustWindow with the given layout data.
47 ///
48 /// On first call, creates the window, connects its signals, and shows it.
49 /// On subsequent calls, reloads the scene with the new SVG and bbox JSON
50 /// so the window reflects the latest layout state.
51 ///
52 /// @param svgPath Absolute native path to the layout SVG file.
53 /// @param bboxJson Piece bounding-box JSON string from appController.getPieceBboxes().
54 Q_INVOKABLE void launchAdjustWindow(const QString& svgPath,
55 const QString& bboxJson);
56
57 /// @brief Close the AdjustWindow on demand (called from QML after canvas reload).
58 Q_INVOKABLE void closeAdjustWindow();
59
60signals:
61 /// @brief Forwarded from AdjustWindow::accepted(); carries the transforms JSON.
62 /// @param transformsJson Compact JSON array of {id, transform} objects.
63 /// Called by AdjustWindow::onApplyClicked() after the user clicks Apply and the transforms JSON is built.
64 void applyRequested(const QString& transformsJson);
65
66 /// @brief Forwarded from AdjustWindow::saveRequested().
67 ///
68 /// QML should respond by calling appController.exitAdjustMode(), which saves
69 /// layout_dom and reloads the right canvas via layoutFinished.
70 void saveRequested();
71
72 /// @brief Forwarded from AdjustWindow::cancelled().
73 void cancelRequested();
74
75 /// @brief Forwarded from AdjustWindow::abandoned().
76 void abandonRequested();
77
78private:
79 /// @brief The managed adjust window (nullptr until first launch).
