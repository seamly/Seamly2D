39// @brief RAII helper that sets an environment variable for the scope of a test and
40// guarantees it is unset again when the test function returns.
41//
42// Task 17: PreferencesModelTests exercises Platform::isAppImage() (checks the APPIMAGE
43// environment variable the AppImage runtime sets) by setting the variable directly. A
44// failed QVERIFY/QCOMPARE inside the test body simply `return`s the enclosing function —
45// it does not throw — so a stack-allocated guard's destructor still runs and the variable
46// never leaks into whichever test runs next in the same process.
47class ScopedEnvVar
48{
49public:
50 ScopedEnvVar(const QByteArray &name, const QByteArray &value) : m_name(name)
51 {
52 qputenv(m_name.constData(), value);
53 } // ScopedEnvVar
54 ~ScopedEnvVar() { qunsetenv(m_name.constData()); } // ~ScopedEnvVar
55private:
56 QByteArray m_name;
57}; // class ScopedEnvVar

365void PreferencesModelTests::resetToDefaults_loadsValuesFromDefaultsFile()
366{
367 // Arrange: write a known defaults file to a temp dir, point the model's
368 // preferencesDirectory/preferencesFile at it, then mutate the in-memory
369 // values away from those defaults.
370 QTemporaryDir tempDir;
371 QVERIFY(tempDir.isValid());
372
373 const QString defaultsPath = tempDir.filePath(QStringLiteral("prefs.json"));
374 QJsonObject defaults;
375 defaults[QStringLiteral("input_directory")] = QStringLiteral("/tmp/in");
376 defaults[QStringLiteral("layout_directory")] = QStringLiteral("/tmp/out");
377 defaults[QStringLiteral("dxf_viewer_path")] = QStringLiteral("https://sharecad.org");
378 defaults[QStringLiteral("projector_path")] = QStringLiteral("https://patternprojector.com");
379
380 QFile f(defaultsPath);
381 QVERIFY(f.open(QIODevice::WriteOnly | QIODevice::Truncate));
382 f.write(QJsonDocument(defaults).toJson(QJsonDocument::Indented));
383 f.close();
384
385 PreferencesModel m;
386 m.setPreferencesDirectory(tempDir.path());
387 m.setPreferencesFile(QStringLiteral("prefs.json"));
388
389 // Pollute current values so we know reset actually overwrote them.
390 m.setInputDirectory(QStringLiteral("/somewhere/else"));
391 m.setDxfViewerPath(QStringLiteral("not-the-default"));
392 m.setProjectorPath(QStringLiteral("not-the-default-projector"));
393
394 // Act
395 QVERIFY(m.resetToDefaults());
396
397 // Assert: each scalar field now matches the defaults file.
398 QCOMPARE(m.inputDirectory(), QStringLiteral("/tmp/in"));
399 QCOMPARE(m.layoutDirectory(), QStringLiteral("/tmp/out"));
400 QCOMPARE(m.dxfViewerPath(), QStringLiteral("https://sharecad.org"));
401 QCOMPARE(m.projectorPath(), QStringLiteral("https://patternprojector.com"));
402}
