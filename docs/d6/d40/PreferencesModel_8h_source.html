5// @file PreferencesModel.h
6// @brief QObject model for SeamlyLayout application preferences.
7//
8// Mirrors the Rust AppSettings data model exposed to the Qt frontend.
9// Exposes all fields as Q_PROPERTY items for QML binding and stores
10// application preferences in qt6_seamlylayout.ini.
11//
12// INI keys use snake_case for compatibility with legacy preferences JSON:
13// input_directory, layout_directory, preferences_directory,
14// settings_directory, settings_file, preferences_file,
15// dxf_viewer_path, pdf_viewer_path, png_viewer_path, data_root
16//
17// Registration:
18// Registered at runtime in main.cpp:
19// qmlRegisterType<PreferencesModel>("SeamlyLayout", 1, 0, "PreferencesModel");
20//
21// Usage in QML:
22// PreferencesModel { id: preferencesModel }
23// Text { text: preferencesModel.inputDirectory }
24// preferencesModel.load(preferencesModel.defaultPreferencesFilePath())
25// preferencesModel.save(preferencesModel.defaultPreferencesFilePath())
26

236 // @brief Parse a viewer-path field into (executable, args).
237 // Resolution order:
238 // 1. Strip a leading `file:///` URL prefix if present.
239 // 2. If the whole trimmed string is an existing file on disk, treat it
240 // as the executable with no preset args. This is what makes Browse-
241 // picked Windows paths with spaces (e.g.
242 // `C:\Program Files\Common Files\eDrawings2026\eDrawings.exe`) work
243 // without quoting.
244 // 3. Otherwise parse with QProcess::splitCommand for shell-style
245 // handling (Chrome PWA wrapper, etc.).
246 // @param viewerPath Raw viewer field text.
247 // @return Stringlist where element 0 is the executable path (already
248 // converted to native separators) and the remainder are preset
249 // arguments. Empty list when the input parses to no tokens.
250 static QStringList parseViewerCommand(const QString &viewerPath);
