538/**
539 * @brief Notch lines of a square piece with one slit notch on its top edge.
540 * @param showCutline the node's cutline notch flag.
541 * @param showSeamline the node's seamline notch flag.
542 * @param includeCutlineNotches forwarded to VPiece::createNotchLines().
543 * @param seamAllowancePoints receives the piece's cutline points.
544 */
545QVector<QLineF> squareNotchLines(bool showCutline, bool showSeamline, bool includeCutlineNotches,
546 QVector<QPointF> &seamAllowancePoints)

554 data->UpdateGObject(3, new VPointF(400, 0, "A3", 0, 0));

556 data->UpdateGObject(5, new VPointF(0, 400, "A5", 0, 0));

559 notchNode.setNotch(true);
560 notchNode.setNotchType(NotchType::Slit);
561 notchNode.setNotchSubType(NotchSubType::Straightforward);
562 notchNode.setNotchLength(5);
563 notchNode.setNotchWidth(5);
564 notchNode.setNotchCount(1);
565 notchNode.setShowNotch(showCutline);
566 notchNode.setShowSeamlineNotch(showSeamline);
567
568 VPiece piece;
569 piece.SetSeamAllowance(true);
570 piece.SetSAWidth(notchTestSAWidth);
571 piece.GetPath().Append(VPieceNode(1, Tool::NodePoint));
572 piece.GetPath().Append(notchNode);
573 piece.GetPath().Append(VPieceNode(3, Tool::NodePoint));
574 piece.GetPath().Append(VPieceNode(4, Tool::NodePoint));
575 piece.GetPath().Append(VPieceNode(5, Tool::NodePoint));

614/**
615 * @brief NotchFlagsSelectLine checks that each notch flag puts its notch on its own line, and that the
616 * seam allowance view setting does not change the notch data.
617 */
618void TST_VPiece::NotchFlagsSelectLine() const
619{
620 QFETCH(bool, showCutline);
621 QFETCH(bool, showSeamline);
622 QFETCH(bool, showSeamAllowances);
623 QFETCH(bool, includeCutlineNotches);
624 QFETCH(int, cutlineNotches);
625 QFETCH(int, seamlineNotches);
626
627 const bool savedShowSeamAllowances = qApp->Settings()->showSeamAllowances();
628 qApp->Settings()->setShowSeamAllowances(showSeamAllowances);
629
630 QVector<QPointF> seamAllowancePoints;
631 const QVector<QLineF> notches = squareNotchLines(showCutline, showSeamline, includeCutlineNotches,
632 seamAllowancePoints);
633
634 qApp->Settings()->setShowSeamAllowances(savedShowSeamAllowances);
635
636 const QVector<QPointF> seamline{QPointF(0, 0), QPointF(400, 0)};
637 const qreal notchLength = ToPixel(5, Unit::Mm);
638
639 int onCutline = 0;
640 int onSeamline = 0;
641 for (const QLineF &notch : notches)
642 {
643 QVERIFY(qAbs(notch.length() - notchLength) < 0.01);
644 const bool p1OnCut = isOnPolyline(notch.p1(), seamAllowancePoints);
645 const bool p2OnCut = isOnPolyline(notch.p2(), seamAllowancePoints);
646 const bool p1OnSeam = isOnPolyline(notch.p1(), seamline);
647 const bool p2OnSeam = isOnPolyline(notch.p2(), seamline);
648 if (p1OnCut || p2OnCut)
649 {
650 ++onCutline;
651 }
652 else if (p1OnSeam || p2OnSeam)
653 {
654 ++onSeamline;
655 }
656 }
657
658 QCOMPARE(notches.size(), cutlineNotches + seamlineNotches);
659 QCOMPARE(onCutline, cutlineNotches);
660 QCOMPARE(onSeamline, seamlineNotches);
661}
