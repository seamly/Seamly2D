5// @file SettingsModel.h
6// @brief QObject model for SeamlyLayout layout settings.
7//
8// Mirrors the Rust LayoutSettings data model exposed to the Qt frontend.
9// Exposes all fields as Q_PROPERTY items for QML binding and handles
10// JSON load/save, unit conversion, and paper/tile size lookup.
11//
12// Registration:
13// Registered at runtime in main.cpp:
14// qmlRegisterType<SettingsModel>("SeamlyLayout", 1, 0, "SettingsModel");
15//
16// Usage in QML:
17// SettingsModel { id: settingsModel }
18// Text { text: settingsModel.unit }
19// settingsModel.load(SettingsModel::defaultSettingsFilePath())
20// settingsModel.save(SettingsModel::defaultSettingsFilePath())
21

39 // @brief Piece-arrangement mode: "alongGrainline" | "withNap".
40 // alongGrainline → grain-up baseline, allow {0°, 180°}
41 // withNap → grain-up baseline, {0°} only (no flip)
42 //
43 Q_PROPERTY(QString layoutMode READ layoutMode WRITE setLayoutMode NOTIFY layoutModeChanged)

45 // @brief Rotation step in degrees.
46 // In current modes this is used only when layoutMode == "withNap":
47 // 0.0 = pieces point up
48 // 180.0 = pieces point down
49 Q_PROPERTY(double rotationStep READ rotationStep WRITE setRotationStep NOTIFY rotationStepChanged)

51 // @brief Whether fabric is cut on the fold (doubles usable width).
52 Q_PROPERTY(bool fabricFolded READ fabricFolded WRITE setFabricFolded NOTIFY fabricFoldedChanged)
53
54 // @brief Minimum clearance between adjacent placed pieces, in active units.
55 // The packer enforces at least this much space between any two pieces in
56 // both axes; sub-unit values (e.g. 0.05") are honored at scaled-int
57 // precision in the polygon path.
58 Q_PROPERTY(double pieceGap READ pieceGap WRITE setPieceGap NOTIFY pieceGapChanged)
59
60 // @brief Piece gap in pixels at 96 px/in (derived from pieceGap + unit).
