36namespace
37{
38#ifdef Q_OS_WIN
39/** The key smsi.wxs writes its answers to. One place, so a rename cannot drift. */
40const QLatin1String installKeyPath("HKEY_LOCAL_MACHINE\\SOFTWARE\\Seamly\\Seamly2D");
41
42//---------------------------------------------------------------------------------------------------------------------
43/**
44 * @brief readInstalledPath reads one path value out of the install key.
45 *
46 * Registry64Format rather than NativeFormat: the apps are 64-bit today, so the two agree,
47 * but the MSI is x64 and always writes the 64-bit view.
48 *
49 * @param name value name under the install key.
50 * @return the value cleaned into Qt's '/' separator form; empty when absent or blank.
51 */
52QString readInstalledPath(const QString &name)
53{
54 const QSettings installKey(installKeyPath, QSettings::Registry64Format);
55 const QString recorded = installKey.value(name).toString().trimmed();
56 if (recorded.isEmpty())
57 {
58 return QString();
59 }
60
61 return QDir::cleanPath(QDir::fromNativeSeparators(recorded));
62}
63#endif
64} // anonymous namespace
65
66namespace InstallerRecord

69//---------------------------------------------------------------------------------------------------------------------
70/**
71 * @brief dataRoot returns the user-data root the Windows installer recorded.
72 *
73 * Setup asks "Where do you keep your work?" and writes the answer to the DataRoot value.
74 * Without this the answer was inert: every app resolved its own default instead, so a user
75 * who was promised C:\Users<user>\Documents\SeamlyData got <Documents>/Seamly
76 * (Task InstWinX64.00).
77 *
78 * The value is machine-wide, so each user adopts it once, on that user's first run, into
79 * their own settings file. A user who later changes the root in Preferences keeps that
80 * choice — VCommonSettings::initializeDataRoot() reads the settings file first and never
81 * comes back here.
82 *
83 * Setup leaves the value empty when nothing chose a root, which is the signal to use the
84 * built-in default.
85 *
86 * @return absolute path recorded by the installer, in Qt's '/' separator form; empty when
87 * no installer recorded one, and always empty off Windows.
88 */
89QString dataRoot()
90{
91#ifdef Q_OS_WIN
92 return readInstalledPath(QStringLiteral("DataRoot"));
93#else
94 return QString();
95#endif
96}
97
98} // namespace InstallerRecord
