36/// @brief Parse a numeric SVG attribute to double; defaults to 0.0 when invalid.
37double parseSvgNumber(const QString& value)
38{
39 bool ok = false;
40 const double parsed = value.trimmed().toDouble(&ok);
41 return ok ? parsed : 0.0;
42}

44/// @brief Find the first DOM element by id, searching depth-first.
45QDomElement findElementById(const QDomElement& root, const QString& targetId)
46{
47 if (root.isNull()) {
48 return QDomElement();
49 }
50
51 if (root.attribute(QStringLiteral("id")) == targetId) {
52 return root;
53 }
54
55 QDomElement child = root.firstChildElement();
56 while (!child.isNull()) {
57 const QDomElement hit = findElementById(child, targetId);
58 if (!hit.isNull()) {
59 return hit;
60 }
61 child = child.nextSiblingElement();
62 }
63
64 return QDomElement();
65}

140int AdjustScene::findUndoStackPositionForPiece(const PieceOverlayItem* item) const
141{
142 if (!item) {
143 return -1;
144 }
145
146 for (int i = m_globalUndoStack.size() - 1; i >= 0; --i) {
147 const int opIndex = m_globalUndoStack[i];
148 if (opIndex < 0 || opIndex >= m_operationLog.size()) {
149 continue;
150 }
151
152 const SceneOperation& op = m_operationLog[opIndex];
153 if (op.pieceId == item->pieceId() && op.pieceHistoryIndex == item->historyIndex()) {
154 return i;
155 }
156 }
157
158 return -1;
159}

241bool AdjustScene::undoSelectedPiece()
242{
243 PieceOverlayItem* item = selectedPiece();
244 if (!item || !item->canUndo()) {
245 return false;
246 }
247
248 const int stackPos = findUndoStackPositionForPiece(item);
249 if (stackPos < 0) {
250 return false;
251 }
252
253 const int opIndex = m_globalUndoStack[stackPos];
254 m_globalUndoStack.removeAt(stackPos);
255 if (!item->undo()) {
256 m_globalUndoStack.insert(stackPos, opIndex);
257 return false;
258 }
259
260 m_globalRedoStack.append(opIndex);
261 refreshActorConflictHighlight(item->pieceId());
262 emitUndoRedoAvailability();
263 return true;
264}

291bool AdjustScene::undoGlobalOperation()
292{
293 if (!canGlobalUndo()) {
294 return false;
295 }
296
297 const int opIndex = m_globalUndoStack.takeLast();
298 if (opIndex < 0 || opIndex >= m_operationLog.size()) {
299 return false;
300 }
301
302 const SceneOperation& op = m_operationLog[opIndex];
303 PieceOverlayItem* item = findPieceById(op.pieceId);
304 if (!item || item->historyIndex() != op.pieceHistoryIndex || !item->undo()) {
305 m_globalUndoStack.append(opIndex);
306 return false;
307 }
308
309 m_globalRedoStack.append(opIndex);
310 refreshActorConflictHighlight(op.pieceId);
311 emitUndoRedoAvailability();
312 return true;
313}
314
315bool AdjustScene::redoGlobalOperation()
316{
317 if (!canGlobalRedo()) {
318 return false;
319 }
320
321 const int opIndex = m_globalRedoStack.takeLast();
322 if (opIndex < 0 || opIndex >= m_operationLog.size()) {
323 return false;
324 }
325
326 const SceneOperation& op = m_operationLog[opIndex];
327 PieceOverlayItem* item = findPieceById(op.pieceId);
328 if (!item || (item->historyIndex() + 1) != op.pieceHistoryIndex || !item->redo()) {
329 m_globalRedoStack.append(opIndex);
330 return false;
331 }
332
333 m_globalUndoStack.append(opIndex);
334 refreshActorConflictHighlight(op.pieceId);
335 emitUndoRedoAvailability();
336 return true;
337}

361/// @brief Clear and reload the scene from an SVG path and bbox JSON string.
362void AdjustScene::loadLayout(const QString& svgPath, const QString& bboxJson)
363{
364 saveHistorySnapshots();
365
366 // Remove all existing items and reset tracking containers.
367 clear();
368 // Note: clear() deletes all items, including the background and pieces, so we must reset our pointers and lists to avoid dangling references.
369 m_pieces.clear();
370 // Reset the background pointer since the old background item has been deleted by clear().
371 m_background = nullptr;
372 m_contentRect = QRectF();
373 m_hasContentRect = false;
374
375 // Parse contentRect bounds from the current SVG so operation-time validation
376 // can catch pieces moved outside printable/content margins.
377 QFile svgFile(svgPath);
378 if (svgFile.open(QIODevice::ReadOnly | QIODevice::Text)) {
379 QDomDocument doc;
380 QString parseError;
381 int parseLine = 0;
382 int parseColumn = 0;
383 if (doc.setContent(&svgFile, &parseError, &parseLine, &parseColumn)) {
384 const QDomElement root = doc.documentElement();
385 const QDomElement contentRect = findElementById(root, QStringLiteral("contentRect"));
386 if (!contentRect.isNull()) {
387 const double x = parseSvgNumber(contentRect.attribute(QStringLiteral("x")));
388 const double y = parseSvgNumber(contentRect.attribute(QStringLiteral("y")));
389 const double w = parseSvgNumber(contentRect.attribute(QStringLiteral("width")));
390 const double h = parseSvgNumber(contentRect.attribute(QStringLiteral("height")));
391 if (w > 0.0 && h > 0.0) {
392 m_contentRect = QRectF(x, y, w, h);
393 m_hasContentRect = true;
394 }
395 }
396 } else {
397 qWarning() << "[AdjustScene] loadLayout(): SVG parse failed for contentRect at"
398 << parseLine << ":" << parseColumn << parseError;
399 }
400 svgFile.close();
401 } else {
402 qWarning() << "[AdjustScene] loadLayout(): could not open SVG for contentRect:" << svgPath;
403 }
404
405 // --- SVG background --------------------------------------------------
406
407 // Create a non-interactive SVG background at z=0.
408 QGraphicsSvgItem* bg = new QGraphicsSvgItem(svgPath);
409 bg->setFlag(QGraphicsItem::ItemIsMovable, false);
410 bg->setFlag(QGraphicsItem::ItemIsSelectable, false);
411 bg->setZValue(0.0);
412 addItem(bg);
413 // Keep track of the background item so we can size the scene to fit it and avoid dangling pointers after clear().
414 m_background = bg;
415
416 // Size the scene to exactly fit the SVG background.
417 setSceneRect(bg->boundingRect());
418
419 // --- Parse bbox JSON -------------------------------------------------
420
421 const QByteArray jsonBytes = bboxJson.toUtf8();
422 const QJsonDocument doc = QJsonDocument::fromJson(jsonBytes);
423
424 if (!doc.isObject()) {
425 // Malformed JSON — background is shown but no pieces are created.
426 return;
427 } // if bad JSON
428
429 // Extract the pieces array from the JSON; each entry describes one piece's bbox data: initial position, size, relative origin, and rotation. Used to create overlay items and track their transforms.
430 const QJsonObject root = doc.object();
431 const QJsonArray pieces = root.value("pieces").toArray();
432
433 // Iterate over piece bbox data and create an interactive PieceOverlayItem for each
434 for (const QJsonValue& val : pieces) {
435 if (!val.isObject()) {
436 // Skip malformed entries.
437 continue;
438 } // if not object
439
440 const QJsonObject obj = val.toObject();
441
442 const QString id = obj.value("id").toString();
443 // Human-readable piece name from the Seamly2D handoff (data-name → data-letter
444 // → id, resolved on the Rust side). Display only; `id` stays the identity.
445 const QString label = obj.value("label").toString();
446 const double x = obj.value("x").toDouble(0.0); // canvas space, absolute coords
447 const double y = obj.value("y").toDouble(0.0); // canvas space, absolute coords
448 const double w = obj.value("w").toDouble(1.0); // can't be 0
449 const double h = obj.value("h").toDouble(1.0); // can't be 0
450 const double ox = obj.value("origin_x_px").toDouble(0.0); // local space
451 const double oy = obj.value("origin_y_px").toDouble(0.0); // local space
452 const double rotDeg = obj.value("rotation_deg").toDouble(0.0); // relative to local origin
453 const QString transformStr = obj.value("transform_str").toString(); // accumulated SVG transform from prior Accept cycles
454
455 qDebug() << "[AdjustScene] loadLayout(): " << id
456 << "pos=(" << x << "," << y << ")"
457 << "size=(" << w << "x" << h << ")"
458 << "origin=(" << ox << "," << oy << ")"
459 << "rotDeg=" << rotDeg
460 << "transformStr='" << transformStr << "'";
461
462 // Create an interactive piece overlay item at z=1.
463 PieceOverlayItem* item = new PieceOverlayItem(id, ox, oy, x, y, w, h, rotDeg, transformStr); // piece placement (x,y) is in absolute canvas coords; rotation is relative to piece's (ox, oy) in piece local space coords; in this application (ox, oy) is always the default value (0,0) at bbox top-left corner.
464 // Attach the display name so the piece's context menu reads "Front Bodice",
465 // not "piece-7". Ignored when the layout carried no label (untagged SVG).
466 item->setDisplayLabel(label);
467 // Set the item's z-value above the background so it receives mouse events.
468 item->setZValue(1.0);
469 // Add the item to the scene
470 addItem(item);
471 // Note: the scene takes ownership of the item, so we do not delete it directly; we just keep track of it in m_pieces for later reference and cleanup.
472 m_pieces.append(item);
473 restoreHistorySnapshot(item);
474 } // for each piece entry
475
476 qDebug() << "[AdjustScene] loadLayout: loaded" << m_pieces.size() << "piece(s)";
477 emitUndoRedoAvailability();
478} // void loadLayout()

484/// @brief Serialise each piece's current SVG transform to a compact JSON array.
485/// Only pieces that have moved or rotated from their load position are included in the output.
486/// Called by AdjustWindow::onApplyClicked() to gather the updated transforms before emitting accepted() and triggering the main application reload.
487QString AdjustScene::getMovedTransform() const
488{
489 QJsonArray transform_arr;
490
491 qDebug() << "[AdjustScene] getMovedTransform(): checking" << m_pieces.size() << "piece(s):";
492 // Iterate over pieces and build a JSON array of {id, transform} objects for those that have moved or rotated.
493 for (const PieceOverlayItem* item : m_pieces) {
494 // process only the item that moved or rotated
495 if (!item->hasMoved()) {
496 continue;
497 }
498 // get the old transform string from the item before rebuilding it
499 const QString xf_old = item->transformStr();
500 qDebug() << "[AdjustScene] getMovedTransform(): '" + item->pieceId() + "' has moved, old transform string: '" + xf_old + "'";
501
502 // buildTransform() returns the current transform string directly, so use
503 // that value for Apply serialization instead of the stale cached string.
504 qDebug() << "[AdjustScene] getMovedTransform(): calling buildTransform() to build new transform string";
505 const QString xf_new = item->buildTransform();
506
507 // Build json object with new transform string
508 qDebug() << "[AdjustScene] getMovedTransform(): '" + item->pieceId() + "' transform='" + xf_new + "'";
509 QJsonObject obj;
510 obj["id"] = item->pieceId();
511 obj["transform"] = xf_new;
512
513 // Append transform object to json array.
514 transform_arr.append(obj);
515 }
516 // serialize the json array
517 const QString transform_str= QString::fromUtf8(QJsonDocument(transform_arr).toJson(QJsonDocument::Compact));
518 qDebug() << "[AdjustScene] getMovedTransform(): final transforms JSON:" << transform_str;
519
520 // return to onApplyClicked()
521 return transform_str;
522} // getMovedTransform()

557/// @brief Return ids of pieces whose axis-aligned bounding rects intersect.
558QStringList AdjustScene::checkOverlaps() const
559{
560 QStringList conflicting;
561
562 // O(n²) pair-wise check — piece counts are small (typically < 100).
563 for (int i = 0; i < m_pieces.size(); ++i) {
564 const QRectF r1 = m_pieces[i]->sceneBoundingRect();
565
566 for (int j = i + 1; j < m_pieces.size(); ++j) {
567 const QRectF r2 = m_pieces[j]->sceneBoundingRect();
568
569 if (r1.intersects(r2)) {
570 // Add each piece id at most once.
571 const QString id1 = m_pieces[i]->pieceId();
572 const QString id2 = m_pieces[j]->pieceId();
573
574 if (!conflicting.contains(id1)) {
575 conflicting.append(id1);
576 } // if id1 not already listed
577
578 if (!conflicting.contains(id2)) {
579 conflicting.append(id2);
580 } // if id2 not already listed
581 } // if rects intersect
582 } // for j (inner piece)
583 } // for i (outer piece)
584
585 return conflicting;
586}

681namespace {
682/// @brief Minimum gap (scene units) required to consider two pieces non-touching.
683/// Touching pieces are excluded so "Abut" never offers a no-op.
684constexpr double kAbutGapTol = 0.5;
685
686/// @brief Return true if [a0, a1] and [b0, b1] overlap on a 1-D axis (open intervals).
687inline bool intervalsOverlap(double a0, double a1, double b0, double b1)
688{
689 return a1 > b0 && b1 > a0;
690}
691} // anonymous namespace
692
693/// @brief Find the nearest piece to the left of the given piece.
694/// Returns the immediate left neighbor (overlapping on the Y axis) only when
695/// there is a real gap > kAbutGapTol. If the immediate neighbor is already
696/// touching, returns nullptr so "Abut Left" is disabled (no-op suppression).
697PieceOverlayItem* AdjustScene::findNearestPieceLeft(const PieceOverlayItem* piece) const
698{
699 if (!piece) {
700 return nullptr;
701 }
702
703 const QRectF myBbox = piece->sceneBoundingRect();
704 PieceOverlayItem* nearest = nullptr;
705 double minDistance = std::numeric_limits<double>::max();
706
707 for (PieceOverlayItem* other : m_pieces) {
708 if (!other || other == piece) {
709 continue;
710 }
711
712 const QRectF otherBbox = other->sceneBoundingRect();
713
714 if (!intervalsOverlap(myBbox.top(), myBbox.bottom(), otherBbox.top(), otherBbox.bottom())) {
715 continue;
716 }
717
718 // Other must sit at or above my left edge (allow tiny overlap as "touching").
719 const double distance = myBbox.left() - otherBbox.right();
720 if (distance < -kAbutGapTol) {
721 continue;
722 }
723 if (distance < minDistance) {
724 minDistance = distance;
725 nearest = other;
726 }
727 }
728
729 return (nearest && minDistance > kAbutGapTol) ? nearest : nullptr;
730}

732/// @brief Find the nearest piece to the right of the given piece.
733/// Returns the immediate right neighbor (overlapping on the Y axis) only when
734/// there is a real gap > kAbutGapTol. Already-touching neighbor → nullptr.
735PieceOverlayItem* AdjustScene::findNearestPieceRight(const PieceOverlayItem* piece) const
736{
737 if (!piece) {
738 return nullptr;
739 }
740
741 const QRectF myBbox = piece->sceneBoundingRect();
742 PieceOverlayItem* nearest = nullptr;
743 double minDistance = std::numeric_limits<double>::max();
744
745 for (PieceOverlayItem* other : m_pieces) {
746 if (!other || other == piece) {
747 continue;
748 }
749
750 const QRectF otherBbox = other->sceneBoundingRect();
751
752 if (!intervalsOverlap(myBbox.top(), myBbox.bottom(), otherBbox.top(), otherBbox.bottom())) {
753 continue;
754 }
755
756 const double distance = otherBbox.left() - myBbox.right();
757 if (distance < -kAbutGapTol) {
758 continue;
759 }
760 if (distance < minDistance) {
761 minDistance = distance;
762 nearest = other;
763 }
764 }
765
766 return (nearest && minDistance > kAbutGapTol) ? nearest : nullptr;
767}
