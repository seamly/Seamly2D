5// @file StartupOptions.cpp
6// @brief Implementation of the SeamlyLayout startup contract — a positional
7// <svg-file>, or an SVG document on standard input.
8//
9// Control flow of parse():
10// 1. Reject an empty argument list outright (nothing to parse).
11// 2. Run QCommandLineParser::parse() — the non-exiting sibling of process(),
12// chosen so this function can be unit tested and can never terminate the
13// application behind main()'s back.
14// 3. Honour --help / --version by handing their text back as
15// Status::ShowInformation; main() displays it and exits 0.
16// 4. Reject --svg-stdin together with a positional file: one document per
17// process, and two sources would silently pick a winner.
18// 5. With --svg-stdin, read standard input to end and validate it as an SVG
19// document (Seamly2D.5).
20// 6. Otherwise require at most one positional argument, and validate it as an
21// existing, readable .svg file, storing its ABSOLUTE path (the app-wide
22// rule: never pass relative paths around, the working directory of a
23// launched process is not the user's).
24//
25// Every failure produces a sentence the user can act on, because the only thing
26// worse than the old behaviour (silently ignoring the argument) is silently
27// ignoring it with an extra step.
28

55// @brief Build the text shown for --version.
56// Falls back to the product name alone when the caller has not set the
57// application metadata yet (QCoreApplication::setApplicationVersion()).
58// @return e.g. "SeamlyLayout 0.1.0".
59QString versionText()
60{
61 const QString name = QCoreApplication::applicationName();
62 const QString version = QCoreApplication::applicationVersion();
63
64 if (name.isEmpty()) {
65 // No metadata set — name the product so the output is never empty.
66 return version.isEmpty() ? QStringLiteral("SeamlyLayout")
67 : QStringLiteral("SeamlyLayout ") + version;
68 } // if name.isEmpty()
69
70 return version.isEmpty() ? name : name + QLatin1Char(' ') + version;
71} // versionText

82// @brief Parse and validate a SeamlyLayout command line.
83// @param arguments Full argument list including the program name at index 0,
84// as returned by QCoreApplication::arguments().
85// @param standardInput Device the SVG document is read from when --svg-stdin is
86// set; nullptr when the caller offers no such channel.
87// @return A fully populated StartupOptions; never throws, never exits.
88StartupOptions StartupOptions::parse(const QStringList &arguments, QIODevice *standardInput)
89{
90 StartupOptions options;
91
92 if (arguments.isEmpty()) {
93 // Not even a program name — nothing to open, and QCommandLineParser
94 // would warn about the missing argv[0].
95 return options;
96 } // if arguments.isEmpty()
97
98 QCommandLineParser parser;
99 parser.setApplicationDescription(QStringLiteral(
100 "SeamlyLayout — pattern layout application of the Seamly Application Suite.\n"
101 "Opens an SVG file, or the SVG document Seamly2D's Layout Mode sends on "
102 "standard input."));
103
104 const QCommandLineOption helpOption = parser.addHelpOption();
105 const QCommandLineOption versionOption = parser.addVersionOption();
106
107 const QCommandLineOption svgStdinOption(
108 QStringList() << svgStdinOptionName,
109 QStringLiteral("Read the SVG document to open from standard input. "
110 "Used by Seamly2D's Layout Mode handoff."));
111 parser.addOption(svgStdinOption);
112
113 const QCommandLineOption documentNameOption(
114 QStringList() << documentNameOptionName,
115 QStringLiteral("Name of the document read from standard input; used for "
116 "default export file names."),
117 QStringLiteral("name"));
118 parser.addOption(documentNameOption);
119
120 parser.addPositionalArgument(
121 QStringLiteral("svg-file"),
122 QStringLiteral("SVG pattern file to open at startup (optional)."),
123 QStringLiteral("[svg-file]"));
124
125 // parse(), not process(): process() prints to a console this GUI-subsystem
126 // application does not have on Windows, and calls exit() on error.
127 if (!parser.parse(arguments)) {
128 // Unknown option, missing option value, etc. errorText() is already a
129 // complete sentence ("Unknown option 'foo'.").
130 options.m_status = Status::Failed;
131 options.m_message = parser.errorText();
132 return options;
133 } // if !parser.parse
134
135 if (parser.isSet(versionOption)) {
136 // --version wins over everything else, matching every other CLI.
137 options.m_status = Status::ShowInformation;
138 options.m_message = versionText();
139 return options;
140 } // if version requested
141
142 if (parser.isSet(helpOption)) {
143 options.m_status = Status::ShowInformation;
144 options.m_message = parser.helpText();
145 return options;
146 } // if help requested
147
148 const QStringList positional = parser.positionalArguments();
149 const bool readStandardInput = parser.isSet(svgStdinOption);
150
151 if (readStandardInput && !positional.isEmpty()) {
152 // Two sources for one canvas. Refuse rather than pick a winner the
153 // caller cannot predict.
154 options.m_status = Status::Failed;
155 options.m_message = QStringLiteral(
156 "--%1 reads the document from standard input, so no file may be given as well:\n%2")
157 .arg(svgStdinOptionName)
158 .arg(positional.join(QStringLiteral("\n")));
159 return options;
160 } // if both transports requested
161
162 if (readStandardInput) {
163 return parseStandardInput(parser.value(documentNameOption), standardInput);
164 } // if the document arrives on standard input
165
166 if (positional.isEmpty()) {
167 // Plain launch — the double-clicked-icon case. Empty canvas, no message.
168 return options;
169 } // if positional.isEmpty()
170
171 if (positional.size() > 1) {
172 // One document per process, deliberately: SeamlyLayout has a single
173 // pair of canvases and no tabs, so a second file has nowhere to go.
174 options.m_status = Status::Failed;
175 options.m_message = QStringLiteral(
176 "SeamlyLayout opens one SVG file at a time, but %1 files were given:\n%2")
177 .arg(positional.size())
178 .arg(positional.join(QStringLiteral("\n")));
179 return options;
180 } // if positional.size() > 1
181
182 // ---------------------------------------------------------------------
183 // Validate the one positional argument.
184 //
185 // Each branch names the file, because a launched process gives the user no
186 // console output to correlate the message with.
187 // ---------------------------------------------------------------------
188 const QFileInfo file(positional.constFirst());
189 // Resolved against the process working directory, which for the Seamly2D
190 // handoff is SeamlyLayout's own install directory — hence absolute only.
191 const QString absolutePath = file.absoluteFilePath();
192
193 if (!file.exists()) {
194 options.m_status = Status::Failed;
195 options.m_message = QStringLiteral("The file does not exist:\n%1").arg(absolutePath);
196 return options;
197 } // if !file.exists()
198
199 if (!file.isFile()) {
200 options.m_status = Status::Failed;
201 options.m_message = QStringLiteral("This is a folder, not an SVG file:\n%1").arg(absolutePath);
202 return options;
203 } // if !file.isFile()
204
205 if (!file.isReadable()) {
206 options.m_status = Status::Failed;
207 options.m_message = QStringLiteral(
208 "The file cannot be read — check its permissions:\n%1").arg(absolutePath);
209 return options;
210 } // if !file.isReadable()
211
212 if (file.suffix().compare(svgSuffix, Qt::CaseInsensitive) != 0) {
213 options.m_status = Status::Failed;
214 options.m_message = QStringLiteral(
215 "SeamlyLayout opens SVG files; this one is not an SVG:\n%1").arg(absolutePath);
216 return options;
217 } // if suffix is not svg
218
219 options.m_status = Status::OpenFile;
220 options.m_filePath = absolutePath;
221 return options;
222} // StartupOptions::parse

224// @brief Read and validate the SVG document offered on standard input.
225//
226// Split out of parse() so the standard-input transport (Seamly2D.5) has one
227// place that decides what a usable document is, and so a test can drive it with
228// a QBuffer instead of the real stdin of the test runner.
229//
230// @param documentName Value of --document-name; empty when it was not given.
231// @param standardInput Device to read to end; nullptr when the caller offers
232// none, which is itself a failure because --svg-stdin was requested.
233// @return OpenDocument with svgDocument() set, or Failed with a message.
234StartupOptions StartupOptions::parseStandardInput(const QString &documentName,
235 QIODevice *standardInput)
236{
237 StartupOptions options;
238
239 if (standardInput == nullptr || !standardInput->isReadable()) {
240 options.m_status = Status::Failed;
241 options.m_message = QStringLiteral(
242 "--%1 was given, but standard input is not readable.").arg(svgStdinOptionName);
243 return options;
244 } // if no readable channel
245
246 // Read to end: Seamly2D writes the whole document, then closes the channel.
247 const QString document = QString::fromUtf8(standardInput->readAll());
248
249 if (document.trimmed().isEmpty()) {
250 options.m_status = Status::Failed;
251 options.m_message = QStringLiteral(
252 "--%1 was given, but standard input was empty.").arg(svgStdinOptionName);
253 return options;
254 } // if nothing arrived
255
256 if (!document.contains(svgRootElement, Qt::CaseInsensitive)) {
257 options.m_status = Status::Failed;
258 options.m_message = QStringLiteral(
259 "The document on standard input is not an SVG — it has no <svg> element.");
260 return options;
261 } // if not an SVG document
262
263 options.m_status = Status::OpenDocument;
264 options.m_svgDocument = document;
265 options.m_documentName = documentName;
266 return options;
267} // StartupOptions::parseStandardInput
