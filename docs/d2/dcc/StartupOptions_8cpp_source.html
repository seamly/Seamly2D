5// @file StartupOptions.cpp
6// @brief Implementation of the SeamlyLayout command-line contract.
7//
8// Control flow of parse():
9// 1. Reject an empty argument list outright (nothing to parse).
10// 2. Run QCommandLineParser::parse() — the non-exiting sibling of process(),
11// chosen so this function can be unit tested and can never terminate the
12// application behind main()'s back.
13// 3. Honour --help / --version by handing their text back as
14// Status::ShowInformation; main() displays it and exits 0.
15// 4. Require at most one positional argument.
16// 5. Validate that argument as an existing, readable .svg file and store its
17// ABSOLUTE path (the app-wide rule: never pass relative paths around, the
18// working directory of a detached launch is not the user's).
19//
20// Every failure produces a sentence the user can act on, because the only thing
21// worse than the old behaviour (silently ignoring the argument) is silently
22// ignoring it with an extra step.
23

34// Seamly2D hands over "<pattern>.pieces.svg", whose QFileInfo::suffix() is "svg".
35const QLatin1String svgSuffix("svg");

37// @brief Build the text shown for --version.
38// Falls back to the product name alone when the caller has not set the
39// application metadata yet (QCoreApplication::setApplicationVersion()).
40// @return e.g. "SeamlyLayout 0.1.0".
41QString versionText()
42{
43 const QString name = QCoreApplication::applicationName();
44 const QString version = QCoreApplication::applicationVersion();
45
46 if (name.isEmpty()) {
47 // No metadata set — name the product so the output is never empty.
48 return version.isEmpty() ? QStringLiteral("SeamlyLayout")
49 : QStringLiteral("SeamlyLayout ") + version;
50 } // if name.isEmpty()
51
52 return version.isEmpty() ? name : name + QLatin1Char(' ') + version;
53} // versionText
54} // namespace

56// @brief Parse and validate a SeamlyLayout command line.
57// @param arguments Full argument list including the program name at index 0,
58// as returned by QCoreApplication::arguments().
59// @return A fully populated StartupOptions; never throws, never exits.
60StartupOptions StartupOptions::parse(const QStringList &arguments)
61{
62 StartupOptions options;
63
64 if (arguments.isEmpty()) {
65 // Not even a program name — nothing to open, and QCommandLineParser
66 // would warn about the missing argv[0].
67 return options;
68 } // if arguments.isEmpty()
69
70 QCommandLineParser parser;
71 parser.setApplicationDescription(QStringLiteral(
72 "SeamlyLayout — pattern layout application of the Seamly2D family.\n"
73 "Opens the tagged pieces SVG that Seamly2D's Layout Mode writes beside "
74 "the pattern file."));
75
76 const QCommandLineOption helpOption = parser.addHelpOption();
77 const QCommandLineOption versionOption = parser.addVersionOption();
78
79 parser.addPositionalArgument(
80 QStringLiteral("svg-file"),
81 QStringLiteral("SVG pattern file to open at startup (optional)."),
82 QStringLiteral("[svg-file]"));
83
84 // parse(), not process(): process() prints to a console this GUI-subsystem
85 // application does not have on Windows, and calls exit() on error.
86 if (!parser.parse(arguments)) {
87 // Unknown option, missing option value, etc. errorText() is already a
88 // complete sentence ("Unknown option 'foo'.").
89 options.m_status = Status::Failed;
90 options.m_message = parser.errorText();
91 return options;
92 } // if !parser.parse
93
94 if (parser.isSet(versionOption)) {
95 // --version wins over everything else, matching every other CLI.
96 options.m_status = Status::ShowInformation;
97 options.m_message = versionText();
98 return options;
99 } // if version requested
100
101 if (parser.isSet(helpOption)) {
102 options.m_status = Status::ShowInformation;
103 options.m_message = parser.helpText();
104 return options;
105 } // if help requested
106
107 const QStringList positional = parser.positionalArguments();
108
109 if (positional.isEmpty()) {
110 // Plain launch — the double-clicked-icon case. Empty canvas, no message.
111 return options;
112 } // if positional.isEmpty()
113
114 if (positional.size() > 1) {
115 // One document per process, deliberately: SeamlyLayout has a single
116 // pair of canvases and no tabs, so a second file has nowhere to go.
117 options.m_status = Status::Failed;
118 options.m_message = QStringLiteral(
119 "SeamlyLayout opens one SVG file at a time, but %1 files were given:\n%2")
120 .arg(positional.size())
121 .arg(positional.join(QStringLiteral("\n")));
122 return options;
123 } // if positional.size() > 1
124
125 // ---------------------------------------------------------------------
126 // Validate the one positional argument.
127 //
128 // Each branch names the file, because a detached launch gives the user no
129 // console output to correlate the message with.
130 // ---------------------------------------------------------------------
131 const QFileInfo file(positional.constFirst());
132 // Resolved against the process working directory, which for the Seamly2D
133 // handoff is SeamlyLayout's own install directory — hence absolute only.
134 const QString absolutePath = file.absoluteFilePath();
135
136 if (!file.exists()) {
137 options.m_status = Status::Failed;
138 options.m_message = QStringLiteral("The file does not exist:\n%1").arg(absolutePath);
139 return options;
140 } // if !file.exists()
141
142 if (!file.isFile()) {
143 options.m_status = Status::Failed;
144 options.m_message = QStringLiteral("This is a folder, not an SVG file:\n%1").arg(absolutePath);
145 return options;
146 } // if !file.isFile()
147
148 if (!file.isReadable()) {
149 options.m_status = Status::Failed;
150 options.m_message = QStringLiteral(
151 "The file cannot be read — check its permissions:\n%1").arg(absolutePath);
152 return options;
153 } // if !file.isReadable()
154
155 if (file.suffix().compare(svgSuffix, Qt::CaseInsensitive) != 0) {
156 options.m_status = Status::Failed;
157 options.m_message = QStringLiteral(
158 "SeamlyLayout opens SVG files; this one is not an SVG:\n%1").arg(absolutePath);
159 return options;
160 } // if suffix is not svg
161
162 options.m_status = Status::OpenFile;
163 options.m_filePath = absolutePath;
164 return options;
165} // StartupOptions::parse
