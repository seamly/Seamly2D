287void SymlinkCopyDirRecursive(const QString &fromDir, const QString &toDir, bool replaceOnConflit)
288{
289 QDir dir;
290 dir.setPath(fromDir);
291
292 foreach (QString copyFile, dir.entryList(QDir::Files))
293 {
294 const QString from = fromDir + QDir::separator() + copyFile;
295 QString to = toDir + QDir::separator() + copyFile;
296
297#ifdef Q_OS_WIN
298 {
299 // To fix issue #702 check each not symlink if it is actually broken symlink.
300 // Also trying to mimic Unix symlink. If a file eaxists do not create a symlink and remove it if exists.
301 QFile fileTo(to);
302 if (fileTo.exists())
303 {
304 if (not fileTo.rename(to + QLatin1String(".lnk")))
305 {
306 QFile::remove(to + QLatin1String(".lnk"));
307 fileTo.rename(to + QLatin1String(".lnk"));
308 }
309
310 QFileInfo info(to + QLatin1String(".lnk"));
311 if (info.symLinkTarget().isEmpty())
312 {
313 fileTo.copy(to);
314 fileTo.remove();
315 continue; // The file already exists, skip creating shortcut
316 }
317 }
318 }
319
320 to = to + QLatin1String(".lnk");
321#endif
322
323 if (QFile::exists(to))
324 {
325 if (replaceOnConflit)
326 {
327 QFile::remove(to);
328 }
329 else
330 {
331 continue;
332 }
333 }
334
335 QFile::link(from, to);
336 }
337
338 foreach (QString copyDir, dir.entryList(QDir::Dirs | QDir::NoDotAndDotDot))
339 {
340 const QString from = fromDir + QDir::separator() + copyDir;
341 const QString to = toDir + QDir::separator() + copyDir;
342
343 if (dir.mkpath(to) == false)
344 {
345 return;
346 }
347
348 SymlinkCopyDirRecursive(from, to, replaceOnConflit);
349 }
350}

473 * @brief commonSettingsFilePath returns the absolute path of the shared qt6_common.ini.
474 *
475 * The file lives under GenericConfigLocation — %LOCALAPPDATA% on Windows, ~/.config on

479 * carry them to a machine where they are wrong. Task SettingsFiles.1.

481 * Every application sets "Seamly", so all of them resolve one file.

495 {
496 organization = QStringLiteral("Seamly");
497 }
498
499 return base + QLatin1Char('/') + organization + QLatin1Char('/') + commonIniFilename
500 + QLatin1String(".ini");
501}

549/**
550 * @brief migrateCommonSettingsLocation brings an existing common settings file forward
551 * into commonSettingsFilePath().
552 *
553 * Candidate sources, newest first; the first one that exists is copied:
554 *
555 * 1. the pre-move file under Qt's IniFormat/UserScope default (%APPDATA%\Seamly on
556 * Windows — identical to the target on Linux and macOS, so skipped there);
557 * 2. the pre-Task-15 "Seamly2DTeam" organization's qt6_common.ini;
558 * 3. the qt5-era common.ini from either of those folders.
559 *
560 * Copy-if-missing and re-entrant: once the target exists, every later call is a cheap
561 * no-op, and no source file is ever modified or deleted — a rollback to an earlier
562 * release keeps working. Called from each application's openSettings() before any
563 * settings value is read.
564 *
565 * @return absolute path of the common settings file at its current location.
566 */
567QString VCommonSettings::migrateCommonSettingsLocation()
568{
569 const QString target = commonSettingsFilePath();
570 QDir().mkpath(QFileInfo(target).absolutePath());
571 if (QFileInfo::exists(target))
572 {
573 return target;
574 }
575
576 // Qt's own IniFormat/UserScope resolution names the pre-move folder, so a
577 // QSettings::setPath() redirection (the test suite) is honoured here too.
578 const QSettings roamingProbe(QSettings::IniFormat, QSettings::UserScope,
579 QCoreApplication::organizationName(), commonIniFilename);
580 const QString roamingDir = QFileInfo(roamingProbe.fileName()).absolutePath();
581
582 static const QString kLegacyOrganizationName = QStringLiteral("Seamly2DTeam");
583 const QSettings legacyProbe(QSettings::IniFormat, QSettings::UserScope,
584 kLegacyOrganizationName, commonIniFilename);
585 const QString legacyDir = QFileInfo(legacyProbe.fileName()).absolutePath();
586
587 const QStringList candidates
588 {
589 roamingDir + QLatin1String("/qt6_common.ini"),
590 legacyDir + QLatin1String("/qt6_common.ini"),
591 roamingDir + QLatin1String("/common.ini"),
592 legacyDir + QLatin1String("/common.ini")
593 };
594
595 for (const QString &candidate : candidates)
596 {
597 if (candidate != target && QFileInfo::exists(candidate))
598 {
599 QFile::copy(candidate, target);
600 break;
601 }
602 }
603
604 return target;
605}

608/**
609 * @brief getDefaultDataRoot returns the built-in default root of the user's data tree.
610 *
611 * Task 60 moved this to <Documents>/Seamly. Two changes, for two reasons:
612 *
613 * - **Documents, not the home directory.** These are files the user creates, opens, saves
614 * and backs up, so they belong where every other application puts documents. Internal
615 * state — settings, caches, logs — stays in the platform's application-data locations
616 * and is deliberately NOT mixed in here.
617 * - **"Seamly", not "seamlyData" or "seamly2d".** The folder holds the whole suite's
618 * work, so naming it after one member (seamly2d) wrongly implies SeamlyMe and
619 * SeamlyLayout belong to Seamly2D, and "Data" is redundant once the parent location

781/**
782 * @brief migrateDataTree copies a whole user-data tree to a new root (Task 60).
783 *
784 * Copies EVERY file and directory found under sourceRoot, not a known list of subfolders.
785 * That is deliberate and is the single most important property of this function: users add
786 * their own directories to the data tree — Projects, bodyscans and others have been seen in
787 * the wild — so migrating a fixed list would silently strand whatever the list did not
788 * mention. The structure is reproduced exactly; only the root's name changes.
789 *
790 * The safety rules, each of which exists because the alternative loses data:
791 *
792 * - **Never a rename or a move.** The source tree is left completely intact so a user can
793 * roll back to an earlier release, which is why the caller can also mark it rather than
794 * delete it. Nothing here removes anything, ever.
795 * - **Merge, never overwrite.** An existing destination file is skipped and counted, not
796 * clobbered — the destination may be an already-populated folder.
797 * - **Verify every copy.** Sizes are compared after each file, because a cloud-synced
798 * target (Google Drive, OneDrive, Dropbox) can report a write complete before it is
799 * durable. A file that does not verify aborts the migration.
800 * - **Fail safe.** On any error the function stops, removes only the partial file it was
801 * writing at that moment, and returns false with the source untouched. A half-copied
802 * destination must never become the configured root, so the caller must not record the
803 * new root unless this returned true.
804 *
805 * @param sourceRoot tree to copy from; must exist.
806 * @param destinationRoot tree to copy to; created if missing.
807 * @param filesCopied optional out-parameter, number of files actually copied.
808 * @param filesSkipped optional out-parameter, number already present at the destination.
809 * @param errorMessage optional out-parameter, human-readable reason for a false return.
810 * @return true when every file is present and verified at the destination.
811 */
812bool VCommonSettings::migrateDataTree(const QString &sourceRoot, const QString &destinationRoot,
813 int *filesCopied, int *filesSkipped, QString *errorMessage)
814{
815 const auto fail = [errorMessage](const QString &reason)
816 {
817 if (errorMessage != nullptr)
818 {
819 *errorMessage = reason;
820 }
821 qWarning() << "Data-tree migration failed:" << reason;
822 return false;
823 };
824
825 if (filesCopied != nullptr) { *filesCopied = 0; }
826 if (filesSkipped != nullptr) { *filesSkipped = 0; }
827 if (errorMessage != nullptr) { errorMessage->clear(); }
828
829 const QString source = QDir::cleanPath(QDir::fromNativeSeparators(sourceRoot.trimmed()));
830 const QString destination = QDir::cleanPath(QDir::fromNativeSeparators(destinationRoot.trimmed()));
831
832 if (source.isEmpty() || destination.isEmpty())
833 {
834 return fail(QStringLiteral("source or destination path is empty"));
835 }
836 if (!QFileInfo(source).isDir())
837 {
838 return fail(QStringLiteral("source '%1' is not a directory").arg(source));
839 }
840#ifdef Q_OS_WIN
841 const Qt::CaseSensitivity caseSensitivity = Qt::CaseInsensitive;
842#else
843 const Qt::CaseSensitivity caseSensitivity = Qt::CaseSensitive;
844#endif
845 if (source.compare(destination, caseSensitivity) == 0)
846 {
847 return fail(QStringLiteral("source and destination are the same directory"));
848 }
849 // Copying a tree into its own subdirectory would recurse without end.
850 if (destination.startsWith(source + QLatin1Char('/'), caseSensitivity))
851 {
852 return fail(QStringLiteral("destination '%1' lies inside the source tree").arg(destination));
853 }
854
855 QDir destinationDir(destination);
856 if (!destinationDir.mkpath(QStringLiteral(".")))
857 {
858 return fail(QStringLiteral("could not create '%1'").arg(destination));
859 }
860
861 const QDir sourceDir(source);
862 int copied = 0;
863 int skipped = 0;
864
865 QDirIterator iterator(source, QDir::Files | QDir::Dirs | QDir::NoDotAndDotDot | QDir::Hidden,
866 QDirIterator::Subdirectories);
867 while (iterator.hasNext())
868 {
869 const QString entryPath = iterator.next();
870 const QFileInfo entry(entryPath);
871 const QString relative = sourceDir.relativeFilePath(entryPath);
872 const QString target = destination + QLatin1Char('/') + relative;
873
874 if (entry.isDir())
875 {
876 if (!destinationDir.mkpath(relative))
877 {
878 return fail(QStringLiteral("could not create '%1'").arg(target));
879 }
880 continue;
881 }
882
883 if (QFileInfo::exists(target))
884 {
885 // Merge, never overwrite. Reported so a collision is visible rather than silent.
886 ++skipped;
887 qDebug() << "Data-tree migration skipped existing file" << QDir::toNativeSeparators(target);
888 continue;
889 }
890
891 // The parent may not exist yet: QDirIterator does not guarantee a directory is
892 // visited before the files inside it.
893 const QString targetParent = QFileInfo(target).absolutePath();
894 if (!QDir().mkpath(targetParent))
895 {
896 return fail(QStringLiteral("could not create '%1'").arg(targetParent));
897 }
898
899 if (!QFile::copy(entryPath, target))
900 {
901 return fail(QStringLiteral("could not copy '%1' to '%2'").arg(entryPath, target));
902 }
903
904 // Verify, because a cloud-synced destination can report success early.
905 if (QFileInfo(target).size() != entry.size())
906 {
907 QFile::remove(target);
908 return fail(QStringLiteral("copy of '%1' did not verify (expected %2 bytes)")
909 .arg(entryPath)
910 .arg(entry.size()));
911 }
912 ++copied;
913 }
914
915 if (filesCopied != nullptr) { *filesCopied = copied; }
916 if (filesSkipped != nullptr) { *filesSkipped = skipped; }
917 return true;
918}

921/**
922 * @brief migrateAdoptedLegacyTree turns a first-run adoption into a Task 60 migration.
923 *
924 * initializeDataRoot() still *adopts* a legacy tree — it resolves and records a path and
925 * touches no files, which is what keeps it safe for the unit tests to call. This function
926 * is the second half, and it is deliberately called only from the applications'
927 * openSettings(), the one place the real home directory is fed in. The tests therefore
928 * cannot copy anything into the developer's home no matter what they resolve, which is the
929 * same rule pruneEmptyLegacyDataRoot() and ensureDataRootTree() follow.
930 *
931 * Fail-safe by construction: the configured root is only repointed at newRoot after the
932 * copy has completed and verified. If anything goes wrong the legacy tree stays configured
933 * and in use, so the worst case is that the user carries on exactly as before.
934 *
935 * @param legacyRoot the adopted tree, e.g. ~/seamly2d.
936 * @param newRoot where it should live now, e.g. <Documents>/Seamly.
937 * @return the root actually in force afterwards — newRoot on success, legacyRoot on failure.
938 */
939QString VCommonSettings::migrateAdoptedLegacyTree(const QString &legacyRoot, const QString &newRoot)
940{
941 if (legacyRoot.isEmpty() || newRoot.isEmpty() || !QFileInfo(legacyRoot).isDir())
942 {
943 return legacyRoot;
944 }
945
946 // Already dealt with on an earlier run: leave the marked tree alone.
947 if (dataTreeWasMigrated(legacyRoot))
948 {
949 return legacyRoot;
950 }
951
952 int copied = 0;
953 int skipped = 0;
954 QString errorMessage;
955 if (!migrateDataTree(legacyRoot, newRoot, &copied, &skipped, &errorMessage))
956 {
957 qWarning() << "Keeping the existing data root" << QDir::toNativeSeparators(legacyRoot)
958 << "because migration failed:" << errorMessage;
959 return legacyRoot;
960 }
961
962 qInfo() << "Migrated the user-data tree from" << QDir::toNativeSeparators(legacyRoot) << "to"
963 << QDir::toNativeSeparators(newRoot) << '-' << copied << "file(s) copied," << skipped
964 << "already present";
965
966 // Only now is it safe to repoint the configured root.
967 QSettings settings(commonSettingsFilePath(), QSettings::IniFormat);
968 settings.setValue(settingPathsDataRoot, newRoot);
969 settings.sync();
970
971 markDataTreeMigrated(legacyRoot, newRoot);
972 return newRoot;
973}

1035#else
1036 const Qt::CaseSensitivity caseSensitivity = Qt::CaseSensitive;
1037#endif
1038

1040 const QString cleanNew = QDir::cleanPath(QDir::fromNativeSeparators(newRoot.trimmed()));
1041

1062/**
1063 * @brief initializeDataRoot resolves the user-data root once, at application start-up.
1064 *
1065 * Called before any data path is read, from every application's openSettings(). Four cases:
1066 *
1067 * 1. A root is already configured — honour it untouched.
1068 * 2. Nothing configured, and the Windows installer recorded one — adopt it. The user
1069 * chose that folder on Setup's "Where do you keep your work?" page and was told the
1070 * apps would use it, so it outranks every default below.
1071 * 3. Nothing configured or recorded, and a populated legacy ~/seamly2d tree exists while
1072 * the default root does not — adopt the legacy tree *in place* as the root. Adoption
1073 * rather than copying is deliberate: an upgrading user's patterns and measurements can
1074 * be many gigabytes and may sit on a cloud-synced drive, so nothing is moved, copied
1075 * or deleted and the data keeps working from the moment the app starts.
1076 * 4. Otherwise — a fresh install — use the built-in default.
1077 *
1078 * The resolved root is written back so later runs take case 1 and the value is visible to
1079 * the other applications and to Preferences → Paths.
1080 *
1081 * Cases 2–4 are DEPRECATED first-run seeding (Task SettingsFiles.3, 2026-08-31). The
1082 * Windows MSI seeds paths/dataRoot at install time (smsi_seed_user_settings.ps1), so an
1083 * installed Windows machine takes case 1. The fallbacks stay only for packages with no
1084 * install hook — the macOS dmg, the Linux AppImage, dev builds — and for other Windows
1085 * accounts on a shared machine. Remove them when those packages gain install-time seeding.
1086 *
1087 * @param adoptedLegacyTree optional out-parameter, set to true when case 3 applied; pass
1088 * null when the caller does not care.
1089 * @return absolute path of the resolved user-data root.
1090 */
1091QString VCommonSettings::initializeDataRoot(bool *adoptedLegacyTree)
1092{
1093 if (adoptedLegacyTree != nullptr)
1094 {
1095 *adoptedLegacyTree = false;
1096 }
1097
1098 QSettings settings(commonSettingsFilePath(), QSettings::IniFormat);
1099
1100 const QString configured = settings.value(settingPathsDataRoot).toString().trimmed();
1101 if (!configured.isEmpty())
1102 {
1103 // Case 1: already chosen, by an earlier run or by the user.
1104 return QDir::cleanPath(QDir::fromNativeSeparators(configured));
1105 }
1106
1107 // Case 2: the Windows installer recorded the folder the user chose. It outranks both the
1108 // legacy tree and the built-in default, because the user was shown that path and told the
1109 // apps would use it. Recorded here, so later runs take case 1 and a change made in
1110 // Preferences is never overridden by the installer.
1111 const QString fromInstaller = InstallerRecord::dataRoot();
1112 if (!fromInstaller.isEmpty())
1113 {
1114 settings.setValue(settingPathsDataRoot, fromInstaller);
1115 settings.sync();
1116 return fromInstaller;
1117 }
1118
1119 const QString resolved = chooseFirstRunDataRoot(getDefaultDataRoot(), getLegacyDataRoot(), adoptedLegacyTree);
1120
1121 settings.setValue(settingPathsDataRoot, resolved);
1122 settings.sync();
1123
1124 return resolved;
1125}

1165/**
1166 * @brief pruneEmptyLegacyDataRoot removes the abandoned legacy data root when it holds no files.
1167 *
1168 * Renaming the default root leaves the old ~/seamly2d behind, and ensureDataRootTree() will
1169 * have stocked it with the nine standard subfolders, so what remains after the move is an
1170 * empty skeleton that looks like data but is not. This deletes that skeleton.
1171 *
1172 * Two conditions gate it, and both matter:
1173 *
1174 * - the legacy root must not be the configured root. Task 34's first-run rule *adopts* an
1175 * existing ~/seamly2d in place, so for an upgrading user that directory is the live data
1176 * tree. Deleting it there would destroy exactly the patterns adoption set out to preserve.
1177 * - the tree must contain no files at any depth. One stray file and nothing is removed.
1178 *
1179 * Only empty directories are then removed, deepest first, via QDir::rmdir() — which cannot
1180 * delete a file and refuses a non-empty directory. removeRecursively() is never used: this
1181 * function must not be capable of deleting anything it has not counted.
1182 *
1183 * @param legacyRoot the legacy root to prune, normally getLegacyDataRoot().
1184 * @param configuredRoot the data root actually in use; pruning is skipped when they match.
1185 * @return true when the legacy root was removed, false when it was kept for any reason.
1186 */
1187bool VCommonSettings::pruneEmptyLegacyDataRoot(const QString &legacyRoot, const QString &configuredRoot)
1188{
1189 // Windows path comparison is case-insensitive; POSIX filesystems are not.
1190#ifdef Q_OS_WIN
1191 const Qt::CaseSensitivity caseSensitivity = Qt::CaseInsensitive;
1192#else
1193 const Qt::CaseSensitivity caseSensitivity = Qt::CaseSensitive;
1194#endif
1195
1196 const QString legacy = QDir::cleanPath(QDir::fromNativeSeparators(legacyRoot.trimmed()));
1197 const QString configured = QDir::cleanPath(QDir::fromNativeSeparators(configuredRoot.trimmed()));
1198
1199 if (legacy.isEmpty() || !QFileInfo(legacy).isDir())
1200 {
1201 return false;
1202 }
1203
1204 // The live data tree of an upgrading user — never touch it.
1205 if (legacy.compare(configured, caseSensitivity) == 0)
1206 {
1207 return false;
1208 }
1209
1210 // A configured root *inside* the legacy root (e.g. ~/seamly2d/patterns) would be taken
1211 // down with its parent, so treat that as occupied too.
1212 if (configured.startsWith(legacy + QLatin1Char('/'), caseSensitivity))
1213 {
1214 return false;
1215 }
1216
1217 QDirIterator files(legacy, QDir::Files | QDir::Hidden | QDir::System | QDir::NoDotAndDotDot,
1218 QDirIterator::Subdirectories);
1219 if (files.hasNext())
1220 {
1221 return false;
1222 }
1223
1224 // Deepest first, so each rmdir() sees an already-emptied directory.
1225 QStringList directories;
1226 QDirIterator subdirectories(legacy, QDir::Dirs | QDir::Hidden | QDir::System | QDir::NoDotAndDotDot,
1227 QDirIterator::Subdirectories);
1228 while (subdirectories.hasNext())
1229 {
1230 directories.append(subdirectories.next());
1231 }
1232
1233 std::sort(directories.begin(), directories.end(),
1234 [](const QString &first, const QString &second) { return first.length() > second.length(); });
1235
1236 for (const QString &directory : qAsConst(directories))
1237 {
1238 QDir().rmdir(directory);
1239 }
1240
1241 if (!QDir().rmdir(legacy))
1242 {
1243 qWarning() << "Could not remove the empty legacy data root" << QDir::toNativeSeparators(legacy);
1244 return false;
1245 }
1246
1247 return true;
1248}
