45/**
46 * @brief createDummyFile creates an empty placeholder file (parent directories
47 * included) standing in for an executable in the lookup tests.
48 *
49 * The lookup only checks existence and file-ness, never content or the
50 * executable bit, so an empty file is a faithful stand-in on every platform.
51 *
52 * @param filePath absolute path of the file to create.
53 * @return true when the file (and any missing parent directory) was created.
54 */
55bool createDummyFile(const QString &filePath)
56{
57 // QFile::open() does not create missing parent directories - mirror the
58 // real installer layout by creating them first.
59 if (!QDir().mkpath(QFileInfo(filePath).absolutePath()))
60 {
61 return false;
62 }
63 QFile file(filePath);
64 if (!file.open(QIODevice::WriteOnly))
65 {
66 return false;
67 }
68 file.close();
69 return true;
70}

136/**
137 * @brief FlatLayoutTakesPrecedence verifies that when both layouts are
138 * present the flat executable wins — it is the one sharing the caller's
139 * runtime, so it must shadow a leftover subdirectory install.
140 *
141 * This scenario is only physically constructible on Windows: there the flat
142 * executable carries the ".exe" suffix (`SeamlyLayout.exe`) that distinguishes
143 * it from the MSI subdirectory (`SeamlyLayout\`), so a flat exe file and a
144 * `SeamlyLayout` subdirectory can share one parent. On every other platform
145 * the executable name has no suffix, so the flat candidate (`SeamlyLayout`, a
146 * file) and the subdirectory (`SeamlyLayout`, a directory) have identical names
147 * and cannot coexist in one directory — the two layouts are mutually exclusive,
148 * so precedence never arises. That mutually-exclusive non-Windows case is
149 * covered instead by DirectoryNamedLikeExecutableIsIgnored().
150 */
151void TST_SeamlyFamilyPaths::FlatLayoutTakesPrecedence() const
152{
153#ifndef Q_OS_WIN
154 QSKIP("Both layouts can coexist only on Windows (the flat exe's \".exe\" "
155 "suffix distinguishes it from the \"SeamlyLayout\" subdirectory); "
156 "elsewhere they are mutually exclusive — see "
157 "DirectoryNamedLikeExecutableIsIgnored().");
158#else
159 QTemporaryDir dir;
160 QVERIFY(dir.isValid());
161
162 const QString flatExe = dir.path() + QLatin1Char('/')
163 + SeamlyFamilyPaths::seamlyLayoutExeName();
164 const QString nestedExe = dir.path() + QLatin1String("/SeamlyLayout/")
165 + SeamlyFamilyPaths::seamlyLayoutExeName();
166 QVERIFY(createDummyFile(flatExe));
167 QVERIFY(createDummyFile(nestedExe));
168
169 const QString found = SeamlyFamilyPaths::locateSeamlyLayout(dir.path());
170 QCOMPARE(found, QFileInfo(flatExe).absoluteFilePath());
171#endif
172}

175/**
176 * @brief DirectoryNamedLikeExecutableIsIgnored verifies a *directory* named
177 * like the executable never counts as a match. On non-Windows platforms the
178 * executable name has no ".exe" suffix, so the MSI-style "SeamlyLayout"
179 * subdirectory itself collides with the flat candidate's name — the isFile()
180 * guard must reject it and let the subdirectory lookup succeed instead.
181 */
182void TST_SeamlyFamilyPaths::DirectoryNamedLikeExecutableIsIgnored() const
183{
184 QTemporaryDir dir;
185 QVERIFY(dir.isValid());
186
187 // A directory whose name matches the flat executable candidate exactly.
188 QVERIFY(QDir(dir.path()).mkpath(SeamlyFamilyPaths::seamlyLayoutExeName()));
189
190 // With nothing else present the lookup must find nothing...
191 QVERIFY(SeamlyFamilyPaths::locateSeamlyLayout(dir.path()).isEmpty());
192
193 // ...and with the real executable inside the MSI-style subdirectory, the
194 // lookup must skip the impostor directory and return the nested file.
195 const QString nestedExe = dir.path() + QLatin1String("/SeamlyLayout/")
196 + SeamlyFamilyPaths::seamlyLayoutExeName();
197 QVERIFY(createDummyFile(nestedExe));
198 QCOMPARE(SeamlyFamilyPaths::locateSeamlyLayout(dir.path()),
199 QFileInfo(nestedExe).absoluteFilePath());
200}

380 * @brief DevBuildStopsBeforeUnboundedWalk verifies the walk is bounded: a
381 * checkout further above the start directory than the depth limit is not found.
382 *
383 * Without the bound the lookup would keep climbing to the filesystem root,
384 * probing directories that have nothing to do with the application.
385 */
386void TST_SeamlyFamilyPaths::DevBuildStopsBeforeUnboundedWalk() const
387{
388 QTemporaryDir dir;
389 QVERIFY(dir.isValid());
390
391 const QString checkout = dir.path();
392
393 // Ten levels below the checkout root — beyond the eight-level limit.
394 const QString deepDirectory =
395 checkout + QLatin1String("/a/b/c/d/e/f/g/h/i/j");
396 QVERIFY(QDir().mkpath(deepDirectory));
397
398 const QString layoutExe = checkout
399 + QLatin1String("/src/app/seamlylayout/qt_frontend/build/Release/")
400 + SeamlyFamilyPaths::seamlyLayoutExeName();
401 QVERIFY(createDummyFile(layoutExe));
402
403 QVERIFY(SeamlyFamilyPaths::locateSeamlyLayoutDevBuild(deepDirectory).isEmpty());
404
405 // Sanity check that the same tree *is* found from within the limit, so the
406 // case above fails for the depth bound and not for a broken fixture.
407 const QString shallowDirectory = checkout + QLatin1String("/a/b/c");
408 QCOMPARE(SeamlyFamilyPaths::locateSeamlyLayoutDevBuild(shallowDirectory),
409 QFileInfo(layoutExe).absoluteFilePath());
410}
