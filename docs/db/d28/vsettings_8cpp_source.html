116/**
117 * @brief copySampleFiles copies files matching nameFilter from sourceDir into destinationDir.
118 *
119 * Shared by SeedSamplePatterns() and SeedSampleMeasurements(): both need the same
120 * missing-source-is-a-no-op and never-overwrite-an-existing-file behaviour, only the file
121 * extension differs.
122 *
123 * @param sourceDir bundled samples folder; a missing folder is a silent no-op.
124 * @param destinationDir the user's folder to seed; created if missing.
125 * @param nameFilter glob filter selecting which bundled files to copy, e.g. "*.sm2d".
126 * @return number of sample files copied.
127 */
128int copySampleFiles(const QString &sourceDir, const QString &destinationDir, const QString &nameFilter)
129{
130 const QDir source(sourceDir);
131 if (!source.exists())
132 {
133 return 0;
134 }
135
136 QDir destination(destinationDir);
137 if (!destination.mkpath(QStringLiteral(".")))
138 {
139 qWarning() << "Could not create the folder" << QDir::toNativeSeparators(destinationDir);
140 return 0;
141 }
142
143 int copied = 0;
144 const QStringList sampleFiles = source.entryList(QStringList(nameFilter), QDir::Files);
145 for (const QString &fileName : sampleFiles)
146 {
147 const QString target = destination.filePath(fileName);
148 if (QFileInfo::exists(target))
149 {
150 continue;
151 }
152
153 if (QFile::copy(source.filePath(fileName), target))
154 {
155 ++copied;
156 }
157 else
158 {
159 qWarning() << "Could not copy sample file" << QDir::toNativeSeparators(source.filePath(fileName))
160 << "to" << QDir::toNativeSeparators(target);
161 }
162 }
163 return copied;
164}
165}

219/**
220 * @brief SeedSamplePatterns copies the bundled sample patterns into the user's patterns folder.
221 *
222 * A standard user has no write access to the Program Files installation folder, so the
223 * bundled samples under sourceDir can be opened but never saved back in place. Copying them
224 * into the writable destinationDir gives the user an editable starting point instead.
225 *
226 * Purely additive and safe to call on every launch: an existing file at the destination is
227 * left untouched, so a sample the user deleted stays deleted and an edited copy is never
228 * overwritten by the original.
229 *
230 * @param sourceDir bundled samples/patterns folder; a missing folder is a silent no-op.
231 * @param destinationDir the user's patterns folder; created if missing.
232 * @return number of sample files copied.
233 */
234int VSettings::SeedSamplePatterns(const QString &sourceDir, const QString &destinationDir)
235{
236 return copySampleFiles(sourceDir, destinationDir, QStringLiteral("*.sm2d"));
237}

779/**
780 * @brief VSettings::GetTiledPDFMargins returns the tiled pdf margins in the given unit. When the setting is
781 * called for the first time, the 4 default margins are 10mm.
782 * @param unit
783 * @return
784 */
785QMarginsF VSettings::GetTiledPDFMargins(const Unit &unit) const
786{
787 // default value is 10mm. We save the margins in mm in the setting.
788 const QMarginsF def = QMarginsF(10,10,10,10);
789
790 const QVariant val = value(settingTiledPDFMargins, QVariant::fromValue(def));
791
792 if (val.canConvert<QMarginsF>())
793 {
794 return UnitConvertor(val.value<QMarginsF>(), Unit::Mm, unit);
795 }
796 return UnitConvertor(def, Unit::Mm, unit);
797}
